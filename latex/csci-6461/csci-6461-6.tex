\begin{frame}{High-Level Function Execution Model}
\large
\begin{itemize}\setlength{\itemsep}{7pt}
  \item \key{Subroutines} provide modular abstraction, reusability, and clean software architecture.
  \item To execute a function call, the CPU must satisfy five core requirements:
  \begin{enumerate}\setlength{\itemsep}{3pt}
    \item \key{Pass Arguments:} Transfer input parameters to the function.
    \item \key{Transfer Control:} Jump to the function's entry point address.
    \item \key{Isolate Local State:} Provide memory space for function local variables.
    \item \key{Return Value:} Pass computed results back to the caller.
    \item \key{Resume Execution:} Return seamlessly to the instruction following the call.
  \end{enumerate}
\end{itemize}
\end{frame}

% ============================================================
% Slide 4
% ============================================================

\begin{frame}{The Link Register (\texttt{x30}) and Branch-with-Link (\texttt{bl})}

\begin{itemize}\setlength{\itemsep}{6pt}
  \item Invoking a function requires remembering where to return when it completes.
  \item \key{Branch with Link (\code{bl label}):} Atomically performs two operations:
  \begin{enumerate}\setlength{\itemsep}{2pt}
    \item Saves return address ($\text{PC} + 4$) into the \key{Link Register (\bits{x30} / \bits{lr})}.
    \item Branches to target function: $\text{PC} \leftarrow \text{label}$.
  \end{enumerate}
\end{itemize}

\vspace{0.2cm}
\centering
\begin{tikzpicture}[scale=0.9, every node/.style={transform shape}]
  \node[draw=navy, fill=lightblue, rounded corners=2pt, minimum width=3.8cm, minimum height=0.6cm, font=\small\bfseries] (save) {1. $x_{30} \leftarrow \text{PC} + 4$};
  \node[draw=gold!80!black, fill=gold!20, line width=1pt, rounded corners=2pt, right=1.2cm of save, minimum width=3.2cm, minimum height=0.6cm, font=\small\bfseries] (jump) {2. $\text{PC} \leftarrow \text{Target Address}$};
  \draw[-{Latex}, thick, navy] (save) -- (jump);
\end{tikzpicture}

\vspace{0.3cm}
\begin{block}{PC-Relative Calling Range}
The \code{bl} instruction encodes a signed 26-bit immediate offset, allowing direct function calls within a $\pm 128\text{ MB}$ window of the current instruction.
\end{block}
\end{frame}

% ============================================================
% Slide 5
% ============================================================

\begin{frame}{Subroutine Return Mechanics (\texttt{ret})}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item \key{Return from Subroutine (\code{ret}):} Redirects execution back to the caller by loading the target address from the Link Register into the Program Counter:
  $$\text{PC} \leftarrow x_{30}$$
  \item Execution resumes at the exact instruction immediately following the original \code{bl}.
\end{itemize}

\vspace{0.2cm}
\begin{block}{Example: Minimal Leaf Return Sequence}
\small
\begin{tabular}{ll}
\code{caller\_func:} & \\
\quad \code{bl  helper\_func}  & \text{\scriptsize // x30 = address of next instruction; PC = helper\_func} \\
\quad \code{mov x1, x0}        & \text{\scriptsize // Execution resumes here after ret!} \\
\code{helper\_func:} & \\
\quad \code{add x0, x0, \#1}  & \text{\scriptsize // Compute result} \\
\quad \code{ret}              & \text{\scriptsize // PC = x30} \\
\end{tabular}
\end{block}
\end{frame}

% ============================================================
% Slide 6
% ============================================================

\begin{frame}{Leaf vs.\ Non-Leaf Functions}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{block}{Leaf Functions}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Calls \key{no other functions}.
  \item Link Register (\bits{x30}) is not overwritten by a nested call.
  \item May still utilize stack space for local variables or register spills if required, but does not need to save \bits{x30} solely for linkage.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Non-Leaf Functions}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Calls one or more \key{nested subroutines}.
  \item Executing a nested \code{bl} \key{overwrites \bits{x30}} with the nested return address.
  \item Must preserve its incoming return address before making another call if it will need it later (though not strictly required to save it immediately upon entry if no calls happen early).
\end{itemize}
\end{block}
\end{column}
\end{columns}

\vspace{0.3cm}
\normalsize
\begin{block}{The Non-Leaf Rule}
If a function invokes other subroutines and needs to return to its caller afterwards, its incoming return address in \bits{x30} must be preserved before being overwritten.
\end{block}
\end{frame}

% ============================================================
% Slide 7
% ============================================================

\begin{frame}{Indirect Subroutine Calls (\texttt{blr})}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item When the target function address is determined dynamically at runtime, use \key{Branch with Link to Register (\code{blr xn})}.
  \item Saves return address ($\text{PC} + 4$) into \bits{x30} and jumps to address held in register \bits{xn}.
\end{itemize}

\vspace{0.2cm}
\begin{block}{Primary Applications}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item \key{C Function Pointers:} \code{void (*func\_ptr)(int) = my\_callback;}
  \item \key{Virtual Method Tables (vtables):} C++ polymorphism and object dispatch.
  \item \key{Shared Library Veneers:} Dynamic linking and Symbol Jump Tables.
\end{itemize}
\end{block}

\vspace{0.2cm}
\small
\code{ldr x2, [x0, \#8]} \quad \text{\scriptsize // Load function pointer from vtable} \\
\code{blr x2}             \quad \text{\scriptsize // Call function indirectly via register x2}
\end{frame}

\sectiondivider{PART 2}{Procedure Call Standard \& Register Roles}

% ============================================================
% Slide 9

\begin{frame}{Purpose of Procedure Call Standards (AAPCS64)}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item \key{AAPCS64:} Procedure Call Standard for the ARM 64-bit Architecture.
  \item Defines a strict software contract between caller and callee functions.
  \item \key{Why Is an ABI Contract Mandatory?}
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item Allows assembly code to seamlessly call C library functions (\code{printf}, \code{malloc}).
    \item Enables C compilers (GCC, Clang) to interoperate with assembly routines.
    \item Dictates parameter passing, return values, and register preservation rules.
  \end{itemize}
\end{itemize}
\end{frame}

% ============================================================
% Slide 10
% ============================================================

\begin{frame}{Parameter Passing Registers (\texttt{x0}--\texttt{x7})}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item Up to \key{8 integer or pointer parameters} are passed directly in registers \bits{x0} through \bits{x7}.
\end{itemize}

\vspace{0.15cm}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{cll}
\toprule
\textbf{Parameter Index} & \textbf{Register} & \textbf{C Language Example} \\
\midrule
1st Argument & \bits{x0} (\bits{w0}) & \code{int64\_t a} \\
2nd Argument & \bits{x1} (\bits{w1}) & \code{char *str} \\
3rd Argument & \bits{x2} (\bits{w2}) & \code{size\_t length} \\
4th--8th Arguments & \bits{x3}--\bits{x7} & Additional scalar arguments \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.2cm}
\normalsize
\begin{block}{Overflow Arguments ($> 8$ Parameters)}
Parameters beyond the 8th argument are allocated on the runtime stack according to AAPCS64 argument layout rules rather than in reverse order.
\end{block}
\end{frame}

% ============================================================
% Slide 11
% ============================================================

\begin{frame}{Return Value Registers}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item Functions return scalar results in registers without accessing memory:
\end{itemize}

\vspace{0.15cm}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{lll}
\toprule
\textbf{Return Data Type} & \textbf{Register Used} & \textbf{Description} \\
\midrule
32-bit Integer / Boolean & \bits{w0} & Lower 32 bits of register 0 \\
64-bit Integer / Pointer & \bits{x0} & Full 64-bit register 0 \\
Small Aggregates / Structs & \bits{x0} (and potentially \bits{x1}) & Some small integer-like aggregates may be returned in \bits{x0}/\bits{x1} \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.25cm}
\normalsize
\begin{block}{Indirect Aggregate Returns}
Aggregates not eligible for register return are returned through the indirect-result pointer in \bits{x8}.
\end{block}
\end{frame}

% ============================================================
% Slide 12
% ============================================================

\begin{frame}{Caller-Saved Registers (\texttt{x0}--\texttt{x17})}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item Also known as \key{Scratch} or \key{Volatile} registers.
  \item The callee function is free to overwrite these registers \key{without restoring them}.
\end{itemize}

\vspace{0.15cm}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{lll}
\toprule
\textbf{Register Range} & \textbf{AAPCS64 Role} & \textbf{Callee Behavior} \\
\midrule
\bits{x0}--\bits{x7}    & Parameter passing / Return values & May overwrite freely \\
\bits{x8}               & Indirect result location pointer  & May overwrite freely \\
\bits{x9}--\bits{x15}   & Temporary caller scratch registers & May overwrite freely \\
\bits{x16}--\bits{x17} & Intra-procedure linker scratch (IP0, IP1) & May overwrite freely \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.2cm}
\begin{block}{Rule for Caller Functions}
If a caller needs to preserve values in \bits{x0}--\bits{x17} across a subroutine call (\code{bl}), it must preserve them (e.g., by saving them to the stack or moving them to callee-saved registers); stack storage is only one option.
\end{block}
\end{frame}

% ============================================================
% Slide 13
% ============================================================

\begin{frame}{Callee-Saved Registers (\texttt{x19}--\texttt{x28})}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item Also known as \key{Preserved} or \key{Non-Volatile} registers.
  \item Registers \bits{x19} through \bits{x28} hold long-lived variables across subroutine invocations.
\end{itemize}

\vspace{0.15cm}
\begin{block}{The Callee Contract}
If a subroutine modifies any register in \bits{x19}--\bits{x28}, its incoming value must be preserved and restored before return.
\end{block}

\vspace{0.25cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{block}{Typical Preservation (Prologue)}
\code{str x19, [sp, \#-16]!} \\
\text{\scriptsize Preserves caller's x19 before modifying it.}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Typical Restoration (Epilogue)}
\code{ldr x19, [sp], \#16} \\
\text{\scriptsize Restores original x19 before ret.}
\end{block}
\end{column}
\end{columns}
\end{frame}

% ============================================================
% Slide 14
% ============================================================

\begin{frame}{Special-Purpose ABI Registers}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{lll}
\toprule
\textbf{Register} & \textbf{ABI Name} & \textbf{AAPCS64 / ABI Role} \\
\midrule
\bits{x8}        & XR & Indirect result location pointer for struct returns \\
\bits{x16}, \bits{x17} & IP0, IP1 & Intra-procedure call temporaries (used by dynamic linker) \\
\bits{x18}       & PR & Platform register (availability depends on platform ABI rules) \\
\bits{x29}       & FP & Frame Pointer (conventionally used as frame pointer when a frame record is maintained) \\
\bits{x30}       & LR & Link Register (holds function return address) \\
\bits{sp}        & SP & Stack Pointer (holds current stack top; 16-byte aligned) \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.25cm}
\normalsize
\key{Takeaway:} Avoid using \bits{x18}, \bits{x29}, and \bits{x30} as general scratch registers unless the applicable ABI/platform rules and linkage requirements permit it.
\end{frame}

% ============================================================

\begin{frame}{Register Responsibility Summary}
\begin{center}
\small
\setlength{\tabcolsep}{5pt}
\begin{tabular}{l>{\raggedright\arraybackslash}p{4.2cm}>{\raggedright\arraybackslash}p{4.5cm}}
\toprule
\textbf{Registers} & \textbf{Role / Usage} & \textbf{Preservation Contract} \\
\midrule
\bits{x0}--\bits{x7}    & Parameters \& Return Values & \key{Caller-saved} (Callee can overwrite) \\[0.2em]
\bits{x8}--\bits{x17}   & Scratch / Temporaries      & \key{Caller-saved} (Callee can overwrite) \\[0.2em]
\bits{x19}--\bits{x28} & Preserved Local Variables  & \key{Callee-saved} (Callee must restore) \\[0.2em]
\bits{x29} (\bits{fp}) & Frame Pointer              & \key{Callee-saved} (Callee must restore) \\[0.2em]
\bits{x30} (\bits{lr}) & Link Register              & \key{Linkage} (saved by caller or function if needed across calls) \\[0.2em]
\bits{sp}              & Stack Pointer              & \key{Preserved} (Must balance exactly) \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\sectiondivider{PART 3}{The Runtime Stack \& Activation Frames}

% ============================================================

\begin{frame}{The Stack Memory Model}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item The runtime stack is a dynamic memory region allocated in process address space.
  \item \key{Key Characteristics:}
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item Grows \textbf{downward} from high memory addresses toward low memory addresses.
    \item Stack Pointer (\bits{sp}) points to the current \key{lowest active address} (top of stack).
    \item Operates on a Last-In, First-Out (LIFO) order.
  \end{itemize}
\end{itemize}

\vspace{0.25cm}
\begin{block}{Allocation vs.\ Deallocation}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item \key{Push / Allocate:} Subtract from \bits{sp} (\code{sub sp, sp, \#32}).
  \item \key{Pop / Deallocate:} Add to \bits{sp} (\code{add sp, sp, \#32}).
\end{itemize}
\end{block}
\end{frame}

% ============================================================
% Slide 17
% ============================================================

\begin{frame}{Frame Pointer (\texttt{x29}) and Frame Records}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item The \key{Frame Pointer (\bits{x29} / \bits{fp})} can provide a fixed reference point within the activation frame for accessing arguments and local variables.
  \item \key{Frame Pointer Conventions:}
  \begin{itemize}\setlength{\itemsep}{2pt}
    \item Under standard AAPCS64 profiles, functions may maintain a frame record linking saved \bits{x29} and \bits{x30}.
    \item When used, this creates a traceable chain of stack frames to assist debuggers (e.g., GDB) with backtraces.
    \item However, platform rules or optimization levels may permit omitting frame records or repurposing \bits{x29}.
  \end{itemize}
\end{itemize}

\begin{frame}{Example Activation Frame Structure}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.52\textwidth}
\begin{itemize}\setlength{\itemsep}{5pt}
  \item An \key{Activation Frame} stores data necessary for a function execution instance.
  \item Common elements in standard frames include:
  \begin{enumerate}\setlength{\itemsep}{2pt}
    \item Saved Link Register (\bits{x30}) and Frame Pointer (\bits{x29}).
    \item Saved Callee-Saved Registers (\bits{x19}--\bits{x28}).
    \item Local Variables \& Buffers.
  \end{enumerate}
\end{itemize}
\end{column}

\begin{column}{0.44\textwidth}
\centering
\begin{tikzpicture}[scale=0.75, every node/.style={transform shape}]

\begin{frame}[fragile]{Function Prologue Mechanics}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item A \key{prologue} is setup code commonly executed near function entry to set up stack space and saved registers.
  \item Typical actions include:
\end{itemize}

\begin{block}{Standard Prologue Template}
\small
\begin{verbatim}
my_func:
    // 1. Push FP and LR; allocate stack frame (32 bytes)
    stp    x29, x30, [sp, #-32]!

    // 2. Set Frame Pointer to bottom of frame link
    mov    x29, sp

    // 3. Save any callee-saved registers used in body
    str    x19, [sp, #16]
\end{verbatim}
\end{block}

\vspace{0.2cm}
\normalsize
Pre-indexed \code{stp} combines stack allocation and saving \bits{x29}/\bits{x30} into a single instruction.
\end{frame}

% ============================================================
% Slide 20
% ============================================================

\begin{frame}[fragile]{Function Epilogue Mechanics}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item The \key{Epilogue} tears down the frame and restores state before returning.
  \item Reverses the exact sequence performed by the prologue:
\end{itemize}

\vspace{0.15cm}
\begin{block}{Standard Epilogue Template}
\small
\begin{verbatim}
    // 1. Restore saved callee registers
    ldr    x19, [sp, #16]

    // 2. Restore original FP and LR; release stack frame
    ldp    x29, x30, [sp], #32

    // 3. Return to caller
    ret
\end{verbatim}
\end{block}

\vspace{0.2cm}
\normalsize
Post-indexed \code{ldp} performs a paired architectural load and writeback in one instruction to restore \bits{x29}/\bits{x30} and deallocate 32 stack bytes.
\end{frame}

Local stack variables are referenced using positive offsets from \bits{sp} or \bits{x29}.
\end{frame}

\begin{frame}{Stack Alignment and Padding Rules}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item AAPCS64 specifically requires that \key{the stack pointer remains 16-byte aligned at public interfaces and whenever memory is accessed via \bits{sp}}.
  \item If a function needs a 16-byte header (FP/LR) plus 20 bytes of local variables, the total is 36 bytes, which must be rounded up to 48 bytes to maintain 16-byte stack alignment.
\end{itemize}

\vspace{0.2cm}
\begin{block}{Calculating Frame Size}
$$\text{Total Frame Size} = \text{align}_{16}(\text{Header (16B)} + \text{Callee Regs} + \text{Local Variables})$$
\end{block}

\vspace{0.2cm}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{lll}
\toprule
\textbf{Header + Requested Space} & \textbf{Padded Frame Allocation} & \textbf{Instruction} \\
\midrule
16B Header + 0B Locals = 16B & 16 bytes & \code{sub sp, sp, \#16} \\
16B Header + 20B Locals = 36B & 48 bytes & \code{sub sp, sp, \#48} \\
16B Header + 32B Locals = 48B & 48 bytes & \code{sub sp, sp, \#48} \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\begin{frame}{Common Function Linkage Bugs in Assembly}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item \key{1. Overwriting Link Register:} Executing \code{bl} in a function that needs to return without preserving its incoming return address destroys the return path, causing infinite loops or crashes.
  \item \key{2. Corrupting Callee Registers:} Modifying \bits{x19}--\bits{x28} without preserving them corrupts caller local variables.
  \item \key{3. Stack Misalignment:} Violating AAPCS64 16-byte stack alignment requirements can cause compatibility issues or runtime faults where hardware SP-alignment checking is enforced.
  \item \key{4. Unbalanced Stack Pointer:} Modifying \bits{sp} in body without restoring exact offsets can corrupt stack-resident state and cause incorrect restores or returns.
\end{itemize}
\end{frame}

\begin{frame}{Topic 6: Comprehensive Summary}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item \key{Linkage Primitives:} \code{bl} saves return context ($\text{PC}+4$) into \bits{x30} and jumps; \code{ret} loads \bits{x30} back into \bits{PC}.
  \item \key{AAPCS64 Register Contract:}
  \begin{itemize}\setlength{\itemsep}{1.5pt}
    \item Parameters: Passed in \bits{x0}--\bits{x7}; returns in \bits{x0}/\bits{x1}.
    \item Scratch Registers: \bits{x0}--\bits{x17} (Caller-saved).
    \item Preserved Registers: \bits{x19}--\bits{x28} (Callee-saved).
  \end{itemize}
  \item \key{Activation Frames:} When a frame pointer is used, \bits{x29} can provide a stable frame reference; stack space is allocated in 16-byte aligned blocks.
  \item \key{Prologue/Epilogue:} Standard pre/post-indexed \code{stp}/\code{ldp} instructions handle saving and restoring \bits{x29}/\bits{x30} cleanly.
\end{itemize}

\end{frame}
