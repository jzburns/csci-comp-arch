\begin{frame}{What Is Architectural State?}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item \key{Definition:} The complete set of software-visible hardware storage locations required to represent a program's execution at any discrete point in time.
  \item If execution is interrupted (context switch, interrupt, exception), saving and restoring this exact state allows program resumption without alteration of behavior.
  \item \key{Core Components in AArch64:}
  \begin{itemize}\setlength{\itemsep}{2.5pt}
    \item General-Purpose Registers (\bits{x0}--\bits{x30}).
    \item Program Counter (\bits{pc}) and Stack Pointer (\bits{sp}).
    \item Current Program Status Register / Condition Flags (\bits{pstate} / NZCV).
    \item System control and configuration registers (e.g., $\text{TTBR0\_EL1}$, $\text{SCTLR\_EL1}$ for exception/virtualization management).
    \item Byte-addressed memory contents.
  \end{itemize}
\end{itemize}

\end{frame}

% ============================================================
% Slide 4
% ============================================================

\begin{frame}{Architectural vs.\ Microarchitectural State}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Architectural State (Visible Contract)}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Defined strictly by the ISA specification.
  \item Software and compilers directly read and write this state.
  \item Examples:
  \begin{itemize}
    \item Registers \bits{x0}--\bits{x30}
    \item Program counter (\bits{pc})
    \item Architectural memory space
    \item Flag bits (N, Z, C, V)
  \end{itemize}
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Microarchitectural State (Hidden)}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Internal hardware implementation details.
  \item Invisible to software correctness; exists solely for performance.
  \item Examples:
  \begin{itemize}
    \item Pipeline register latches
    \item L1/L2/L3 cache line contents
    \item Branch Target Buffers (BTB)
    \item Physical register rename tables
  \end{itemize}
\end{itemize}
\end{block}
\end{column}
\end{columns}

\vspace{0.25cm}
\key{Architectural Boundary:} Hardware may execute instructions out of order across hidden buffers, but must commit results sequentially to architectural state.
\end{frame}

% ============================================================
% Slide 5
% ============================================================

\begin{frame}{The AArch64 Execution State}

\begin{itemize}\setlength{\itemsep}{4pt}
  \item \key{AArch64:} 64-bit execution state introduced in ARMv8-A.
  \item \key{Flat 64-bit Virtual Addressing:} Registers hold 64-bit pointers (architectural address space up to $2^{64}$ bytes; OS implementations typically configure 48-bit or 52-bit spaces).
  \item \key{Uniform 32-bit Instructions:} Fixed 4-byte width, aligned to 4-byte memory boundaries.
  \item \key{Clean-Slate RISC Design (Eliminates A32 Quirks):}
  \begin{itemize}\setlength{\itemsep}{1.5pt}
    \item No universal conditional execution on every instruction.
    \item No load/store-multiple (replaced by paired \code{ldp}/\code{stp}).
    \item Program Counter (\bits{pc}) is decoupled from general-purpose registers.
  \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}{General-Purpose Registers: \texttt{x0}--\texttt{x30}}

\begin{itemize}\setlength{\itemsep}{3.5pt}
  \item AArch64 provides \key{31 general-purpose registers} (\bits{x0}--\bits{x30}), each \key{64 bits wide} ($8\text{ bytes}$).
  \item \key{Orthogonal Design:} Any register can serve as source, destination, or base address pointer.
  \item \key{Simplified ISA:} Eliminates dedicated register constraints from legacy architectures.
\end{itemize}

\vspace{0.25cm}

  \node[anchor=south, font=\tiny\bfseries] at (8.7, 0.5) {Bit 0};
\end{tikzpicture}

\end{frame}

\begin{frame}{32-Bit Register Views: \texttt{w0}--\texttt{w30}}

\begin{itemize}\setlength{\itemsep}{4pt}
  \item Every 64-bit \bits{x} register has a corresponding 32-bit view denoted as \bits{w0} through \bits{w30}.
  \item \bits{wN} represents the \key{lower 32 bits (bits 31:0)} of register \bits{xN}.
  \item Enables efficient execution of 32-bit integer operations without separate hardware register files.
\end{itemize}

\vspace{0.15cm}

  \node[anchor=south, font=\tiny\bfseries] at (0.2, 0.65) {Bit 63};
  \node[anchor=south, font=\tiny\bfseries] at (6.0, 0.65) {Bit 32};
  \node[anchor=south, font=\tiny\bfseries] at (11.3, 0.65) {Bit 0};

\end{tikzpicture}

\vspace{0.2cm}
\begin{block}{Zero-Extension Rule on 32-Bit Writes}
Writing to a \bits{w} register \key{automatically clears (zero-extends)} the upper 32 bits of the corresponding \bits{x} register to zero.
\end{block}
\end{frame}

% ============================================================
% Slide 9
% ============================================================

\begin{frame}{The Zero Register: \texttt{xzr} and \texttt{wzr}}

\begin{itemize}\setlength{\itemsep}{4pt}
  \item Encoded as register number 31 in most instruction contexts.
  \item \key{Read Behavior:} Always returns the constant value \bits{0} (64-bit \bits{0} for \bits{xzr}, 32-bit \bits{0} for \bits{wzr}).
  \item \key{Write Behavior:} Discards any value written to it (serves as a discard target).
  \item \key{Architectural Benefit:} Eliminates dedicated zero-loading instructions and comparison formats.
\end{itemize}

\vspace{0.15cm}
\begin{block}{Common Idioms Using the Zero Register}
\small
\begin{tabular}{lll}
\toprule
\textbf{Assembly Instruction} & \textbf{Equivalent Operation} & \textbf{Purpose} \\
\midrule
\code{mov x0, xzr}            & $x_0 \leftarrow 0$            & Clear register $x_0$ to zero \\
\code{subs xzr, x1, x2}       & Evaluate $x_1 - x_2 \to \text{flags}$ & Compare $x_1$ and $x_2$ (alias: \code{cmp x1, x2}) \\
\code{str xzr, [x1]}          & $\text{Mem}[x_1] \leftarrow 0$ & Zero out a 64-bit memory word \\
\bottomrule
\end{tabular}
\end{block}
\end{frame}

% ============================================================
% Slide 10
% ============================================================

\begin{frame}{Special-Purpose Registers: \texttt{sp}, \texttt{pc}, \texttt{lr}}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Stack Pointer (\texttt{sp})}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Dedicated register holding current stack top address.
  \item Must maintain \key{16-byte alignment} when accessing memory in standard AArch64 ABI.
  \item Arithmetic on \bits{sp} is restricted to explicit stack allocations/deallocations.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Program Counter (\texttt{pc})}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Holds memory address of executing instruction.
  \item Advances by $+4$ bytes on sequential execution.
  \item \key{Cannot} be read or written directly as a general data register (no direct $\code{mov x0, pc}$), but can be accessed indirectly via PC-relative addressing (\code{adr}, \code{adrp}).
  \item Modified via branch and call instructions.
\end{itemize}
\end{block}
\end{column}
\end{columns}

\vspace{0.2cm}
\begin{block}{Link Register (\texttt{x30} / \texttt{lr})}
Holds the return address generated by subroutine branch-with-link instructions (\code{bl}, \code{blr}).
\end{block}
\end{frame}

% ============================================================
% Slide 11: Definitions First (Caller vs Callee-Saved) - with negative vspace
% ============================================================

\begin{frame}{The Calling Convention: Caller- vs.\ Callee-Saved}
\vspace{-0.7cm}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Caller-Saved Registers (\texttt{x0}--\texttt{x17})}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Used for temporary computations and arguments.
  \item The caller must save them to memory/stack before making a subroutine call if their values are needed later.
  \item The called function (callee) is free to overwrite them without restoring.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Callee-Saved Registers (\texttt{x19}--\texttt{x28})}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Used to hold long-lived variables across calls.
  \item If a called function modifies any of these, it \key{must save the original value to the stack} and restore it before returning.
  \item Protects caller state across subroutines.
\end{itemize}
\end{block}
\end{column}
\end{columns}

\vspace{0.25cm}
\begin{block}{Design Trade-off}
Balancing caller- and callee-saved registers minimizes unnecessary memory traffic during deeply nested function calls.
\end{block}
\end{frame}

\begin{frame}{Process State: \texttt{pstate} and Condition Flags}

\begin{itemize}\setlength{\itemsep}{4pt}
  \item The processor maintains an internal process state register (\bits{pstate}).
  \item The upper 4 bits contain the \key{NZCV Arithmetic Condition Flags}:
\end{itemize}

\vspace{0.15cm}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{cll}
\toprule
\textbf{Flag} & \textbf{Name} & \textbf{Condition Under Which Flag Is Set to 1} \\
\midrule
\key{N} & \textbf{Negative} & Result bit 63 (or bit 31) is 1 (negative signed value) \\
\key{Z} & \textbf{Zero}     & Computed result is exactly zero \\

\begin{block}{Selective Flag Updates}
Standard ALU instructions (\code{add}, \code{sub}, \code{and}) do \key{not} modify flags. Only instructions with an ``\code{s}'' suffix (\code{adds}, \code{subs}, \code{ands}) or comparison instructions (\code{cmp}) update NZCV.
\end{block}
\end{frame}

\begin{frame}{Memory Alignment Rules}

\begin{itemize}\setlength{\itemsep}{4pt}
  \item An access to an $n$-byte operand is \key{naturally aligned} if its memory address is an exact multiple of $n$: $\text{Address} \pmod n = 0$.
  \item \key{Alignment Requirements:}
  \begin{itemize}\setlength{\itemsep}{2pt}
    \item 32-bit Word access: Address must be a multiple of 4 ($\text{Address} \pmod 4 = 0$).
    \item 64-bit Doubleword access: Address must be a multiple of 8 ($\text{Address} \pmod 8 = 0$).
    \item Instruction fetch: Address must be a multiple of 4.
  \end{itemize}
\end{itemize}

\begin{frame}{Instruction Syntax Conventions}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item Standard AArch64 assembly follows a strict operand ordering:
  $$\code{mnemonic\quad destination,\; source1,\; source2\_or\_immediate}$$

  \item \key{Three-Operand Format:} Non-destructive; source registers are preserved.
\end{itemize}

\toprule
\textbf{Assembly Example} & \textbf{RTL Operation} & \textbf{Description} \\

\code{add w0, w1, w2}     & $w_0 \leftarrow w_1 + w_2$             & 32-bit addition (clears upper 32 bits of $x_0$) \\
\code{add x0, x1, \#16}   & $x_0 \leftarrow x_1 + 16$              & Addition with immediate constant \\
\bottomrule
\end{tabular}

\begin{frame}{Memory Access: Basic Load/Store Architecture}

\begin{itemize}\setlength{\itemsep}{4pt}
  \item Memory is accessed exclusively through explicit load (\code{ldr}) and store (\code{str}) instructions.
  \item Register size specifier determines data transfer width:

\begin{center}

\setlength{\tabcolsep}{6pt}
\begin{tabular}{llll}
\toprule
\textbf{Instruction} & \textbf{Register Type} & \textbf{Transfer Width} & \textbf{Action} \\
\midrule

\begin{frame}{Addressing Modes: Base with Immediate Offset}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Mechanism ($\text{EA} = x_n + \text{offset}$)}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item Instruction: \code{ldr xt, [xn, \#offset]}
  \item Accesses memory at $x_n + \text{offset}$.
  \item \key{No side effects:} Base register $x_n$ remains unchanged.

\end{tabular}
\end{block}
\end{column}
\end{columns}

\vspace{0.15cm}
\begin{block}{Hardware Offset Range}
\footnotesize Standard immediate offsets are unsigned 12-bit scaled values (for 64-bit doublewords, offsets up to $+32760$ bytes).

\begin{frame}{Advanced Addressing: Pre- and Post-Indexed}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Pre-Indexed (Update Before Access)}
$$\code{ldr xt, [xn, \#imm]!}$$
\begin{enumerate}\setlength{\itemsep}{2pt}
  \item Update base: $x_n \leftarrow x_n + \text{imm}$.
  \item Access memory at updated address: $x_t \leftarrow \text{Mem}[x_n]$.
\end{enumerate}
\vspace{0.1cm}
Note the exclamation mark (\code{!}) syntax.
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Post-Indexed (Update After Access)}
$$\code{ldr xt, [xn], \#imm}$$
\begin{enumerate}\setlength{\itemsep}{2pt}
  \item Access memory at current address: $x_t \leftarrow \text{Mem}[x_n]$.
  \item Update base: $x_n \leftarrow x_n + \text{imm}$.
\end{enumerate}
\vspace{0.1cm}
Base update happens \emph{after} the load.
\end{block}
\end{column}
\end{columns}

\vspace{0.25cm}
\key{Efficiency Impact:} Eliminates separate \code{add} instructions when traversing arrays or pushing/popping stack elements.
\end{frame}

\begin{frame}{Loading and Storing Pairs (\texttt{ldp} and \texttt{stp})}

\begin{itemize}\setlength{\itemsep}{6pt}
  \item \key{Core Purpose:} Transfer two adjacent 64-bit registers in a single instruction.
  \item \key{Primary Use Case (Stack Management):}
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item Combines two register movements into one atomic operation.
    \item Efficiently saves and restores function linkage (Frame Pointer $x_{29}$ and Link Register $x_{30}$) during function calls.
  \end{itemize}
  \item \key{Performance Impact:} Issues a single 128-bit memory bus transaction, doubling data throughput compared to individual loads or stores.
\end{itemize}

\end{frame}

\begin{itemize}\setlength{\itemsep}{4pt}

\begin{itemize}\setlength{\itemsep}{4pt}

\setlength{\tabcolsep}{6pt}

\begin{frame}{Topic 3: Comprehensive Summary}

\begin{itemize}\setlength{\itemsep}{6pt}
  \item \key{Architectural State:} Precise hardware-software contract defined by registers (\bits{x0}--\bits{x30}), flags (NZCV), stack pointer (\bits{sp}), program counter (\bits{pc}), system control registers, and byte-addressed memory.
  \item \key{Register Structure:} 31 orthogonal 64-bit registers (\bits{x0}--\bits{x30}) with 32-bit aliases (\bits{w0}--\bits{w30}); writing to a \bits{w} register automatically clears the upper 32 bits.
  \item \key{ABI Rules:} \bits{x0}--\bits{x7} handle arguments and returns (caller-saved); \bits{x19}--\bits{x28} hold local variables across calls (callee-saved).
  \item \key{Load/Store Operations:} Arithmetic occurs exclusively in registers; memory is accessed via \code{ldr}/\code{str} using offset, pre-indexed, post-indexed, or paired (\code{ldp}/\code{stp}) modes.
  \item \key{Control Flow:} Branching evaluates NZCV flags updated by \code{adds}/\code{subs}/\code{cmp}; conditional select (\code{csel}) eliminates branch penalties for small conditionals.
\end{itemize}

\end{frame}
