\begin{frame}{Arithmetic Primitives: Addition \& Subtraction}
\normalsize
\begin{itemize}\setlength{\itemsep}{4pt}
  \item Many AArch64 data-processing instructions use a \key{3-operand non-destructive format}. Flag-setting forms such as \code{adds} and \code{subs} update NZCV; \code{add} and \code{sub} do not.
\end{itemize}

\vspace{0.08cm}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{lll}
\toprule
\textbf{Instruction} & \textbf{Operation} & \textbf{Flags Updated?} \\
\midrule
\code{add  x0, x1, x2}  & $x_0 \leftarrow x_1 + x_2$ & No \\
\code{adds x0, x1, x2}  & $x_0 \leftarrow x_1 + x_2$ & \key{Yes (NZCV)} \\
\code{sub  x0, x1, x2}  & $x_0 \leftarrow x_1 - x_2$ & No \\
\code{subs x0, x1, x2}  & $x_0 \leftarrow x_1 - x_2$ & \key{Yes (NZCV)} \\
\code{neg  x0, x1}      & $x_0 \leftarrow 0 - x_1$   & No; alias of \code{sub x0, xzr, x1} \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.08cm}
\begin{block}{Immediate Constant Arithmetic}
\small
\code{add x0, x1, \#42} uses a 12-bit immediate. The add/sub immediate form can also shift that field left by 12 bits. The corresponding \bits{w}-register form performs 32-bit arithmetic.
\end{block}
\end{frame}

% ============================================================
% Slide 5
% ============================================================

\begin{frame}{Shifted-Register Operands and Shift Operations}
\begin{itemize}\setlength{\itemsep}{4pt}
  \item Many AArch64 data-processing instructions can encode a shifted register operand directly in the instruction.
  \item Execution latency for shifted operands is implementation-dependent.
\end{itemize}

\vspace{0.12cm}
\begin{center}
\scriptsize
\setlength{\tabcolsep}{4pt}
\begin{tabular}{p{0.22\textwidth}p{0.15\textwidth}p{0.55\textwidth}}
\toprule
\textbf{Shift Type} & \textbf{Mnemonic} & \textbf{Operation Meaning} \\
\midrule
Logical Shift Left & \code{lsl \#k} & Multiply by $2^k$ modulo the register width; zeros enter the low bits. \\
Logical Shift Right & \code{lsr \#k} & Unsigned floor division by $2^k$; zeros enter the high bits. \\
Arithmetic Shift Right & \code{asr \#k} & Sign-preserving right shift; the sign bit is replicated into the high bits. \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.10cm}
\begin{block}{Integrated Shift-Add Examples}
\footnotesize
\begin{tabular}{ll}
\code{add x0, x1, x2, lsl \#3} & $x_0 \leftarrow x_1 + (x_2 \times 8)$ \\
\code{sub x0, x1, x2, lsr \#1} & $x_0 \leftarrow x_1 - \lfloor x_2/2 \rfloor$ for unsigned $x_2$ \\
\end{tabular}
\end{block}
\end{frame}

% ============================================================
% Slide 6
% ============================================================

\begin{frame}{Integer Multiplication \& Division}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item AArch64 defines instructions for integer multiplication and division, leaving the underlying microarchitectural execution units implementation-dependent.
\end{itemize}

\vspace{0.15cm}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{lll}
\toprule
\textbf{Instruction} & \textbf{RTL Semantic Action} & \textbf{Description} \\
\midrule
\code{mul  x0, x1, x2}        & $x_0 \leftarrow x_1 \times x_2$         & Lower 64 bits of the product \\
\code{madd x0, x1, x2, x3}    & $x_0 \leftarrow x_3 + (x_1 \times x_2)$ & Multiply-Add \\
\code{msub x0, x1, x2, x3}    & $x_0 \leftarrow x_3 - (x_1 \times x_2)$ & Multiply-Subtract \\
\code{sdiv x0, x1, x2}        & $x_0 \leftarrow x_1 / x_2$             & Signed division \\
\code{udiv x0, x1, x2}        & $x_0 \leftarrow x_1 / x_2$             & Unsigned division \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.25cm}
\normalsize
\begin{block}{Division by Zero}
AArch64 integer division by zero does \textbf{not} raise an exception; it returns \bits{0} in the destination register. For signed division, the overflow case $\mathrm{INT64\_MIN}/(-1)$ also does not trap; the destination receives the architecturally defined 64-bit result.
\end{block}
\end{frame}

% Slide 7
% ============================================================

\begin{frame}{Bitwise Logic Primitives}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item Bitwise operations apply independently to each corresponding bit position.
\end{itemize}

\vspace{0.15cm}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{lll}
\toprule
\textbf{Instruction} & \textbf{Operation} & \textbf{Bitwise RTL Formula} \\
\midrule
\code{and x0, x1, x2} & Bitwise AND & $x_0 \leftarrow x_1 \land x_2$ \\
\code{orr x0, x1, x2} & Bitwise OR  & $x_0 \leftarrow x_1 \lor x_2$ \\
\code{eor x0, x1, x2} & Bitwise XOR & $x_0 \leftarrow x_1 \oplus x_2$ \\
\code{bic x0, x1, x2} & Bit Clear (AND NOT) & $x_0 \leftarrow x_1 \land (\sim x_2)$ \\
\code{mvn x0, x1}     & Bitwise NOT & $x_0 \leftarrow \sim x_1$ \quad (Alias: \code{orn x0, xzr, x1}) \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.25cm}
\normalsize
\begin{block}{Bit Clear (\texttt{bic}) Idiom}
\code{bic x0, x1, x2} sets bits to \bits{0} in $x_1$ wherever the mask in $x_2$ has \bits{1}s.
\end{block}
\end{frame}

% Slide 8
% ============================================================

\begin{frame}{Bitfield Extraction: \texttt{ubfx}}
\begin{itemize}\setlength{\itemsep}{4pt}
  \item AArch64 provides bitfield instructions that extract packed fields without explicit mask-and-shift sequences.
  \item \code{ubfx xd, xn, \#lsb, \#width}: unsigned bitfield extract.
\end{itemize}

\vspace{0.12cm}
\begin{center}
\begin{tikzpicture}[scale=0.82,every node/.style={transform shape}]
\draw[fill=lightblue,draw=navy] (0,0) rectangle (7,0.55);
\draw[fill=gold!30,draw=navy] (3.0,0) rectangle (5.0,0.55);
\node[font=\tiny\bfseries] at (0.15,0.85) {Bit 63};
\node[font=\tiny\bfseries] at (3.0,0.85) {Bit (lsb + width - 1)};
\node[font=\tiny\bfseries] at (5.0,0.85) {Bit lsb};
\node[font=\tiny\bfseries] at (6.85,0.85) {Bit 0};
\node[font=\small\bfseries] at (4.0,0.28) {Extracted Field};
\end{tikzpicture}
\end{center}

\vspace{0.15cm}
\begin{block}{Example: Extract 8-bit Field Starting at Bit 16}
\footnotesize
\begin{tabular}{ll}
\textbf{Mask + shift:} & \code{lsr x0, x1, \#16} $\to$ \code{and x0, x0, \#0xff} \\
\textbf{Bitfield extract:} & \code{ubfx x0, x1, \#16, \#8} \\
\end{tabular}
\end{block}
\end{frame}

\begin{frame}{NZCV Condition Flags for Control Flow}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item Topic 4 is where we use the \bits{pstate} condition flags in assembly programs.
  \item The NZCV bits record the status of the most recent instruction that updates them.
  \item Common flag-setting forms include instructions with an ``\code{s}'' suffix and explicit comparisons.
\end{itemize}

\vspace{-0.2cm}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{cll}
\toprule
\textbf{Flag} & \textbf{Name} & \textbf{Condition Set to 1} \\
\midrule
\key{N} & \textbf{Negative} & The most-significant bit of the result is 1. \\
\key{Z} & \textbf{Zero}     & Result is exactly zero. \\
\key{C} & \textbf{Carry}    & Unsigned carry-out (addition) or \key{no borrow} (subtraction). \\
\key{V} & \textbf{Overflow} & The signed mathematical result cannot be represented in the destination width. \\
\bottomrule
\tabularnewline
\end{tabular}
\end{center}

\vspace{-0.5cm}
\begin{block}{Carry Flag in Subtraction}
In AArch64 subtraction ($A - B$), $\text{C} = 1$ indicates $A \ge B$ (no borrow required). $\text{C} = 0$ indicates $A < B$ (unsigned borrow occurred).
\end{block}
\end{frame}

% ============================================================
% Slide 10
% ============================================================

\begin{frame}{Comparison Instructions: \texttt{cmp}, \texttt{cmn}, and \texttt{tst}}
\begin{itemize}\setlength{\itemsep}{4pt}
  \item Comparison aliases set NZCV flags and discard the computed result; later branches interpret those flags.
\end{itemize}

\vspace{0.10cm}
\begin{center}
\footnotesize
\setlength{\tabcolsep}{4pt}
\begin{tabular}{p{0.17\textwidth}p{0.31\textwidth}p{0.44\textwidth}}
\toprule
\textbf{Instruction} & \textbf{Equivalent Operation} & \textbf{Flags Set} \\
\midrule
\code{cmp x0, x1} & \code{subs xzr, x0, x1} & NZCV from $x_0-x_1$ \\
\code{cmn x0, x1} & \code{adds xzr, x0, x1} & NZCV from $x_0+x_1$ \\
\code{tst x0, x1} & \code{ands xzr, x0, x1} & N/Z from the AND result; C/V cleared \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.12cm}
\begin{block}{Checking Flags After \texttt{cmp x0, x1}}
\small
\begin{itemize}\setlength{\itemsep}{1pt}
  \item $x_0 == x_1$: \key{$Z=1$}.
  \item $x_0 < x_1$ (signed): \key{$N \neq V$}.
  \item $x_0 < x_1$ (unsigned): \key{$C=0$}.
\end{itemize}
\end{block}
\end{frame}

% ============================================================
% Slide 11
% ============================================================

\begin{frame}{Condition Codes: Signed vs.\ Unsigned Branches}
\begin{center}
\small
\setlength{\tabcolsep}{6pt}
\begin{tabular}{llll}
\toprule
\textbf{Branch} & \textbf{Meaning} & \textbf{Flag Condition} & \textbf{Comparison Context} \\
\midrule
\code{b.eq} & Equal                     & $\text{Z} == 1$           & Signed / Unsigned \\
\code{b.ne} & Not Equal                 & $\text{Z} == 0$           & Signed / Unsigned \\
\midrule
\code{b.lt} & Signed Less Than ($<$)    & $\text{N} \neq \text{V}$  & \key{Signed} \\
\code{b.le} & Signed Less or Equal ($\le$) & $(\text{Z} == 1) \lor (\text{N} \neq \text{V})$ & \key{Signed} \\
\code{b.gt} & Signed Greater Than ($>$) & $(\text{Z} == 0) \land (\text{N} == \text{V})$ & \key{Signed} \\
\code{b.ge} & Signed Greater or Equal ($\ge$) & $\text{N} == \text{V}$ & \key{Signed} \\
\midrule
\code{b.lo} & Unsigned Lower ($<$)      & $\text{C} == 0$           & \key{Unsigned} \\
\code{b.hs} & Unsigned Higher or Same ($\ge$) & $\text{C} == 1$      & \key{Unsigned} \\
\code{b.hi} & Unsigned Higher ($>$)     & $(\text{C} == 1) \land (\text{Z} == 0)$ & \key{Unsigned} \\
\code{b.ls} & Unsigned Lower or Equal ($\le$) & $(\text{C} == 0) \lor (\text{Z} == 1)$ & \key{Unsigned} \\
\bottomrule
\end{tabular}
\end{center}

\vspace{-0.35cm}
\normalsize
\begin{block}{Usage Context}
Unsigned conditions are appropriate when the program semantics treat the values as unsigned; they are also commonly used for address and range comparisons.
\end{block}
\end{frame}

\begin{frame}{Compare-and-Branch Instructions: \texttt{cbz} and \texttt{cbnz}}
\begin{itemize}\setlength{\itemsep}{4pt}
  \item \key{\code{cbz} / \code{cbnz}} test a register directly and do not alter NZCV.
  \item They encode the zero test and branch in one architectural instruction instead of a separate compare and conditional branch.
\end{itemize}

\vspace{0.12cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{block}{Null Pointer Validation}
\footnotesize
\begin{tabular}{ll}
\code{cbz x0, .Lnull} & \text{\scriptsize // branch if null} \\
\code{ldr x1, [x0]}   & \text{\scriptsize // dereference} \\
\end{tabular}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Loop Counter Decrement}
\footnotesize
\begin{tabular}{ll}
\code{.Lloop:} & \\
\code{sub  x1, x1, \#1} & \text{\scriptsize // count--} \\
\code{cbnz x1, .Lloop} & \text{\scriptsize // repeat if nonzero} \\
\end{tabular}
\end{block}
\end{column}
\end{columns}

\vspace{0.12cm}
\small
\key{Single-Bit Test:} \code{tbz x0, \#0, label} branches if bit 0 is zero.
\end{frame}

% ============================================================
% Slide 13
% ============================================================

\begin{frame}[fragile]{Translating \texttt{if-then-else} Blocks}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.44\textwidth}
\begin{block}{High-Level C Code}
\small
\begin{verbatim}
if (x > y) {
    result = x - y;
} else {
    result = y - x;
}
\end{verbatim}
\end{block}
\end{column}

\begin{column}{0.52\textwidth}
\begin{block}{AArch64 Assembly Mapping}
\small
\begin{tabular}{ll}
\multicolumn{2}{l}{\textit{// Assume: x, y are signed int64\_t}} \\
\multicolumn{2}{l}{\textit{// x0=x, x1=y, x2=result}} \\
\code{cmp  x0, x1}        & \text{\scriptsize // Compare x and y} \\
\code{b.le .Lelse}        & \text{\scriptsize // Invert test: jump if x <= y} \\
\code{sub  x2, x0, x1}    & \text{\scriptsize // Then-body: result = x - y} \\
\code{b    .Lend}         & \text{\scriptsize // Skip else block} \\
\code{.Lelse:}            & \\
\code{sub  x2, x1, x0}    & \text{\scriptsize // Else-body: result = y - x} \\
\code{.Lend:}             & \\
\end{tabular}
\end{block}
\end{column}
\end{columns}

\vspace{0.3cm}
\normalsize
\key{Assembly Pattern:} Invert the high-level condition to fall through into the ``then'' path.
\end{frame}

\begin{frame}[fragile]{Loop Translation: Inverted Loop Pattern}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.44\textwidth}
\begin{block}{High-Level C Loop}
\small
\begin{verbatim}
// Compute array sum
int64_t i = 0;
int64_t n = ...; // signed bound
while (i < n) {
    sum += arr[i];
    i++;
}
\end{verbatim}
\end{block}
\end{column}

\begin{column}{0.52\textwidth}
\begin{block}{Bottom-Tested AArch64 Loop}
\small
\begin{tabular}{ll}
\multicolumn{2}{l}{\textit{// x0=arr, x1=n, x2=i, x3=sum}} \\
\code{mov  x2, xzr}       & \text{\scriptsize // i = 0} \\
\code{mov  x3, xzr}       & \text{\scriptsize // sum = 0} \\
\code{b    .Ltest}        & \text{\scriptsize // Jump to test first} \\
\code{.Lbody:}            & \\
\code{ldr  x4, [x0, x2, lsl \#3]} & \text{\scriptsize // Load arr[i]} \\
\code{add  x3, x3, x4}    & \text{\scriptsize // sum += arr[i]} \\
\code{add  x2, x2, \#1}   & \text{\scriptsize // i++} \\
\code{.Ltest:}            & \\
\code{cmp  x2, x1}        & \text{\scriptsize // i < n?} \\
\code{b.lt .Lbody}        & \text{\scriptsize // Loop if true} \\
\end{tabular}
\end{block}
\end{column}
\end{columns}

\vspace{0.25cm}
\normalsize
\key{Common Loop Layout:} Placing the test at the bottom gives the steady-state loop body one loop-closing conditional branch.
\end{frame}

\begin{frame}{Branch Misprediction \& Conditional Selection (\texttt{csel})}
\small
\begin{itemize}\setlength{\itemsep}{2pt}
  \item A mispredicted conditional branch discards younger speculative work; the cost is implementation-dependent.
  \item \key{\code{csel}} chooses between two register operands from NZCV without changing control flow.
  \item \key{Trade-off:} Conditional selection avoids prediction for that decision but can extend a data-dependency chain.
\end{itemize}

\vspace{0.08cm}
\begin{block}{Signed Maximum: $\code{result} = (x_1 > x_2) \mathbin{?} x_1 : x_2$}
\footnotesize
\begin{tabular}{ll}
\code{cmp  x1, x2}          & \text{\scriptsize // set NZCV from x1 - x2} \\
\code{csel x0, x1, x2, gt}  & \text{\scriptsize // signed max; use hi for unsigned} \\
\end{tabular}
\end{block}

\vspace{0.08cm}
\begin{block}{Related Forms}
\footnotesize
\code{cset w0, eq} sets 0 or 1 from a condition; \code{cinc x0, x1, eq} conditionally increments \bits{x1}.
\end{block}
\end{frame}

\begin{frame}{Topic 4: Comprehensive Summary}
\small
\begin{itemize}\setlength{\itemsep}{4pt}
  \item \key{Arithmetic \& Shifts:} Regular register and immediate arithmetic includes shifted-register operands in many data-processing instructions.
  \item \key{Bitwise Operations:} \code{and}, \code{orr}, \code{eor}, \code{bic}, and bitfield instructions operate directly on packed binary data.
  \item \key{NZCV Flags:} Flag-setting arithmetic and aliases such as \code{cmp} and \code{tst} support signed and unsigned conditions.
  \item \key{Control Flow:} Compilers commonly arrange one path as fall-through and use bottom-tested loops where appropriate.
  \item \key{Conditional Selection:} \code{csel} and related forms can replace short conditional branches when data-dependent selection is preferable.
\end{itemize}
\end{frame}
