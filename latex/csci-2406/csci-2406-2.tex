\sectiondivider{Section 01}{RISC Design Principles \&\\the Load/Store Paradigm}

% --- Slide 4: Core RISC Principles ---
\begin{frame}{Foundational RISC Design Principles}
    Reduced Instruction Set Computers (RISC) optimize the hardware-software interface by prioritizing simplicity and regularity:
    \vspace{0.2em}
    \begin{itemize}
        \item \key{Simpler Control Circuitry}: Regular instruction formats and orthogonal opcodes drastically reduce decode and control unit complexity.
        \item \key{Smaller Silicon Footprint}: Eliminating complex microcoded multi-cycle instructions frees die area for larger register files and caches.
        \item \key{Lower Energy Consumption}: Streamlined decoding reduces gate switching activity, yielding high performance-per-watt.
        \item \key{Predictable Pipelining}: Uniform fixed-width instruction lengths ensure smooth, hazard-free pipeline flow.
    \end{itemize}
\end{frame}

% --- Slide 5: The Load/Store Model ---
\begin{frame}{The Load/Store Architectural Paradigm}
    \begin{block}{Separation of Memory Access and Computation}
        In a pure RISC architecture, data processing instructions (arithmetic, logic) operate \key{exclusively on registers}. Memory is accessed strictly through dedicated \key{load} and \key{store} instructions.
    \end{block}
    \vspace{0.3em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Arithmetic Isolation}
                \begin{itemize}
                    \item ALU inputs must reside in the register file.
                    \item ALU outputs are committed to a register.
                    \item No arithmetic instruction can target a memory address directly.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Explicit Memory Staging}
                \begin{itemize}
                    \item \code{ldr}: Reads data from memory into a register.
                    \item \code{str}: Writes data from a register into memory.
                    \item Simplifies pipeline address generation and memory stages.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 6: The Load/Store Execution Sequence ---
\begin{frame}{Executing Memory Mutation in AArch64}
    To mutate a 64-bit variable in RAM (e.g., \code{x = x + 5}), AArch64 strictly sequences three distinct operations:
    \vspace{0.3em}
    \begin{enumerate}
        \item \key{Load Stage}: Read 64-bit value from memory base pointer \code{[x0]} into register \code{x1}.
        \item \key{Compute Stage}: Perform 64-bit register addition using immediate value \code{\#5}.
        \item \key{Store Stage}: Write modified 64-bit value in \code{x1} back to memory address \code{[x0]}.
    \end{enumerate}
    \vspace{0.2em}
    \begin{center}
        \begin{tikzpicture}[node distance=0.6cm, auto,
            codebox/.style={rectangle, draw=navy, fill=lightblue, text=darkgray, font=\small\ttfamily, text width=9.5cm, align=left, rounded corners=2pt, minimum height=0.45cm}]
            \node [codebox] (c1) {ldr x1, [x0] \hfill // 1. Load 64-bit x from RAM into register x1};
            \node [codebox, below of=c1] (c2) {add x1, x1, \#5 \hfill // 2. Perform arithmetic within register file};
            \node [codebox, below of=c2] (c3) {str x1, [x0] \hfill // 3. Write updated 64-bit result back to RAM};
        \end{tikzpicture}
    \end{center}
\end{frame}

\sectiondivider{Section 02}{CISC versus RISC\\Architectural Trade-Offs}

% --- Slide 9: CISC Philosophy ---
\begin{frame}{Complex Instruction Set Computers (CISC)}
    CISC architectures maximize work accomplished per individual instruction:
    \vspace{0.2em}
    \begin{itemize}
        \item \key{Rich, Multi-Step Operations}: A single instruction combines memory fetching, arithmetic calculation, and memory write-back.
        \item \key{Direct Memory Arithmetic}: Instructions can operate directly on operands residing in memory (e.g., \code{add [rax], rbx}).
        \item \key{Variable-Length Formats}: Instruction encodings range from 1 to 15 bytes, requiring complex multi-stage decoders.
        \item \key{Dense Static Code}: High semantic density minimizes program size in RAM.
    \end{itemize}
    \vspace{0.2em}
    \begin{alertblock}{Design Penalty}
        Heavy silicon area and energy cost allocated to instruction decoding and complex hardware microcode control paths.
    \end{alertblock}
\end{frame}

% --- Slide 10: Concrete Comparison: x += 5 ---
\begin{frame}{Code Comparison: Implementing \code{x += 5}}
    Consider incrementing a 64-bit memory variable \code{x} by immediate constant 5:
    \vspace{0.3em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{CISC Implementation (x86-64)}
                \begin{itemize}
                    \item \textbf{Instruction}:
                    \code{add QWORD PTR [x], 5}
                    \item Single variable-length instruction.
                    \item CPU decodes operand, issues memory read, performs ALU add, and writes to memory in microcode.
                    \item Compact binary size, but complex execution timing.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{RISC Implementation (ARM64)}
                \begin{itemize}
                    \item \textbf{Instruction Stream}:
                    \begin{itemize}
                        \item \code{ldr x1, [x0]}
                        \item \code{add x1, x1, \#5}
                        \item \code{str x1, [x0]}
                    \end{itemize}
                    \item Three fixed 32-bit instructions.
                    \item Higher instruction count, but simple 1-cycle decode and pipeline predictability.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 11: Trade-Off Matrix ---
\begin{frame}{CISC vs. RISC Trade-Off Matrix}
    \begin{table}[]
        \centering
        \small
        \begin{tabular}{p{3.0cm} p{4.5cm} p{4.5cm}}
            \toprule
            \textbf{Dimension} & \textbf{CISC (e.g., x86-64)} & \textbf{RISC (e.g., AArch64, RISC-V)} \\
            \midrule
            \key{Instruction Size} & Variable (1 to 15 bytes) & Fixed (strictly 32-bit) \\
            \key{Memory Access} & Combined with computation & Strict Load/Store only \\
            \key{Addressing Modes} & Many, complex modes & Few, regular base-register modes \\
            \key{Decode Complexity} & High (requires decode ROM/$\mu$ops) & Low (hardwired, orthogonal fields) \\
            \key{Register Count} & 16 visible GPRs (\code{rax}--\code{r15}) & 31 visible GPRs (\code{x0}--\code{x30}) \\
            \key{Pipelining} & Difficult due to multi-cycle instructions & Highly efficient and deterministic \\
            \bottomrule
        \end{tabular}
    \end{table}
\end{frame}

\sectiondivider{Section 03}{AArch64 Architectural Overview\\and Encoding Model}

% --- Slide 13: ARM64 Architecture Overview ---
\begin{frame}{ARM64 (AArch64) Architecture Overview}
    \key{AArch64} is ARM's modern 64-bit RISC instruction set architecture designed for clean, scalable, high-performance computing:
    \vspace{0.2em}
    \begin{itemize}
        \item \key{Modern Lineage}: Introduced with ARMv8-A; powers Apple Silicon (M-series), AWS Graviton, and flagship mobile processors.
        \item \key{Cleaned Architecture}: Removed legacy ARM32 complexities (such as conditional execution fields on every opcode and exposed pipeline prefetch offsets).
        \item \key{Orthogonal Register Set}: 31 general-purpose 64-bit registers (\code{x0}--\code{x30}) plus dedicated \code{sp}, \code{pc}, and zero register \code{xzr}.
    \end{itemize}
    \vspace{0.2em}
    \begin{block}{Architectural Contract vs.\ Microarchitecture}
        The AArch64 ISA guarantees identical software behavior across energy-efficient IoT cores and high-throughput server chips.
    \end{block}
\end{frame}

% --- Slide 14: Fixed-Width Instruction Encoding ---
\begin{frame}{AArch64 Fixed-Width 32-Bit Instruction Encoding}
    In AArch64, every instruction is encoded into an exact \key{fixed 32-bit word}:
    \vspace{0.4em}
    \begin{center}
        \begin{tikzpicture}[
            scale=1.0,
            fld/.style={draw=navy, thick, minimum height=1.1cm, align=center, font=\scriptsize\bfseries}
        ]
            % sf (1 bit) + op (31:21)
            \node[fld, fill=navy, text=white, minimum width=2.2cm] (op) at (0,0) {
                Opcode / sf\\{\tiny[31:21]}
            };
            % Rm / Src2 [20:16] (5 bits)
            \node[fld, fill=deepblue, text=white, minimum width=2.2cm, right=0cm of op] (rm) {
                Rm (Src2)\\{\tiny[20:16]}
            };
            % Imm6 / Shift [15:10] (6 bits)
            \node[fld, fill=lightblue, text=navy, minimum width=2.2cm, right=0cm of rm] (shift) {
                Shift / Option\\{\tiny[15:10]}
            };
            % Rn / Src1 [9:5] (5 bits)
            \node[fld, fill=softgray, text=navy, minimum width=2.2cm, right=0cm of shift] (rn) {
                Rn (Src1)\\{\tiny[9:5]}
            };
            % Rd / Dest [4:0] (5 bits)
            \node[fld, fill=gold!40, text=navy, minimum width=2.2cm, right=0cm of rn] (rd) {
                Rd (Dest)\\{\tiny[4:0]}
            };
        \end{tikzpicture}
    \end{center}
    \vspace{0.2em}
    \begin{itemize}
        \item \key{5-Bit Register Specifiers}: Exactly 5 bits each for \code{rd}, \code{rn}, and \code{rm} to index all 31 registers ($2^5 = 32$).
        \item \key{Size Flag (\code{sf})}: Bit 31 selects between 32-bit operation (\code{w}-registers, $\code{sf}=0$) and 64-bit operation (\code{x}-registers, $\code{sf}=1$).
        \item \key{Consistent Alignment}: Instructions are strictly 4-byte aligned in memory.
    \end{itemize}
\end{frame}

% --- Slide 15: 32-Bit (W) vs. 64-Bit (X) Register Views ---
\begin{frame}{Dual Register Views: \code{X} (64-Bit) and \code{W} (32-Bit)}
    Each physical 64-bit register can be accessed in two distinct architectural widths:
    \vspace{0.3em}
    \begin{center}
        \begin{tikzpicture}[scale=0.9, every node/.style={font=\small}]
            % Full 64-bit X register
            \draw[thick, fill=navy, text=white] (0,1.2) rectangle (10,2.0);
            \node[text=white, font=\bfseries] at (5,1.6) {\code{Xn} (Full 64-Bit Register: Bits [63:0])};
            
            % Lower 32-bit W register
            \draw[thick, fill=gold!40, text=navy] (5,0.0) rectangle (10,0.8);
            \node[text=navy, font=\bfseries] at (7.5,0.4) {\code{Wn} (Lower 32-Bit View: Bits [31:0])};
            \draw[thick, dashed, fill=softgray, text=darkgray] (0,0.0) rectangle (5,0.8);
            \node[text=darkgray] at (2.5,0.4) {\textit{Zeroed on 32-bit write}};
        \end{tikzpicture}
    \end{center}
    \vspace{0.2em}
    \begin{block}{Zero-Extension Rule}
        When an instruction writes to a 32-bit destination register (\code{Wn}), hardware automatically \key{zeroes out the upper 32 bits} (bits [63:32]) of the corresponding \code{Xn} register, preventing stale data leakage.
    \end{block}
\end{frame}

\sectiondivider{Section 04}{Registers and Programmer-Visible State}

% --- Slide 17: AArch64 Register Model ---
\begin{frame}{AArch64 Programmer-Visible Register File}
    AArch64 exposes \key{31 general-purpose registers} (\code{x0}--\code{x30}), plus special registers:
    \vspace{0.2em}
    \begin{center}
        \small
        \begin{tabular}{p{2.5cm} p{3.2cm} p{5.5cm}}
            \toprule
            \textbf{Register} & \textbf{Architectural Role} & \textbf{Standard Software Convention} \\
            \midrule
            \code{x0}--\code{x7} & Parameter / Return & Function arguments 0--7; \code{x0}--\code{x1} return values \\
            \code{x8} & Indirect Result & Location pointer for large struct returns \\
            \code{x9}--\code{x15} & Caller-Saved & Temporary scratch registers \\
            \code{x16}--\code{x17} & IP0 / IP1 & Intra-procedure call scratch registers \\
            \code{x18} & Platform Register & Reserved for platform/OS runtime use \\
            \code{x19}--\code{x28} & Callee-Saved & Preserved across function calls \\
            \rowcolor{gold!15} \code{x29} (\code{fp}) & \key{Frame Pointer} & Points to base of current stack activation record \\
            \rowcolor{gold!15} \code{x30} (\code{lr}) & \key{Link Register} & Holds return address for function calls (\code{bl}) \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

% --- Slide 18: Special Registers: XZR, SP, and PC ---
\begin{frame}{Special Architectural Registers in AArch64}
    AArch64 eliminates ARM32 register aliasing by introducing dedicated registers:
    \vspace{0.3em}
    \begin{itemize}
        \item \key{Zero Register (\code{xzr} / \code{wzr})}:
        \begin{itemize}
            \item Reads always return numerical value \code{0}; writes are silently discarded.
            \item Encoded as register specifier 31 in most data-processing opcodes.
        \end{itemize}
        \vspace{0.2em}
        \item \key{Stack Pointer (\code{sp} / \code{wsp})}:
        \begin{itemize}
            \item Dedicated register for stack alignment (strictly 16-byte aligned at runtime).
            \item Encoded as register specifier 31 in load/store and stack arithmetic.
        \end{itemize}
        \vspace{0.2em}
        \item \key{Program Counter (\code{pc})}:
        \begin{itemize}
            \item Holds current instruction address; \key{not a general-purpose register}.
            \item Cannot be read or modified by arithmetic instructions (no \code{add pc, pc, \#4}).
            \item Updated exclusively by branch, call (\code{bl}), and exception return instructions.
        \end{itemize}
    \end{itemize}
\end{frame}

% --- Slide 19: PSTATE and Condition Flags ---
\begin{frame}{Processor State (PSTATE) and Condition Flags}
    In AArch64, status flags reside in specialized fields within \key{PSTATE} (accessed via \code{NZCV}):
    \vspace{0.2em}
    \begin{center}
        \begin{tikzpicture}[
            scale=1.0,
            fld/.style={draw=navy, thick, minimum height=1.0cm, align=center, font=\scriptsize\bfseries}
        ]
            \node[fld, fill=navy, text=white, minimum width=1.4cm] (n) at (0,0) {N\\{\tiny[31]}};
            \node[fld, fill=navy, text=white, minimum width=1.4cm, right=0cm of n] (z) {Z\\{\tiny[30]}};
            \node[fld, fill=navy, text=white, minimum width=1.4cm, right=0cm of z] (c) {C\\{\tiny[29]}};
            \node[fld, fill=navy, text=white, minimum width=1.4cm, right=0cm of c] (v) {V\\{\tiny[28]}};
            \node[fld, fill=softgray, text=darkgray, minimum width=4.5cm, right=0cm of v] (res) {Reserved / System Control Bits};
            \node[fld, fill=gold!40, text=navy, minimum width=1.6cm, right=0cm of res] (el) {EL\\{\tiny[3:2]}};
        \end{tikzpicture}
    \end{center}
    \vspace{0.1em}
    \begin{block}{The Arithmetic Condition Flags (NZCV)}
        \begin{itemize}
            \item \key{N (Negative)}: Set if result bit 63 (or 31) is 1 (negative signed value).
            \item \key{Z (Zero)}: Set if the result of the operation equals 0.
            \item \key{C (Carry)}: Set if an unsigned addition generated a carry-out (or unsigned sub did not borrow).
            \item \key{V (oVerflow)}: Set if signed operation exceeded 64-bit 2's complement bounds.
        \end{itemize}
    \end{block}
\end{frame}

% --- Slide 20: Complete 64-Bit Architectural State ---
\begin{frame}{Complete AArch64 Architectural State}
    \begin{block}{Formal Definition}
        \key{Architectural State} comprises all hardware registers and memory contents that determine the exact execution context of software.
    \end{block}
    \vspace{0.3em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Programmer-Visible State}
                \begin{itemize}
                    \item 31 General Registers: \code{x0}--\code{x30}.
                    \item Zero Register: \code{xzr} / \code{wzr}.
                    \item Dedicated Stack Pointer: \code{sp}.
                    \item Program Counter: \code{pc}.
                    \item Condition Flags: \code{PSTATE.NZCV}.
                    \item 64-Bit Linear Address Space (RAM).
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Microarchitectural Separation}
                \begin{itemize}
                    \item Physical rename registers ($128+$ entries).
                    \item Branch target buffers (BTB).
                    \item L1/L2/L3 cache hierarchies.
                    \item Speculative execution buffers.
                    \item \textit{All internal and invisible to software}.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\sectiondivider{Section 05}{Core AArch64 Instruction Classes}

% --- Slide 22: The Three Instruction Classes ---
\begin{frame}{AArch64 Instruction Categories}
    Every operation in the AArch64 ISA partitions into three primary functional classes:
    \vspace{0.4em}
    \begin{columns}[T]
        \begin{column}{0.32\textwidth}
            \begin{block}{1. Data Processing}
                \begin{itemize}
                    \item Arithmetic \& logic.
                    \item Register-only computation.
                    \item Suffix \code{'s'} sets flags (\code{adds}).
                    \item Examples: \code{add}, \code{sub}, \code{mul}, \code{mov}, \code{cmp}.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.32\textwidth}
            \begin{block}{2. Memory Access}
                \begin{itemize}
                    \item Base register addressing.
                    \item Explicit sizing (\code{b}, \code{h}, \code{w}, \code{x}).
                    \item Pre/post-indexing modes.
                    \item Examples: \code{ldr}, \code{str}, \code{ldp}, \code{stp}.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.32\textwidth}
            \begin{block}{3. Control Flow}
                \begin{itemize}
                    \item Unconditional branches.
                    \item Conditional branches.
                    \item Subroutine calls (\code{bl}).
                    \item Examples: \code{b}, \code{b.eq}, \code{b.ne}, \code{b.lt}, \code{ret}.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 23: Data Processing Instructions ---
\begin{frame}{Class 1: Data Processing Instructions}
    Data processing instructions compute mathematical or logical operations within registers:
    \vspace{0.3em}
    \begin{center}
        \small
        \begin{tabular}{p{3.5cm} p{3.5cm} p{4.5cm}}
            \toprule
            \textbf{Instruction Syntax} & \textbf{Semantic Operation} & \textbf{Architectural Effect} \\
            \midrule
            \code{add x3, x1, x2} & 64-Bit Addition & $\code{x3} \leftarrow \code{x1} + \code{x2}$ \\
            \code{sub x3, x3, x2} & 64-Bit Subtraction & $\code{x3} \leftarrow \code{x3} - \code{x2}$ \\
            \code{mul x3, x1, x2} & 64-Bit Multiplication & $\code{x3} \leftarrow \code{x1} \times \code{x2}$ \\
            \code{mov x3, x4} & Register Copy & $\code{x3} \leftarrow \code{x4}$ \\
            \code{and x0, x1, x2} & Bitwise Logical AND & $\code{x0} \leftarrow \code{x1} \text{ AND } \code{x2}$ \\
            \code{orr x0, x1, x2} & Bitwise Logical OR & $\code{x0} \leftarrow \code{x1} \text{ OR } \code{x2}$ \\
            \bottomrule
        \end{tabular}
    \end{center}
    \vspace{0.2em}
    \begin{block}{32-Bit Variant}
        \code{add w3, w1, w2} performs 32-bit addition and zero-extends the upper 32 bits of \code{x3}.
    \end{block}
\end{frame}

% --- Slide 24: Memory Instructions ---
\begin{frame}{Class 2: Memory Access (Load/Store)}
    Memory access instructions transfer data between RAM and the 64-bit register file:
    \vspace{0.3em}
    \begin{center}
        \begin{tikzpicture}[node distance=0.6cm, auto,

            \node [membox, below of=m2] (m3) {ldr x1, [x2, \#8] \hfill // Immediate offset: Load from address (x2 + 8) into x1};

    \end{center}
    \vspace{0.2em}
    \begin{block}{Bracket Notation \code{[x2]}}
        Brackets represent \key{memory dereferencing}. In AArch64, memory base pointers must reside in 64-bit registers (\code{x0}--\code{x30} or \code{sp}).
    \end{block}
\end{frame}

% --- Slide 25: Comparison Instructions ---
\begin{frame}{Comparison Operations and the \code{cmp} Instruction}
    Conditional decision-making begins with explicit comparison instructions:
    \vspace{0.3em}
    \begin{block}{\code{cmp x0, x1} (Alias for \code{subs xzr, x0, x1})}
        \begin{itemize}
            \item Performs internal 64-bit subtraction: $\text{Temp} \leftarrow \code{x0} - \code{x1}$.
            \item \key{Discards Result}: Writes output to the zero register (\code{xzr}).
            \item \key{Updates NZCV}: Sets condition flags in \code{PSTATE} based on the subtraction.
        \end{itemize}
    \end{block}
    \vspace{0.3em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{If $\code{x0} == \code{x1}$}
                $\code{x0} - \code{x1} = 0 \rightarrow$ \key{Z flag set to 1}.
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{If $\code{x0} < \code{x1}$}
                $\code{x0} - \code{x1} < 0 \rightarrow$ \key{N flag set to 1}.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 26: Class 3: Control Flow Instructions ---
\begin{frame}{Class 3: Control Flow and Branching}
    Branch instructions alter program flow by loading new target addresses into the \code{pc}:
    \vspace{0.3em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Unconditional Branch}
                \code{b target\_label}
                \begin{itemize}
                    \item Forces execution to diverge.
                    \item \code{bl label}: Branch with Link (writes return address into \code{x30/lr}).
                    \item \code{ret}: Returns to address in \code{lr} (\code{x30}).
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Conditional Branches}
                \code{b.<cond> target\_label}
                \begin{itemize}
                    \item Evaluates NZCV flags in \code{PSTATE}.
                    \item If condition holds true $\rightarrow$ updates \code{pc}.
                    \item If false $\rightarrow$ falls through ($PC \leftarrow PC + 4$).
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    \vspace{0.2em}
    \begin{center}
        \small
        \begin{tabular}{lll}
            \toprule
            \textbf{Mnemonic} & \textbf{Meaning} & \textbf{Tested Flag Condition} \\
            \midrule
            \code{b.eq} & Branch if Equal & $\text{Z} == 1$ \\
            \code{b.ne} & Branch if Not Equal & $\text{Z} == 0$ \\
            \code{b.lt} & Branch if Less Than (Signed) & $\text{N} \neq \text{V}$ \\
            \code{b.gt} & Branch if Greater Than (Signed) & $(\text{Z} == 0) \text{ and } (\text{N} == \text{V})$ \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

% --- Slide 27: Assembly Decisions (Fixed Alignment) ---
\begin{frame}{Assembly Decisions: The Compare-and-Branch Pattern}
    High-level \code{if-else} statements are implemented via a 2-step assembly pattern:
    \vspace{0.3em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Step 1: Evaluate Condition}
                \code{cmp x1, x2}
                \vspace{0.2em}
                \begin{itemize}
                    \item Subtracts \code{x2} from \code{x1} internally.
                    \item Updates \key{NZCV} flags in \code{PSTATE}.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Step 2: Test Flags \& Branch}
                \code{b.eq equal\_label}
                \vspace{0.2em}
                \begin{itemize}
                    \item Evaluates flag state ($Z == 1$).
                    \item If true $\rightarrow$ updates \code{pc} to label.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    \begin{block}{Complete Conditional Pattern}
        \vspace{0.1em}
        \centering
        \small
        \begin{tabular}{@{}ll@{}}
            \code{cmp x1, x2}           & \code{// 1. Compare x1 and x2} \\
            \code{b.eq equal\_path}      & \code{// 2. If x1 == x2, branch to equal\_path} \\
            \code{sub x0, x1, x2}       & \code{// Fall-through: executed if x1 != x2} \\
            \code{b end\_block}          & \code{// Bypass true-block} \\
            \code{equal\_path:}          & \\
            \code{\quad add x0, x1, x2} & \code{// True-block: executed if x1 == x2} \\
            \code{end\_block:}           & 
        \end{tabular}
        \vspace{0.1em}
    \end{block}
\end{frame}

% --- Slide 28: Modern Dedicated Compare-and-Branch ---
\begin{frame}{Optimized Branching: \code{cbz} and \code{cbnz}}
    AArch64 introduces instructions that combine testing and branching into a single cycle:
    \vspace{0.3em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Compare and Branch on Zero}
                \code{cbz x0, target\_label}
                \vspace{0.2em}
                \begin{itemize}
                    \item Tests if register \code{x0} is equal to 0.
                    \item If $\code{x0} == 0 \rightarrow$ branch to label.
                    \item \key{Does not alter NZCV flags}.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Compare and Branch on Non-Zero}
                \code{cbnz x0, target\_label}
                \vspace{0.2em}
                \begin{itemize}
                    \item Tests if register \code{x0} is non-zero.
                    \item If $\code{x0} \neq 0 \rightarrow$ branch to label.
                    \item Highly efficient for loop termination.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    \begin{alertblock}{Microarchitectural Advantage}
        Fuses comparison and control-flow evaluation into a single instruction without clobbering condition flag state.
    \end{alertblock}
\end{frame}

% --- Slide 30: Topic Summary ---
\begin{frame}{Topic 2 Comprehensive Summary}
    \begin{itemize}
        \item \key{RISC Design Philosophy}: Emphasizes simple control circuitry, regular instruction formats, fixed-width encodings, and high energy efficiency.
        \item \key{Load/Store Architecture}: Strict isolation of memory access (\code{ldr}, \code{str}) from computation (\code{add}, \code{sub}).
        \item \key{AArch64 Register File}: 31 visible 64-bit registers (\code{x0}--\code{x30}), zero register \code{xzr}, frame pointer \code{x29}, and link register \code{x30} (\code{lr}).
        \item \key{Dedicated Program Counter (\code{pc})}: \code{pc} is not a general-purpose register and cannot be accessed as an arithmetic operand.
        \item \key{PSTATE and Flags}: The \code{cmp} instruction updates \key{N, Z, C, V} flags to drive conditional branches (\code{b.eq}, \code{b.ne}, \code{b.lt}).
        \item \key{Instruction Classes}: All AArch64 instructions partition into Data Processing, Memory Access, or Control Flow.
    \end{itemize}
\end{frame}

% --- Slide 31: Self-Test: Questions 1 & 2 ---
\begin{frame}{Self-Test: Questions 1 \& 2}
    \begin{block}{Question 1: RISC vs.\ CISC Principles}
        What are the main architectural principles that distinguish RISC from CISC? Why does simpler instruction decoding reduce processor power consumption?
    \end{block}
    \vspace{0.4em}
    \begin{block}{Question 2: The Load/Store Sequence}
        How does the AArch64 instruction sequence below implement an update to a 64-bit memory variable? Why can't ARM perform this operation in a single arithmetic instruction?
        \begin{center}
            \code{ldr x1, [x0]} \quad $\rightarrow$ \quad \code{add x1, x1, \#5} \quad $\rightarrow$ \quad \code{str x1, [x0]}
        \end{center}
    \end{block}
\end{frame}

% --- Slide 32: Self-Test: Solutions 1 & 2 ---
\begin{frame}{Self-Test: Solutions 1 \& 2}
    \begin{block}{Solution 1}
        RISC uses fixed 32-bit instructions, a load/store model, and regular operand fields. CISC uses variable-length encodings and direct memory arithmetic. Simpler decoding reduces silicon gate count and dynamic switching activity ($P \propto C \cdot V^2 \cdot f$), lowering power draw.
    \end{block}
    \vspace{0.3em}
    \begin{block}{Solution 2}
        The sequence explicitly stages memory mutation: \code{ldr} loads 64 bits into \code{x1}, \code{add} computes in the ALU, and \code{str} writes back to RAM. ARM cannot combine these because the ALU datapath is isolated from memory buses, adhering to the pure RISC load/store principle.
    \end{block}
\end{frame}

\begin{frame}{Self-Test: Questions 3 \& 4}
    \begin{block}{Question 3: AArch64 Registers and Special Roles}
        How many general-purpose registers does AArch64 expose? What are the functions of \code{xzr}, \code{sp}, \code{x30} (\code{lr}), and how is \code{pc} protected?
    \end{block}
    \vspace{0.4em}
    \begin{block}{Question 4: Instruction Classes}
        What are the three core instruction classes in AArch64? Give one specific instruction example for each class.
    \end{block}
\end{frame}

% --- Slide 34: Self-Test: Solutions 3 \& 4 ---
\begin{frame}{Self-Test: Solutions 3 \& 4}
    \begin{block}{Solution 3}
        \begin{itemize}
            \item \code{x0}--\code{x30}: 31 general-purpose 64-bit registers.
            \item \code{xzr}: Zero register (reads zero, writes discarded); \code{sp}: Dedicated Stack Pointer.
            \item \code{x30} (\code{lr}): Link Register holding return addresses from \code{bl}.
            \item \code{pc} is \textbf{not} a general register and can only be updated via control-flow branches.
        \end{itemize}
    \end{block}
    \vspace{0.3em}
    \begin{block}{Solution 4}
        \begin{enumerate}
            \item \key{Data Processing}: \code{add x3, x1, x2} (computes within registers).
            \item \key{Memory Access}: \code{ldr x1, [x2]} (loads 64-bit word from memory).
            \item \key{Control Flow}: \code{b.eq target} or \code{cbz x0, target} (branching).
        \end{enumerate}
    \end{block}
\end{frame}

% --- Slide 35: Self-Test: Question 5 & Solution ---
\begin{frame}{Self-Test: Question 5 \& Solution}
    \begin{block}{Question 5: Compare-and-Branch Execution}
        How do \code{cmp} and a conditional branch such as \code{b.lt} work together to implement high-level decision making in AArch64? What status field stores the intermediate outcome?
    \end{block}
    \vspace{0.3em}
    \pause
    \begin{block}{Solution 5}
        \begin{itemize}
            \item \code{cmp x0, x1} performs internal subtraction ($\code{x0} - \code{x1}$), discards the result, and updates the \key{N, Z, C, V} condition flags in \key{PSTATE}.
            \item \code{b.lt label} subsequently inspects the flags ($N \neq V$). If true, it loads the label address into \code{pc}; otherwise, it falls through to the next sequential instruction.
            \item Intermediate comparison outcome is stored in \key{PSTATE.NZCV}.
        \end{itemize}
    \end{block}
\end{frame}
