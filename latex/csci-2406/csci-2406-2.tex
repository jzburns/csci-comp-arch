% ============================================================
% Slide: What Does RISC Mean?
% ============================================================

\begin{frame}{What Does RISC Mean?}

\begin{block}{Reduced Instruction Set Computer}
RISC is a design philosophy emphasizing
\key{regularity},
\key{simple instruction forms},
and a clean separation between computation
and memory access.
\end{block}

\vspace{0.15cm}

\begin{itemize}
  \item RISC does \key{not} simply mean
        ``a processor with very few instructions.''
  \item ``Reduced'' is better understood as
        \key{reduced instruction complexity and irregularity}.
  \item Arithmetic and logical operations
        are primarily register-based.
  \item Load and store instructions
        transfer data between registers and memory.
\end{itemize}

\end{frame}

% ============================================================
% Slide: Foundational RISC Design Principles
% ============================================================

\begin{frame}{Foundational RISC Design Principles}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\begin{block}{Instruction Regularity}
\begin{itemize}
  \item Regular instruction formats.
  \item Consistent operand fields.
  \item Predictable instruction alignment.
\end{itemize}
\end{block}

\column{0.48\textwidth}

\begin{block}{Register-Centered Computation}
\begin{itemize}
  \item Arithmetic uses registers.
  \item Memory access uses loads and stores.
  \item Regularity simplifies decoding.
\end{itemize}
\end{block}

\end{columns}

\vspace{0.15cm}

\begin{block}{Implementation Benefit}
These characteristics support regular
instruction fetch,
decode,
and pipelined implementation.
\end{block}

\end{frame}

% ============================================================
% Slide: The Load/Store Architectural Model
% ============================================================

\begin{frame}{The Load/Store Architectural Model}

\begin{block}{Separation of Memory Access and Computation}
In a load/store architecture such as AArch64,
arithmetic and logical instructions operate on registers.
Memory is accessed using dedicated
\key{load}
and
\key{store}
instructions.
\end{block}

\vspace{0.15cm}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\begin{block}{Data Processing}
\begin{itemize}
  \item Operands are registers.
  \item Results are written to registers.
\end{itemize}

\begin{center}
\large
\code{add x1, x2, x3}
\end{center}
\end{block}

\column{0.48\textwidth}

\begin{block}{Memory Access}
\begin{itemize}
  \item Loads read memory into registers.
  \item Stores write registers to memory.
\end{itemize}

\begin{center}
\large
\code{ldr x1, [x0]}
\end{center}
\end{block}

\end{columns}

\end{frame}

that must be increased by 5.

The \code{add} instruction
does not directly read
or modify a memory operand.
\end{block}

% Slide: AArch64 vs. x86-64: Instruction Length
% ============================================================

\begin{frame}{AArch64 vs. x86-64: Instruction Length}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\begin{block}{AArch64 / A64}

\begin{center}
\large
\key{32 bits = 4 bytes}
\end{center}

\begin{itemize}
  \item Fixed instruction width.
  \item Instructions are 4-byte aligned.
  \item Boundaries are known immediately.
\end{itemize}

\end{block}

\column{0.48\textwidth}

\begin{block}{x86-64}

\begin{center}
\large
\key{1 to 15 bytes}
\end{center}

\begin{itemize}
  \item Variable instruction length.
  \item Length depends on the encoding.
  \item Boundaries must be discovered.
\end{itemize}

\end{block}

\end{columns}

\vspace{0.15cm}

\begin{block}{What a 16-Byte Fetch Might Contain}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\vspace{0.05cm}

\centering

\textbf{A64}

\vspace{0.05cm}

\code{4B | 4B | 4B | 4B}

\vspace{0.05cm}

Exactly four instructions.

\column{0.48\textwidth}

\centering

\vspace{0.05cm}

\textbf{x86-64}

\code{1B | 4B | 2B | 7B | 2B}

Five instructions of different lengths.

\end{columns}

\end{block}

\vspace{0.05cm}

\begin{center}
\key{A64 starts with known boundaries; x86-64 must determine them.}
\end{center}

\end{frame}

% ============================================================
% Slide: How RISC Simplicity Shapes the Hardware
% ============================================================

\begin{frame}{How RISC Simplicity Shapes the Hardware}

\begin{block}{Reduced Front-End Irregularity}
A regular ISA can reduce some of the circuitry
required to \key{find}, \key{align}, and \key{decode}
instructions.
\end{block}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\begin{block}{Potential Hardware Benefits}
\begin{itemize}
  \item Simpler instruction-boundary detection.
  \item More regular decode paths.
  \item Less handling of legacy encoding cases.
  \item Potentially lower front-end area
        and switching activity.
\end{itemize}
\end{block}

\column{0.48\textwidth}

\begin{block}{But Not a Simple CPU}

Modern AArch64 cores may still contain:

\begin{itemize}
  \item out-of-order execution
  \item register renaming
  \item branch predictors
  \item large caches
  \item SIMD/vector units
\end{itemize}

\end{block}

\end{columns}

\vspace{0.1cm}

\begin{center}
\key{RISC reduces ISA irregularity, not necessarily total CPU complexity.}
\end{center}

\end{frame}

% ============================================================
% Slide: Why AArch64 Fits Mobile and Energy-Constrained Systems
% ============================================================

\begin{frame}{Why AArch64 Fits Mobile and Energy-Constrained Systems}

AArch64 provides a regular architectural foundation
that works well under tight power and thermal budgets.

\vspace{-0.2cm}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\begin{block}{Architectural Characteristics}
\begin{itemize}
  \item Fixed 32-bit A64 instructions.
  \item Regular instruction alignment.
  \item Explicit load/store memory access.
  \item Large register set.
  \item Comparatively regular decoding.
\end{itemize}
\end{block}

\column{0.48\textwidth}

\begin{block}{System-Level Advantages}
\begin{itemize}
  \item Efficient processor front ends.
  \item Strong performance-per-watt designs.
  \item Highly integrated ARM-based SoCs.
  \item Long-established mobile ecosystem.
  \item Hardware can be specialized
        for a tight energy budget.
\end{itemize}
\end{block}

\end{columns}

\vspace{-0.1cm}

\begin{block}{Important Qualification}
AArch64 does not guarantee lower power.
Energy consumption depends on the complete
microarchitecture, semiconductor process,
clock frequency, voltage, caches, and workload.
\end{block}

\end{frame}

\begin{frame}{Same Operation, Different ISA Style}

Suppose a 64-bit memory value
must be increased by 5.

\vspace{0.15cm}

\begin{columns}[T,onlytextwidth]

\column{0.46\textwidth}

\begin{block}{x86-64 Style}

Some x86-64 instructions permit
memory operands directly:

\begin{center}

\begin{minipage}{0.9\textwidth}
\raggedright
\large
\code{add qword ptr [rax], 5}
\end{minipage}

\end{center}

One instruction describes
both memory access and arithmetic.

\end{block}

\column{0.50\textwidth}

\begin{block}{AArch64 Load/Store Style}

\begin{center}

\begin{minipage}{0.62\textwidth}
\raggedright
\large
\code{ldr x1, [x0]}\\[0.15cm]
\code{add x1, x1, \#5}\\[0.15cm]
\code{str x1, [x0]}
\end{minipage}

\end{center}

Memory transfer and arithmetic
are represented by separate instructions.

\end{block}

\end{columns}

\end{frame}

% ============================================================
% Slide: ARM and AArch64
% ============================================================

\begin{frame}{ARM and AArch64}

\begin{block}{ARM Architecture}
ARM defines a family
of instruction set architectures
used in embedded systems,
mobile computing,
desktop systems,
and servers.
\end{block}

\vspace{0.15cm}

\begin{itemize}
  \item \key{AArch64}
        is the 64-bit execution state
        introduced with ARMv8-A.
  \item It provides a modern
        64-bit programmer-visible register model.
  \item It follows a load/store design
        with regular instruction formats.
  \item Different processors can implement AArch64
        using different microarchitectures.
\end{itemize}

\end{frame}

% ============================================================
% Slide: AArch64 and A64 Terminology
% ============================================================

\begin{frame}{AArch64 and A64 Terminology}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\begin{block}{AArch64}
\begin{itemize}
  \item The architecture / execution state.
  \item Defines programmer-visible state.
  \item Includes registers,
        memory behavior,
        and instruction semantics.
\end{itemize}
\end{block}

\column{0.48\textwidth}

\begin{block}{A64}
\begin{itemize}
  \item The instruction set encoding.
  \item Used while executing
        in AArch64 state.
  \item A64 instructions
        are fixed 32-bit words.
\end{itemize}
\end{block}

\end{columns}

\vspace{0.15cm}

\begin{block}{Course Terminology}
Use \key{AArch64} by default.
Use \key{A64}
only when referring specifically
to the instruction set encoding.
\end{block}

\end{frame}

% ============================================================
% Slide: Key Characteristics of AArch64
% ============================================================

\begin{frame}{Key Characteristics of AArch64}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\begin{block}{Instruction Model}
\begin{itemize}
  \item Load/store architecture.
  \item A64 instructions are 32 bits wide.
  \item Instructions are 4-byte aligned.
  \item Register fields are regular.
\end{itemize}
\end{block}

\column{0.48\textwidth}

\begin{block}{Register Model}
\begin{itemize}
  \item 31 general-purpose registers.
  \item 64-bit \code{x} register views.
  \item 32-bit \code{w} register views.
  \item Special roles for
        \code{sp},
        \code{xzr},
        PC state,
        and NZCV.
\end{itemize}
\end{block}

\end{columns}

\end{frame}

\begin{block}{Fixed-Width Encoding}
Every instruction
in the A64 instruction set
occupies exactly \key{32 bits}.
\end{block}

\vspace{0.2cm}

\begin{center}

\begin{tikzpicture}[

  }

\end{center}

\vspace{0.2cm}

\begin{itemize}
  \item Fixed width simplifies
        instruction alignment and initial decode.
  \item Exact bit-field layouts
        vary by instruction class.
\end{itemize}

\vspace{0.1cm}

\small

Detailed machine-code field layouts
are covered later
in the instruction-encoding topic.

\end{frame}

\begin{frame}{Dual Register Views: \code{X} and \code{W}}

Each general-purpose register
can be accessed using either
a 64-bit or 32-bit architectural view.

\vspace{0.15cm}

\begin{center}

\begin{tikzpicture}[
  scale=.82,

\draw[

\draw[

]

rectangle

{\code{wn}: bits [31:0]};

\end{tikzpicture}

\end{center}

\vspace{0.15cm}

\begin{block}{32-Bit Write Rule}
Writing to a \code{wn} destination
automatically clears bits [63:32]
of the corresponding \code{xn} register.
\end{block}

\end{frame}

% ============================================================
% Slide: Example: 32-Bit Write Clears Upper Bits
% ============================================================

\begin{frame}{Example: 32-Bit Write Clears Upper Bits}

\begin{block}{Instruction}

\begin{center}
\large
\code{add w3, w1, w2}
\end{center}

\end{block}

\vspace{0.15cm}

\begin{itemize}
  \item The operation uses
        the lower 32-bit views:
        \code{w1}
        and
        \code{w2}.
  \item The 32-bit result
        is written into
        \code{w3}.
  \item The upper half
        of
        \code{x3}
        is cleared automatically.
\end{itemize}

\vspace{0.15cm}

\begin{center}

\begin{minipage}{0.40\textwidth}
\raggedright
\code{x3[63:32] <- 0}\\[0.15cm]
\code{x3[31:0] <- result}
\end{minipage}

\end{center}

\end{frame}

% ============================================================
% Slide: AArch64 General-Purpose Registers
% ============================================================

\begin{frame}{AArch64 General-Purpose Registers}

\begin{block}{Register Set}
AArch64 provides
\key{31 general-purpose registers},
named
\code{x0}
through
\code{x30}.
Each register is 64 bits wide.
\end{block}

\vspace{-0.1cm}

\begin{columns}[T,onlytextwidth]

\column{0.47\textwidth}

\begin{block}{64-Bit View}

\begin{center}
\large
\code{x0}--\code{x30}
\end{center}

\begin{itemize}
  \item Each \code{xn}
        holds 64 bits.
  \item Used for integer values,
        addresses,
        and general computation.
\end{itemize}

\end{block}

\column{0.47\textwidth}

\begin{block}{32-Bit View}

\begin{center}
\large
\code{w0}--\code{w30}
\end{center}

\begin{itemize}
  \item Each \code{wn}
        refers to bits [31:0]
        of \code{xn}.
  \item Writing \code{wn}
        clears bits [63:32].
\end{itemize}

\end{block}

\end{columns}

\vspace{-0.1cm}

\begin{block}{Important}
The ISA treats
\code{x0}--\code{x30}
as general-purpose registers.
Software conventions assign
additional roles during procedure calls.
\end{block}

\end{frame}

% ============================================================
% Slide: The Zero Register
% ============================================================

\begin{frame}{The Zero Register}

\begin{block}{\code{xzr} / \code{wzr}}
The zero register has two special behaviors:
\key{reads return zero}
and
\key{writes are discarded}.
\end{block}

\vspace{-0.25cm}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\begin{block}{Use as a Zero Operand}

\begin{center}
\large
\code{mov x0, xzr}
\end{center}

\begin{itemize}
  \item Reading \code{xzr}
        produces the value \code{0}.
  \item Therefore,
        \code{x0} receives zero.
\end{itemize}

\end{block}

\column{0.48\textwidth}

\begin{block}{Use as a Discard Destination}

\begin{center}
\large
\code{cmp x1, x2}
\end{center}

\begin{itemize}
  \item \code{cmp} is an alias for
        \code{subs xzr, x1, x2}.
  \item The subtraction result is discarded,
        while NZCV flags are updated.
\end{itemize}

\end{block}

\end{columns}

\vspace{0.1cm}

\begin{block}{Key Idea}
\code{xzr} behaves like a permanent zero value
when read,
and like a discard destination when written.
\end{block}

\end{frame}

% ============================================================
% Slide: The Stack Pointer
% ============================================================

\begin{frame}{The Stack Pointer}

\begin{block}{\code{sp}}
The stack pointer identifies
the current stack location
and has dedicated architectural semantics.
\end{block}

\vspace{0.15cm}

\begin{itemize}
  \item Used by stack-related
        load/store and arithmetic forms.
  \item Represents a memory address,
        not a normal scratch register.
  \item Software conventions impose
        alignment and usage requirements.
  \item Stack behavior is studied
        more deeply with procedures
        and function calls.
\end{itemize}

\vspace{0.15cm}

\begin{center}
\code{stp x29, x30, [sp, \#-16]!}
\end{center}

\end{frame}

% ============================================================
% Slide: Encoding Note: Register Number 31
% ============================================================

\begin{frame}{Encoding Note: Register Number 31}

\begin{block}{Context-Dependent Meaning}
In several A64 instruction formats,
register encoding value 31
has special meaning.
\end{block}

\vspace{0.15cm}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\begin{block}{Often Means Zero Register}

\begin{center}
\large
\code{xzr} / \code{wzr}
\end{center}

Common in data-processing contexts.

\end{block}

\column{0.48\textwidth}

\begin{block}{Sometimes Means Stack Pointer}

\begin{center}
\large
\code{sp} / \code{wsp}
\end{center}

Common in stack
and memory-addressing contexts.

\end{block}

\end{columns}

\vspace{0.15cm}

\small

The instruction definition
determines the meaning.

\end{frame}

% ============================================================
% Slide: The Program Counter
% ============================================================

\begin{frame}{The Program Counter}

\begin{block}{Program Counter State}
The program counter identifies
the instruction address
associated with the current flow
of execution.
\end{block}

\vspace{0.15cm}

\begin{itemize}
  \item Sequential execution advances
        to the next instruction.
  \item Branches and calls redirect execution
        to another address.
  \item Unlike AArch32 \code{r15},
        the AArch64 program counter
        is \key{not}
        an ordinary general-purpose register.
  \item AArch64 arithmetic instructions
        cannot use \code{pc}
        as a normal register operand.
\end{itemize}

\end{frame}

Some AArch64 instructions update
four condition flags
used by later operations.

\begin{itemize}

  \item \key{C}:
        carry-out for addition;

  \item \key{V}:
        signed arithmetic overflow.
\end{itemize}

\begin{frame}{Class 1: Data Processing}

Data-processing instructions
perform computation on register values.

\begin{center}

\small

\renewcommand{\arraystretch}{1.05}

\begin{tabular}{

{\color{white}\bfseries Instruction}
&
{\color{white}\bfseries Operation}

\\

\rowcolor{softgray}

\\

\end{tabular}

\begin{block}{Key Characteristic}
The operands are registers;

\end{block}

\begin{frame}{Class 2: Memory Access}

Memory instructions transfer data
between registers and memory.

\vspace{0.15cm}

\begin{center}

\begin{tikzpicture}[
  node distance=.4cm,

    rectangle,

  }

\node[

  membox,

\end{tikzpicture}

\end{center}

\vspace{0.15cm}

\begin{block}{Bracket Notation}
\code{[x2]}
means:
access memory using
the address contained in
\code{x2}.
\end{block}

\end{frame}

\begin{frame}{Class 3: Control Flow}

Control-flow instructions
change which instruction executes next.

\vspace{0.15cm}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\begin{block}{Unconditional}
\begin{itemize}
  \item \code{b target}:
        branch
  \item \code{bl function}:
        call
  \item \code{ret}:
        return
\end{itemize}
\end{block}

\column{0.48\textwidth}

\begin{block}{Conditional}
\begin{itemize}
  \item \code{b.eq target}:
        branch if equal
  \item \code{b.ne target}:
        branch if not equal
  \item Conditional branches
        may test NZCV flags.
\end{itemize}
\end{block}

\end{columns}

\end{frame}

% ============================================================

% ============================================================

\begin{frame}{Comparison and Branching}

\begin{block}{Common Pattern}

\begin{center}
\large
\code{cmp x0, x1}\\[0.15cm]
\code{b.eq equal}
\end{center}

\end{block}

\vspace{0.15cm}

\begin{itemize}
  \item \code{cmp}
        compares two register values.
  \item It updates condition flags
        rather than storing a data result.
  \item \code{b.eq}
        uses those flags
        to decide whether to branch.
\end{itemize}

\end{frame}

\begin{frame}{Topic 2 Summary}

\begin{itemize}
  \item \key{RISC Design}:
        emphasizes reduced instruction complexity,
        regular formats,
        register-based computation,

        ARM's 64-bit architectural execution state.

        \code{x0}--\code{x30},
        \code{sp},
        \code{xzr}/\code{wzr},
        PC state,
        and NZCV.
  \item \key{Instruction Classes}:
        Data Processing,

\end{itemize}
