\sectiondivider{PART 1}{Subroutine Linkage \& Control Flow}


\begin{frame}{The Link Register (\texttt{x30}) and Branch-with-Link (\texttt{bl})}

\begin{itemize}\setlength{\itemsep}{6pt}
  \item The \key{caller} invokes another function, the \key{callee}, and must be able to resume after it returns.
  \item \key{Branch with Link (\code{bl label}):} Atomically performs two operations:
  \begin{enumerate}\setlength{\itemsep}{2pt}
    \item Saves return address ($\text{PC} + 4$) into the \key{Link Register (\bits{x30} / \bits{lr})}.
    \item Branches to target function: $\text{PC} \leftarrow \text{label}$.
  \end{enumerate}
\end{itemize}

\vspace{0.2cm}
\centering
\begin{tikzpicture}[scale=0.9, every node/.style={transform shape}]
  \node[draw=navy, fill=lightblue, rounded corners=2pt, minimum width=3.8cm, minimum height=0.6cm, font=\small\bfseries] (save) {1. $x_{30} \leftarrow \text{PC} + 4$};
  \node[draw=gold!80!black, fill=gold!20, line width=1pt, rounded corners=2pt, right=1.2cm of save, minimum width=3.2cm, minimum height=0.6cm, font=\small\bfseries] (jump) {2. $\text{PC} \leftarrow \text{Target Address}$};
  \draw[-{Latex}, thick, navy] (save) -- (jump);
\end{tikzpicture}

\vspace{0.3cm}
\begin{block}{Direct and Indirect Calls}
\code{bl label} calls a PC-relative target. For a function address held in a register, \code{blr x2} calls through \bits{x2}; it also writes the return address to \bits{x30}.
\end{block}
\end{frame}


\begin{frame}{Subroutine Return Mechanics (\texttt{ret})}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item \key{Return from Subroutine (\code{ret}):} Branches to the address held in a register. With no explicit operand, \code{ret} uses the link register: $\text{PC} \leftarrow x_{30}$.
  \item A normal \code{bl} / \code{ret} pair therefore resumes at the instruction following the original call.
\end{itemize}

\vspace{-0.05cm}
\begin{block}{Example: Minimal Leaf Return Sequence}
\small
\begin{tabular}{ll}
\code{caller\_func:} & \\
\quad \code{bl  helper\_func}  & \text{\scriptsize // x30 = return address; branch to helper} \\
\quad \code{mov x1, x0}        & \text{\scriptsize // resumes here after ret} \\
\code{helper\_func:} & \\
\quad \code{add x0, x0, \#1}  & \text{\scriptsize // compute result} \\
\quad \code{ret}               & \text{\scriptsize // default target register is x30} \\
\end{tabular}
\end{block}
\end{frame}

\begin{block}{Control Flow}
\code{bl square} writes the address of the next instruction into \bits{x30} and branches to \code{square}. The callee computes the result in \bits{x0}; \code{ret} resumes at the instruction after \code{bl}.
\end{block}

\begin{frame}{Leaf vs.\ Non-Leaf Functions}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{block}{Leaf Functions}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Call \key{no other functions}.
  \item \bits{x30} is not overwritten by a nested \code{bl}.
  \item May still use stack space for local data, but need not save \bits{x30} solely for linkage.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Non-Leaf Functions}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Call one or more \key{nested subroutines}.
  \item A nested \code{bl} overwrites \bits{x30} with the nested return address.
  \item Preserve the incoming return address before that happens if it will be needed later.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\vspace{0.2cm}
\begin{block}{The Non-Leaf Rule}
If a function makes another call and must later return to its own caller, its incoming return address must be preserved before \bits{x30} is overwritten.
\end{block}
\end{frame}


\sectiondivider{PART 2}{Procedure Call Standard \& Register Roles}


\begin{frame}{Why a Calling Convention Is Needed}
\small
\setbeamerfont{itemize/enumerate body}{size=\small}
\begin{block}{Separately Compiled Code Must Agree}
A program may combine separately compiled C, assembly, and libraries. These components need shared rules: \code{bl} transfers control and records a return address, but does not define argument locations or register preservation.
\end{block}
\begin{itemize}\setlength{\itemsep}{6pt}
\item \key{ABI --- Application Binary Interface:} rules that let compiled components work together, including data layout, procedure calls, and object-file conventions.
\item \key{AAPCS64 --- Procedure Call Standard for the Arm 64-bit Architecture:} the calling convention within the Arm ABI: where arguments/results go, which registers survive calls, and how the stack is aligned.
\item \key{Example:} for a function adding two 64-bit integers, both sides agree that \bits{x0}/\bits{x1} carry the arguments and \bits{x0} returns the result. Disagreement can corrupt data or execution.
\end{itemize}
\vspace{0.15cm}
\small
\key{ISA vs.\ ABI:} A64 defines instruction behavior; the ABI defines how compiled components cooperate.
\end{frame}


\begin{frame}{Integer Arguments and Return Values}
\begin{itemize}\setlength{\itemsep}{8pt}
\item \key{Arguments:} ordinary 64-bit scalar integer and pointer arguments use \bits{x0}--\bits{x7}, subject to AAPCS64 classification rules. Ordinary 32-bit integers use the corresponding \bits{w} registers.
\item \key{Return value:} a 64-bit integer or pointer result normally uses \bits{x0}; a 32-bit integer result uses \bits{w0}.
\item \key{Beyond the registers:} additional arguments assigned to memory use a stack argument area prepared by the caller. The stack is defined in the next section.
\end{itemize}
\vspace{0.18cm}
\begin{block}{Example: Two Arguments, One Result}
\code{add x0, x0, x1}\\[0.10cm]
\code{ret}
\end{block}
\small
The incoming values in \bits{x0} and \bits{x1} are added; the result returns in \bits{x0}. Floating-point and aggregate classification are outside this example.
\end{frame}


\begin{frame}{Caller-Saved and Callee-Saved Registers}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{block}{Caller-Saved: \bits{x0}--\bits{x17}}
\begin{itemize}\setlength{\itemsep}{6pt}
\item A call may overwrite these registers.
\item The caller must preserve values it still needs after the call.
\item \bits{x0}--\bits{x7}: arguments/results; \bits{x8}: indirect result address when required; \bits{x9}--\bits{x17}: temporary uses, including IP0/IP1 in \bits{x16}/\bits{x17}.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Callee-Saved: \bits{x19}--\bits{x29}}
\begin{itemize}\setlength{\itemsep}{6pt}
\item A function that modifies one of these registers must restore its incoming value before returning.
\item Values kept in them survive ordinary calls.
\item \bits{x29} has the additional conventional role of frame pointer (\bits{fp}).
\end{itemize}
\end{block}
\end{column}
\end{columns}
\vspace{0.18cm}
\key{Example:} a loop counter needed after a call can use \bits{x19}, provided the function preserves its caller's original \bits{x19}.
\end{frame}


\begin{frame}{Special-Purpose ABI Registers}

\begin{itemize}\setlength{\itemsep}{7pt}
  \item \bits{x29} / \bits{fp}: frame pointer: a stable reference for accessing saved state and local storage on the stack.
  \item \bits{x30} / \bits{lr}: link register; \code{bl} and \code{blr} write the return address here.
  \item \bits{sp}: stack pointer; marks the active stack boundary and must stay properly aligned.
  \item \bits{x18}: platform register; do not use it in hand-written examples unless the target platform says it is available.
\end{itemize}


\end{frame}


\sectiondivider{PART 3}{The Runtime Stack \& Stack Frames}


\begin{frame}{The Runtime Stack}
\small
\begin{block}{Definition}
The runtime stack is a \key{per-thread data structure in memory} for temporary function-call state. The OS and runtime establish its memory region. Each thread has its own stack pointer (\bits{sp}).
\end{block}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.57\textwidth}
\begin{itemize}\setlength{\itemsep}{5pt}
  \item Saved return address (\bits{lr}/\bits{x30}) and caller's frame pointer (\bits{fp}/\bits{x29}), when needed.
  \item Other saved register values.
  \item Stack arguments, e.g., integer arguments beyond the first eight register arguments.
  \item Short-lived local variables, arrays, and temporary values that need memory.
\end{itemize}
\end{column}
\begin{column}{0.39\textwidth}
\centering
\begin{tikzpicture}[
  every node/.style={font=\scriptsize\ttfamily,inner sep=4pt},
  stackrow/.style={draw=navy,minimum width=3.65cm,minimum height=0.65cm,align=center}
]
\node at (1.825,0.35) {Higher addresses};
\node[stackrow,fill=lightblue] at (1.825,-0.325) {Earlier call data};
\node[stackrow,fill=gold!20] at (1.825,-0.975) {Current call data};
\node[stackrow,draw=navy,dashed,fill=softgray] at (1.825,-1.625) {Available stack space};

\node at (1.825,-2.3) {Lower addresses};
\end{tikzpicture}
\end{column}
\end{columns}
\vspace{0.15cm}
\small
\key{Direction:} AAPCS64 uses a descending stack: growth is from high to low addresses. This is a convention, not a property of every stack data structure.
\end{frame}

\begin{frame}{Where Does the Compiler Store Local Variables?}
\small
\begin{block}{Compiler Rule}
The compiler chooses storage while preserving observable behavior. A local declaration does not require a stack slot.
\end{block}
\begin{itemize}\setlength{\itemsep}{7pt}
\item \key{Register:} keep a value in an available register when no memory address is needed.
\item \key{Stack storage:} typically holds addressable locals, arrays that need memory, and values displaced from registers (\key{spills}).
\item \key{No storage:} eliminate unused variables, substitute constants, or recompute values.
\end{itemize}
\vspace{0.15cm}
\small
\key{Taking an address:} a local whose address is passed to a function that accesses it needs memory, usually a stack slot, unless optimization removes that need.
\vspace{0.12cm}

\key{Build settings:} unoptimized builds commonly use more stack slots; optimization can reduce them.
\end{frame}

\begin{frame}{Stack Lifetime, Balance, and Limits}
\begin{itemize}\setlength{\itemsep}{7pt}
\item \key{Short-lived storage:} call-related data remains live while needed by the function. Nested callees can allocate more space while the caller's data remains allocated.
\item \key{Balance before return:} release all space allocated by the function and restore required saved registers. \bits{sp} must equal its entry value; several allocations can be released together.
\item \key{Stack leak:} unreleased allocations that accumulate consume capacity. An incorrect \bits{sp} at return can also corrupt execution.
\item \key{Finite capacity:} each thread's stack limit depends on the OS, executable, and runtime configuration. Deep recursion or large locals can cause stack overflow.
\item \key{After release:} the space can be reused, so pointers to former locals must not be used. Releasing space does not erase its bytes.
\end{itemize}
\end{frame}




\begin{frame}{Allocating and Releasing Stack Space}
\large
\begin{itemize}\setlength{\itemsep}{6pt}
  \item Stack storage is reserved and released by changing the stack pointer (\bits{sp}).
  \item \key{Stack behavior:}
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item Grows \textbf{downward} from high memory addresses toward low memory addresses.
    \item Stack Pointer (\bits{sp}) points to the current \key{lowest active address} (top of stack).
    \item Follows \key{Last-In, First-Out (LIFO)} order: the most recently allocated storage is released first.
  \end{itemize}
\end{itemize}

\vspace{0.22cm}
\begin{block}{Allocation vs.\ Deallocation}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item \key{Allocate:} subtract from \bits{sp}, e.g., \code{sub sp, sp, \#32}.
  \item Here \code{\#32} means \key{32 bytes}: room for four 64-bit \bits{x} register slots.
  \item \key{Release:} add the same amount back, e.g., \code{add sp, sp, \#32}.
\end{itemize}
\end{block}
\end{frame}


\begin{frame}{What Is a Stack Frame?}

\begin{itemize}\setlength{\itemsep}{6pt}
  \item Each function call may reserve a region of the runtime stack called a \key{stack frame}.
  \item A stack frame can hold:
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item an optional \key{frame record}: the 16-byte pair holding saved old \bits{x29} and saved \bits{x30},
    \item other saved callee-saved registers and space for outgoing stack arguments,
    \item local variables, temporary storage, and alignment padding.
  \end{itemize}
  \item The stack pointer (\bits{sp}) identifies the current stack boundary and changes as stack storage is allocated or released.
\end{itemize}

\vfill

\begin{block}{Frame Pointer (fp)}
\bits{x29} (fp) provides a \key{stable reference within the frame}, not necessarily its beginning. When a frame record is maintained, fp points to that record. Its address remains fixed while \bits{sp} tracks allocations and releases.
\end{block}

\end{frame}


\begin{frame}{Frame Pointer (\texttt{x29}) and Frame Records}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.55\textwidth}
\begin{itemize}\setlength{\itemsep}{5pt}
  \item The \key{stack frame} is the whole allocation; the \key{frame record} is a 16-byte part of it.
  \item The record has two 64-bit fields:
  \begin{itemize}\setlength{\itemsep}{2pt}
    \item \code{[x29]}: saved previous \bits{x29}
    \item \code{[x29, \#8]}: saved \bits{x30}
  \end{itemize}
  \item fp points to the saved old \bits{x29} slot. Its stored value points to the caller's frame record, forming a chain.
\end{itemize}
\end{column}
\begin{column}{0.4\textwidth}
\centering

\begin{frame}{Example Stack Frame Structure}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.52\textwidth}
\begin{itemize}\setlength{\itemsep}{5pt}
  \item A \key{stack frame} stores the stack-resident state associated with one function invocation.
  \item In the layout used here:
  \begin{enumerate}\setlength{\itemsep}{2pt}
    \item saved link register (\bits{x30}) and frame pointer (\bits{x29}),
    \item saved callee-saved registers, such as \bits{x19} in the diagram, come below it,
    \item local variables and buffers extend farther downward.
  \end{enumerate}
  \item Arrows identify slots; pointers hold their lowest byte addresses.
\end{itemize}
\end{column}

\begin{frame}{Stack Alignment and Padding Rules}
\begin{itemize}\setlength{\itemsep}{4pt}
  \item AAPCS64 requires \bits{sp} to satisfy 16-byte alignment at public interfaces and when used for stack memory accesses.
  \item Stack-frame allocation therefore rounds the required stack space up to a multiple of 16 bytes.
\end{itemize}
\vspace{0.1cm}
\small
\key{Padding} is unused space added for alignment. Round the total required bytes up to a multiple of 16: 36 bytes becomes 48 bytes.

\begin{frame}[fragile]{Function Prologue Mechanics}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item A \key{prologue} allocates stack storage and saves required state on function entry.
  \item Here the \bits{x29}/\bits{x30} frame record stays at the high-address end of this function's stack frame.
  \item Saving one 8-byte \bits{x} register still reserves 16 bytes so \bits{sp} remains aligned.
\end{itemize}

\vspace{0.05cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.55\textwidth}
\begin{block}{Prologue Template}
\footnotesize

my_func:

    stp    x29, x30, [sp, #-16]!

    // fp = address of saved old x29
    mov    x29, sp

    // save x19 below the frame record
    str    x19, [sp, #-16]!
\end{verbatim}
\end{block}

\begin{frame}[fragile]{Function Epilogue Mechanics}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item An \key{epilogue} restores saved state and releases stack storage before return.
  \item Restore any saved callee registers before restoring the frame record and returning.
\end{itemize}

\vspace{0.15cm}
\begin{block}{Epilogue Template}
\small
\begin{verbatim}
    // 1. Restore saved callee register; release its slot
    ldr    x19, [sp], #16

    // 2. Restore original fp and LR; release frame record
    ldp    x29, x30, [sp], #16

    // 3. Return to caller
    ret
\end{verbatim}
\end{block}

\small
\code{ldp} restores both the caller\textquotesingle s fp and the saved LR; \code{ret} then branches to LR (\bits{x30}). Post-indexing releases each allocation after loading it.
\end{frame}

    sub  x0, x0, #1

    mul  x0, x0, x19

\begin{frame}{Common Function Linkage Bugs in Assembly}
\begin{itemize}\setlength{\itemsep}{5pt}
  \item \key{1. Overwriting the Return Address:} A nested \code{bl} overwrites \bits{x30}; failing to preserve the incoming return address can return control to the wrong location.
  \item \key{2. Corrupting Callee-Saved Registers:} Modifying \bits{x19}--\bits{x28} without restoring them violates the calling convention.
  \item \key{3. Stack Misalignment:} Violating AAPCS64 stack-alignment rules breaks ABI assumptions and can fault where SP-alignment checking is enabled.
  \item \key{4. Unbalanced Stack Pointer:} Failing to restore \bits{sp} correctly can corrupt stack-resident state and later restores.
\end{itemize}
\end{frame}


\sectiondivider{SUMMARY \& ASSESSMENT}{Review and Self-Test Questions}


\begin{frame}{Summary}
\begin{itemize}\setlength{\itemsep}{5pt}
  \item \key{Linkage Primitives:} \code{bl}/\code{blr} write the return address to \bits{x30}; \code{ret} normally returns through \bits{x30}.
  \item \key{AAPCS64 Register Contract:}
  \begin{itemize}\setlength{\itemsep}{1.5pt}
    \item Scalar arguments commonly begin in \bits{x0}--\bits{x7}; most scalar results return in \bits{x0}.
    \item \bits{x0}--\bits{x17} are caller-saved; \bits{x19}--\bits{x28} are callee-saved.
  \end{itemize}
  \item \key{Stack Frames:} A stack frame holds saved linkage state, preserved registers, and local storage while maintaining required alignment.
  \item \key{Prologue/Epilogue:} Paired \code{stp}/\code{ldp} instructions commonly save and restore \bits{x29}/\bits{x30} while updating \bits{sp}.
\end{itemize}
\end{frame}


\begin{frame}{Self-Test Questions: Linkage \& AAPCS64}

\begin{enumerate}\setlength{\itemsep}{6pt}
  \item \key{Leaf vs.\ Non-Leaf Function Linkage:} How do leaf and non-leaf functions differ regarding link-register management and return-address preservation?

  \item \key{Register Preservation Contracts:} Suppose function \code{foo()} calls \code{bar()}. If \code{foo()} needs to maintain a loop counter value across the call, should it store that counter in \bits{x9} or \bits{x19}? Why?

  \item \key{Parameter Passing Limits:} How are integer parameters handled when calling a C function declared with 10 scalar arguments: \code{void test(int a1, ..., int a10)}?
\end{enumerate}

\end{frame}


\begin{frame}{Self-Test Questions: Stack Frame Management}

\begin{enumerate}\setlength{\itemsep}{6pt}
  \item \key{Stack Allocation Alignment:} A non-leaf function saves \bits{x29}/\bits{x30} (16 bytes) and needs 20 bytes of local buffer space. What is the minimum stack-frame size it must allocate to maintain AAPCS64 compliance?

  \item \key{Prologue/Epilogue Verification:} Identify the bug in the following epilogue:
  \small
  \begin{tabular}{l}
  \code{ldp x29, x30, [sp], \#16} \\
  \code{ldr x19, [sp, \#16]} \\
  \code{ret} \\
  \end{tabular}

  \item \key{Frame Pointer Chains:} Why need fp not point to the beginning of the whole frame? What do \code{[x29]} and \code{[x29, \#8]} contain, and how can a debugger follow the chain?
\end{enumerate}

\end{frame}
