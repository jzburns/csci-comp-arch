\sectiondivider{Section 01}{Procedure Abstraction and Call-Return Model}

% --- Slide 5: The Call-Return Lifecycle ---
\begin{frame}{The Subroutine Call-Return Lifecycle}
    Executing a procedure involves several distinct operational phases:
    \vspace{0.2em}
    \begin{enumerate}
        \item \textbf{Setup}: Caller places function arguments into designated parameter registers (\code{x0}--\code{x7}).
        \item \textbf{Transfer}: Branch to function label via \code{bl}, saving the return address in \code{x30} (\code{lr}).
        \item \textbf{Prologue}: Allocate stack space and save registers that must be preserved.
        \item \textbf{Execution}: Perform the core computation of the subroutine.
        \item \textbf{Epilogue}: Restore saved registers, deallocate stack space, and return via \code{ret}.
    \end{enumerate}
\end{frame}

% --- Slide 6: Caller vs. Callee Responsibilities ---
\begin{frame}{Caller vs. Callee Responsibilities}
    Maintaining modular integrity requires dividing responsibilities between functions:
    \vspace{0.2em}
    \begin{itemize}
        \item \key{The Caller Responsibilities}:
        \begin{itemize}
            \small
            \item Prepares and passes arguments in registers \code{x0}--\code{x7} (or on the stack if exceeding 8).
            \item Invokes the subroutine using \code{bl}, which saves the return address in \code{x30} (\code{lr}).
            \item Preserves any caller-saved volatile registers (\code{x0}--\code{x15}) across the call if their values are still needed.
        \end{itemize}
        \item \key{The Callee Responsibilities}:
        \begin{itemize}
            \small
            \item Allocates local stack frame space and preserves callee-saved registers (\code{x19}--\code{x29}) and \code{x30} if executing nested calls.
            \item Computes the result and places return values into \code{x0}.
            \item Restores saved registers, deallocates stack memory, and returns via \code{ret}.
        \end{itemize}
    \end{itemize}
\end{frame}

% --- Slide 7: C Function Invocation Mapping ---
\begin{frame}{Mapping C Functions to AArch64 Labels}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{C Function Example}
                \small
                \begin{tabbing}
                    \quad \code{int find\_max(int a, int b) \{} \\
                    \quad \quad \code{int max = (a > b) ? a : b;} \\
                    \quad \quad \code{return max;} \\
                    \quad \code{\}}
                \end{tabbing}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{AArch64 Translation Principle}
                \small
                \begin{itemize}
                    \item AArch64 assembly does not have native function declarations—only symbolic labels.
                    \item Arguments are passed in \code{x0} and \code{x1}.
                    \item The return value is placed in \code{x0}.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\sectiondivider{Section 02}{Subroutine Linkage (BL, RET, and Link Register)}

% --- Slide 9: Branch with Link (BL) ---
\begin{frame}{Branch with Link: The \code{bl} Instruction}
    \begin{block}{Syntax and Semantics}
        $$\code{bl function\_label}$$
        Performs a subroutine call by branching to the target label while simultaneously saving the return address (the instruction immediately following the \code{bl}) into the Link Register (\code{x30} / \code{lr}).
    \end{block}
    \vspace{0.3em}
    \begin{itemize}
        \item Essential for enabling nested function calls.
        \item The Link Register acts as a hardware-supported return address holder for the current scope.
    \end{itemize}
\end{frame}

% --- Slide 10: Subroutine Return (RET) ---
\begin{frame}{Returning from Subroutines: The \code{ret} Instruction}
    \begin{block}{Syntax and Semantics}
        $$\code{ret}$$
        Unconditionally branches back to the address held in the Link Register (\code{x30}), resuming execution in the calling function.
    \end{block}
    \vspace{0.3em}
    \begin{itemize}
        \item Equivalent to executing an indirect branch: \code{br x30}.
        \item Modern microarchitectures optimize \code{ret} using hardware return-stack predictors (unlike legacy \code{mov pc, lr}).
    \end{itemize}
\end{frame}

% --- Slide 11: Handling Nested Function Calls ---
\begin{frame}{Handling Nested Subroutine Calls}
    What happens if Function A calls Function B using \code{bl}?
    \vspace{0.2em}
    \begin{itemize}
        \item The \code{bl} to Function B overwrites the Link Register (\code{x30}) with Function B's return address, erasing Function A's return address.
        \item \key{The Solution}: Before calling another function, a callee must save its incoming \code{x30} value onto the stack during its prologue.
    \end{itemize}
    \vspace{0.2em}
    \begin{block}{Nested Call Assembly Pattern}
        \small
        \begin{tabbing}
            \quad \code{outer\_func:} \\
            \quad \quad \code{stp x29, x30, [sp, \#-16]!} \quad \= \textit{// Save outer return address (x30) on stack} \\
            \quad \quad \code{bl inner\_func} \> \textit{// Overwrites x30 with inner return address} \\
            \quad \quad \code{ldp x29, x30, [sp], \#16} \> \textit{// Restore original outer return address} \\
            \quad \quad \code{ret} \> \textit{// Correctly returns to caller of outer\_func}
        \end{tabbing}
    \end{block}
\end{frame}

\sectiondivider{Section 03}{Stack Organization, Activation Records, and Frame Pointers}

% --- Slide 13: Stack Architecture in AArch64 ---
\begin{frame}{The Stack Memory Structure}
    The system stack provides temporary workspace, register storage, and local variables:
    \vspace{0.2em}
    \begin{itemize}
        \item \key{Last-In, First-Out (LIFO)}: Operates like a stack of plates; data is pushed and popped strictly at the top.
        \item \key{Full-Descending Model}: The stack grows downward in memory toward lower addresses. The Stack Pointer (\code{sp} / \code{x31}) points directly to the last valid item pushed.
        \item \key{Strict Alignment}: The stack pointer must maintain 16-byte alignment at all times to prevent hardware faults.
    \end{itemize}
    \vspace{0.2em}
    \begin{block}{Stack vs. Heap Growth}
        The stack grows downward while the heap grows upward, maximizing available runtime memory efficiency within process address spaces.
    \end{block}
\end{frame}

% --- Slide 14: Frame Pointers and Stack Frames ---
\begin{frame}{Frame Pointers and Stack Activation Records}
    Each active function maintains a localized stack frame (activation record):
    \vspace{0.2em}
    \begin{itemize}
        \item \key{Frame Pointer (\code{x29})}: Points to the base of the current stack frame, anchoring local variables and saved registers.
        \item \key{Prologue Stack Growth}: \code{stp x29, x30, [sp, \#-16]!} reserves space and saves both frame pointer and link register.
        \item \key{Epilogue Restoration}: \code{ldp x29, x30, [sp], \#16} restores them and releases stack space.
    \end{itemize}
\end{frame}

% --- Slide 15: Stack Layout Diagram ---
\begin{frame}{An AArch64 Stack Frame Layout}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \begin{itemize}
                \item \textbf{Higher Addresses}: Caller's stack frame data.
                \item \textbf{Saved Frame Pointer (\code{x29})}: Links back to previous frame.
                \item \textbf{Saved Link Register (\code{x30})}: Return address to caller.
                \item \textbf{Local Variables}: Scratch space for function variables.
                \item \textbf{Lower Addresses}: Current Stack Pointer (\code{sp}).
            \end{itemize}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{block}{Stack Growth Direction ($\downarrow$)}
                \small
                \begin{tabular}{l}
                    \toprule
                    Higher Memory Addresses \\
                    \midrule
                    Caller Frame \\
                    \hline
                    Saved Frame Pointer (\code{x29}) \\
                    \hline
                    Saved Link Register (\code{x30}) \\
                    \hline
                    Local Variables / Buffers \\
                    \bottomrule
                \end{tabular}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\sectiondivider{Section 04}{Parameter Passing and Return Values}

% --- Slide 17: Argument Passing via Registers ---
\begin{frame}{Passing Function Arguments (\code{x0}--\code{x7})}
    To maximize speed, AArch64 passes arguments through general-purpose registers:
    \vspace{0.3em}
    \begin{itemize}
        \item \key{Registers \code{x0} through \code{x7}}: Designated for passing the first 8 integer/pointer arguments to a subroutine.
        \item \key{Overflow Arguments}: If a function takes more than 8 arguments, any additional parameters are pushed onto the stack by the caller.
    \end{itemize}
    \vspace{0.3em}
    \begin{block}{Order Convention}
        Argument 1 goes in \code{x0}, Argument 2 in \code{x1}, up to Argument 8 in \code{x7}.
    \end{block}
\end{frame}

% --- Slide 11: Handling Nested Function Calls ---
\begin{frame}{Handling Nested Subroutine Calls}
    What happens if Function A calls Function B using \code{bl}?
    \vspace{0.2em}
    \begin{itemize}
        \item The \code{bl} to Function B overwrites the Link Register (\code{x30}) with Function B's return address, erasing Function A's return address.
        \item \key{The Solution}: Before calling another function, a callee must save its incoming \code{x30} value onto the stack during its prologue.
    \end{itemize}
    \vspace{0.2em}
    \begin{block}{Nested Call Assembly Pattern}
        \small
        \begin{tabbing}
            \quad \code{outer\_func:} \\
            \quad \quad \code{stp x29, x30, [sp, \#-16]!} \quad \= \textit{// Save outer return address (x30) on stack} \\
            \quad \quad \code{bl inner\_func} \> \textit{// Overwrites x30 with inner return address} \\
            \quad \quad \code{ldp x29, x30, [sp], \#16} \> \textit{// Restore original outer return address} \\
            \quad \quad \code{ret} \> \textit{// Correctly returns to caller of outer\_func}
        \end{tabbing}
    \end{block}
\end{frame}

\sectiondivider{Section 05}{The Procedure Call Standard (AAPCS64)}

% --- Slide 20: Register Preservation Rules (AAPCS64) ---
\begin{frame}{Register Preservation Conventions (AAPCS64)}
    Functions share a common register file, requiring strict rules on preservation:
    \vspace{0.2em}
    \begin{itemize}
        \item \key{Caller-Saved (Volatile) --- \code{x0}--\code{x15}}: Can be freely overwritten by a callee. If a caller needs values in these registers to survive a function call, it must save them beforehand.
        \item \key{Callee-Saved (Non-Volatile) --- \code{x19}--\code{x29}}: Must be preserved across calls. If a callee modifies any of these registers, it must save their original values to the stack in its prologue and restore them in its epilogue.
    \end{itemize}
\end{frame}

% --- Slide 21: Complete Procedure Example ---
\begin{frame}{A Complete Function Prologue and Epilogue}
    Example of a callee function that allocates local stack space and preserves registers:
    \vspace{0.3em}
    \begin{tabbing}
        \quad \code{my\_func:} \\
        \quad \quad \code{stp x29, x30, [sp, \#-32]!} \quad \= \textit{// Allocate 32 bytes, save FP \& LR} \\
        \quad \quad \code{mov x29, sp} \> \textit{// Set up new frame pointer} \\
        \quad \quad \code{str x19, [x29, \#16]} \> \textit{// Save callee-saved register x19} \\
        \quad \quad \textit{// ... Function Body Execution ...} \\
        \quad \quad \code{ldr x19, [x29, \#16]} \> \textit{// Restore x19} \\
        \quad \quad \code{ldp x29, x30, [sp], \#32} \> \textit{// Restore FP \& LR, deallocate stack} \\
        \quad \quad \code{ret} \> \textit{// Return to caller}
    \end{tabbing}
\end{frame}

\sectiondivider{Section 06}{Summary \& Self-Test}

% --- Slide 23: Topic Summary ---
\begin{frame}{Topic 8 Comprehensive Summary}
    \begin{itemize}
        \item \key{Call-Return Model}: Procedures rely on standardized linkage to pass control, arguments, and return values safely.
        \item \key{Link Register}: \code{bl} stores return addresses in \code{x30} (\code{lr}), and \code{ret} uses them to jump back.
        \item \key{Stack Organization}: The stack grows downward, managed via \code{sp} and anchored by frame pointer \code{x29}.
        \item \key{Parameter Passing}: Registers \code{x0}--\code{x7} handle arguments; \code{x0} returns results.
        \item \key{AAPCS64 Conventions}: Strict caller-saved vs. callee-saved register preservation rules maintain modular safety.
    \end{itemize}
\end{frame}

% --- Slide 24: Self-Test: Questions 1 \& 2 ---
\begin{frame}{Self-Test: Questions 1 \& 2}
    \begin{block}{Question 1: The BL Instruction}
        What two actions are performed simultaneously when the processor executes a \code{bl target\_label} instruction?
    \end{block}
    \vspace{0.4em}
    \begin{block}{Question 2: Parameter Limits}
        How many integer or pointer arguments can be passed to a subroutine using registers alone under the AAPCS64 standard, and where do extra arguments go?
    \end{block}
\end{frame}

% --- Slide 25: Self-Test: Solutions 1 \& 2 ---
\begin{frame}{Self-Test: Solutions 1 \& 2}
    \begin{block}{Solution 1}
        It saves the return address (the next sequential instruction address) into the Link Register (\code{x30}) and branches execution to the target label.
    \end{block}
    \vspace{0.3em}
    \begin{block}{Solution 2}
        Up to 8 arguments can be passed via registers \code{x0} through \code{x7}. Any additional arguments exceeding 8 must be pushed onto the stack by the caller.
    \end{block}
\end{frame}

% --- Slide 26: Self-Test: Questions 3 \& 4 ---
\begin{frame}{Self-Test: Questions 3 \& 4}
    \begin{block}{Question 3: Callee-Saved Registers}
        Why must a subroutine preserve callee-saved registers (\code{x19}--\code{x29}) if it intends to modify them during execution?
    \end{block}
    \vspace{0.4em}
    \begin{block}{Question 4: Nested Function Calls}
        Why is it mandatory for a function that calls another subroutine to save the Link Register (\code{x30}) onto the stack during its prologue?
    \end{block}
\end{frame}

% --- Slide 27: Self-Test: Solutions 3 \& 4 ---
\begin{frame}{Self-Test: Solutions 3 \& 4}
    \begin{block}{Solution 3}
        To protect data belonging to the caller. If a callee overwrites a callee-saved register without restoring it, the calling function's data would become corrupted upon return.
    \end{block}
    \vspace{0.3em}
    \begin{block}{Solution 4}
        Executing a nested \code{bl} instruction overwrites the Link Register (\code{x30}) with a new return address, destroying the original return address needed to get back to the original caller.
    \end{block}
\end{frame}

% --- Slide 28: Self-Test: Questions 5 \& 6 ---
\begin{frame}{Self-Test: Questions 5 \& 6}
    \begin{block}{Question 5: Stack Alignment}
        What byte alignment constraint does AArch64 enforce on the Stack Pointer (\code{sp}) during procedure calls, and why is it required?
    \end{block}
    \vspace{0.4em}
    \begin{block}{Question 6: Frame Pointers}
        What role does the Frame Pointer (\code{x29}) play when managing activation records on the stack?
    \end{block}
\end{frame}

% --- Slide 29: Self-Test: Solutions 5 \& 6 ---
\begin{frame}{Self-Test: Solutions 5 \& 6}
    \begin{block}{Solution 5}
        The stack pointer must maintain strict 16-byte alignment to prevent hardware alignment faults and support wide load/store pair operations (\code{ldp}/\code{stp}).
    \end{block}
    \vspace{0.3em}
    \begin{block}{Solution 6}
        The Frame Pointer anchors the base of the current stack frame, allowing local variables and saved registers to be referenced via fixed positive or negative offsets.
    \end{block}
\end{frame}
