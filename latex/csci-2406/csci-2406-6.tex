% ============================================================
% Slide 4
% Slide Header: AArch64 Is a Load/Store Architecture
% ============================================================
\begin{frame}{AArch64 Is a Load/Store Architecture}
\begin{block}{Separation of Memory Access and Computation}
Arithmetic and logical instructions operate on register values. Dedicated load and store instructions transfer values between registers and memory.
\end{block}
\vspace{0.25cm}
\large
\hspace{3.5cm}\code{ldr x1, [x0]}\\[0.12cm]
\hspace{3.5cm}\code{add x1, x1, \#5}\\[0.12cm]
\hspace{3.5cm}\code{str x1, [x0]}
\normalsize
\vspace{0.25cm}
\begin{itemize}
\item \code{ldr} reads a value from memory into a register.
\item \code{add} changes the register value.
\item \code{str} writes the register value back to memory.
\end{itemize}
\end{frame}

% ============================================================
% Slide 5
% Slide Header: Memory Is Byte Addressed
% ============================================================
\begin{frame}{Memory Is Byte Addressed}
\begin{columns}[T,onlytextwidth]
\column{0.56\textwidth}
\begin{itemize}
\item Each memory address identifies one byte.
\item Multi-byte values occupy consecutive addresses.
\item The instruction determines how many bytes are transferred.
\item The base address for a load/store is held in a 64-bit base register or in \code{sp}.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Example}
A 64-bit value beginning at \hex{1000} occupies byte addresses \hex{1000} through \hex{1007}.
\end{block}
\column{0.40\textwidth}
\begin{center}
\small
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{c c}
\toprule
\textbf{Address} & \textbf{Byte} \\
\midrule
\hex{1000} & byte 0 \\
\hex{1001} & byte 1 \\
\hex{1002} & byte 2 \\
\multicolumn{2}{c}{$\vdots$} \\
\hex{1007} & byte 7 \\
\bottomrule
\end{tabular}
\end{center}
\end{columns}
\end{frame}

% ============================================================
% Slide 6
% Slide Header: Bracket Notation Means Memory Access
% ============================================================
\begin{frame}{Bracket Notation Means Memory Access}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Register Operand}
\large
\hspace{1.5cm}\code{add x0, x1, x2}
\normalsize
\vspace{0.1cm}
\begin{itemize}
\item Uses values already in registers.
\item No data-memory operand appears.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Memory Operand}
\large
\hspace{1.5cm}\code{ldr x0, [x1]}
\normalsize
\vspace{0.1cm}
\begin{itemize}
\item \code{x1} contains an address.
\item Brackets mean access memory at that address.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.25cm}
\begin{block}{Do Not Confuse Address and Data}
The value in \code{x1} is the address. The bytes stored at that address are the data transferred by the load or store.
\end{block}
\end{frame}

% ============================================================
% Slide 7
% Slide Header: Load Register
% ============================================================
\begin{frame}{Loading Data with \code{ldr}}
\begin{block}{64-Bit Load}
\large
\hspace{4.0cm}\code{ldr x0, [x1]}
\normalsize
\end{block}
\vspace{0.2cm}
\begin{itemize}
\item The effective address is the 64-bit value in \code{x1}.
\item Eight bytes are read from memory.
\item The 64-bit value is written into \code{x0}.
\end{itemize}
\vspace{0.15cm}
\begin{block}{32-Bit Load}
\code{ldr w0, [x1]} reads four bytes into \code{w0}. Any write to \code{w0} clears bits [63:32] of the corresponding \code{x0} register.
\end{block}
\end{frame}

% ============================================================
% Slide 8
% Slide Header: Store Register
% ============================================================
\begin{frame}{Storing Data with \code{str}}
\begin{block}{64-Bit Store}
\large
\hspace{4.0cm}\code{str x0, [x1]}
\normalsize
\end{block}
\vspace{0.2cm}
\begin{itemize}
\item The effective address is the value in \code{x1}.
\item The eight-byte value in \code{x0} is written to memory.
\item The source register is not changed by the store.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Narrow Stores}
\code{strb} writes the low 8 bits of a register, and \code{strh} writes the low 16 bits. \code{str w0, [x1]} writes the low 32 bits.
\end{block}
\end{frame}

% ============================================================
% Slide 10
% Slide Header: Load Width Determines Bytes Read
% ============================================================
\begin{frame}{Load Width Determines Bytes Read}
\begin{center}
\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{>{\raggedright\arraybackslash}m{3.0cm} >{\raggedright\arraybackslash}m{2.5cm} >{\raggedright\arraybackslash}m{5.6cm}}
\rowcolor{navy}
{\color{white}\bfseries Instruction} &
{\color{white}\bfseries Bytes Read} &
{\color{white}\bfseries Destination Result} \\
\rowcolor{softgray}
\code{ldrb w0, [x1]} &
1 byte &
Zero-extended to 32 bits in \code{w0} \\
\rowcolor{white}
\code{ldrh w0, [x1]} &
2 bytes &
Zero-extended to 32 bits in \code{w0} \\
\rowcolor{softgray}
\code{ldr w0, [x1]} &
4 bytes &
Written to \code{w0}; upper half of \code{x0} becomes zero \\
\rowcolor{white}
\code{ldr x0, [x1]} &
8 bytes &
Full 64-bit value written to \code{x0} \\
\end{tabular}
\end{center}
\vspace{0.25cm}
\begin{block}{Width Comes from the Instruction}
A memory address does not carry a type. The load/store instruction determines the number of bytes transferred.
\end{block}
\end{frame}

% ============================================================
% Slide 11
% Slide Header: Signed Loads Preserve Negative Values
% ============================================================
\begin{frame}{Signed Loads Preserve Negative Values}
\begin{block}{Sign Extension}
A signed load copies the sign bit of the narrow source into the new upper bits so the two's-complement value is preserved.
\end{block}
\vspace{0.15cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{>{\raggedright\arraybackslash}m{3.2cm} >{\raggedright\arraybackslash}m{2.4cm} >{\raggedright\arraybackslash}m{5.4cm}}
\rowcolor{navy}
{\color{white}\bfseries Instruction} &
{\color{white}\bfseries Source Width} &
{\color{white}\bfseries Result} \\
\rowcolor{softgray}
\code{ldrsb x0, [x1]} &
8 bits &
Sign-extended to 64 bits \\
\rowcolor{white}
\code{ldrsh x0, [x1]} &
16 bits &
Sign-extended to 64 bits \\
\rowcolor{softgray}
\code{ldrsw x0, [x1]} &
32 bits &
Sign-extended to 64 bits \\
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{block}{Example}
If memory contains the byte \hex{ff}, \code{ldrb} interprets the transferred bits as 255 after zero extension, while \code{ldrsb x0, [x1]} produces the 64-bit representation of $-1$.
\end{block}
\end{frame}

% ============================================================
% Slide 12
% Slide Header: Load Width and Extension Example
% ============================================================
\begin{frame}{Load Width and Extension Example}
Assume memory at \hex{2000} contains the byte \hex{80}.
\vspace{0.25cm}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Unsigned Byte Load}
\large
\hspace{1.1cm}\code{ldrb w0, [x1]}
\normalsize
\vspace{0.12cm}
\code{w0 = 0x00000080}
\vspace{0.08cm}
\begin{itemize}
\item Decimal value: 128
\item Upper bits filled with zeros
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Signed Byte Load}
\large
\hspace{1.0cm}\code{ldrsb x0, [x1]}
\normalsize
\vspace{0.12cm}
\code{x0 = 0xffffffffffffff80}
\vspace{0.08cm}
\begin{itemize}
\item Decimal value: $-128$
\item Sign bit replicated upward
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\sectiondivider{Section 03}{AArch64 Addressing Modes}

% ============================================================
% Slide 14
% Slide Header: Effective Address
% ============================================================
\begin{frame}{The Effective Address}
\begin{block}{Definition}
The effective address is the memory address computed for a particular load or store instruction.
\end{block}
\vspace{0.25cm}
\begin{itemize}
\item A base register supplies the starting address.
\item An optional immediate or register offset can modify that address.
\item Some addressing forms also write an updated address back to the base register.
\end{itemize}
\vspace{0.2cm}
\begin{block}{Base Register}
AArch64 memory operands normally use a 64-bit general-purpose register or \code{sp} as the base. A \code{w} register is not used as the base address register.
\end{block}
\end{frame}

% ============================================================
% Slide 15
% Slide Header: Base Register Addressing
% ============================================================
\begin{frame}{Base Register Addressing}
\begin{block}{Syntax}
\large
\hspace{4.0cm}\code{ldr x0, [x1]}
\normalsize
\end{block}
\vspace{0.25cm}
\begin{itemize}
\item Effective address $= \code{x1}$.
\item \code{x1} is unchanged.
\item This corresponds closely to dereferencing a pointer already holding the required address.
\end{itemize}
\vspace{0.2cm}
\begin{block}{C Analogy}
If \code{x1} represents pointer \code{p}, then the memory operand \code{[x1]} corresponds conceptually to \code{*p}.
\end{block}
\end{frame}

% ============================================================
% Slide 16
% Slide Header: Immediate Offset Addressing
% ============================================================
\begin{frame}{Immediate Offset Addressing}
\begin{block}{Example}
\large
\hspace{3.6cm}\code{ldr x0, [x1, \#16]}
\normalsize
\end{block}
\vspace{0.25cm}
\begin{itemize}
\item Effective address $= \code{x1} + 16$.
\item The base register \code{x1} is unchanged.
\item Useful for structure fields and fixed-position array elements.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Encoding Detail}
A64 has more than one load/store immediate encoding. Common unsigned offsets are scaled by transfer size; unscaled forms use a smaller signed byte offset.
\end{block}
\end{frame}

% ============================================================
% Slide 17
% Slide Header: Register Offset Addressing
% ============================================================
\begin{frame}{Register Offset Addressing}
\begin{block}{Unscaled Register Offset}
\large
\hspace{3.6cm}\code{ldr x0, [x1, x2]}
\normalsize
\vspace{0.05cm}
Effective address $= \code{x1} + \code{x2}$.
\end{block}
\vspace{0.15cm}
\begin{block}{Scaled Index for 64-Bit Elements}
\large
\hspace{3.0cm}\code{ldr x0, [x1, x2, lsl \#3]}
\normalsize
\vspace{0.05cm}
Effective address $= \code{x1} + (\code{x2} \times 8)$.
\end{block}
\vspace{0.15cm}
Register-offset forms are well suited to array indexing because the index can be scaled to the element size.
\end{frame}

% ============================================================
% Slide 18
% Slide Header: 32-Bit Index with Extension
% ============================================================
\begin{frame}{Using a 32-Bit Index Register}
AArch64 register-offset addressing can extend a 32-bit index while forming a 64-bit address.
\vspace{0.25cm}
\begin{block}{Unsigned 32-Bit Index}
\large
\hspace{2.8cm}\code{ldr x0, [x1, w2, uxtw \#3]}
\normalsize
\end{block}
\vspace{0.15cm}
\begin{itemize}
\item \code{w2} is zero-extended to address width.
\item \code{lsl \#3} scales the index by 8 bytes.
\item This is useful when the source-language array index is a 32-bit unsigned value.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Signed Index}
\code{sxtw} can be used instead when the 32-bit index must be sign-extended before address formation.
\end{block}
\end{frame}

% ============================================================
% Slide 19
% Slide Header: Pre-Indexed Addressing
% ============================================================
\begin{frame}{Pre-Indexed Addressing}
\begin{block}{Example}
\large
\hspace{3.5cm}\code{ldr x0, [x1, \#8]!}
\normalsize
\end{block}
\vspace{0.25cm}
\begin{enumerate}
\item Compute \code{x1 + 8}.
\item Use the updated address for the memory access.
\item Write the updated address back into \code{x1}.
\end{enumerate}
\vspace{0.15cm}
\begin{block}{Syntax Marker}
The exclamation mark indicates writeback. In A64, pre-indexed writeback uses an immediate offset, not a general-purpose register offset.
\end{block}
\end{frame}

% ============================================================
% Slide 20
% Slide Header: Post-Indexed Addressing
% ============================================================
\begin{frame}{Post-Indexed Addressing}
\begin{block}{Example}
\large
\hspace{3.5cm}\code{ldr x0, [x1], \#8}
\normalsize
\end{block}
\vspace{0.25cm}
\begin{enumerate}
\item Access memory using the original value of \code{x1}.
\item After the access, compute \code{x1 + 8}.
\item Write the updated address back into \code{x1}.
\end{enumerate}
\vspace{0.15cm}
\begin{block}{Common Use}
Post-indexing is useful when processing the current element and then advancing a pointer to the next element.
\end{block}
\end{frame}

% ============================================================
% Slide 21
% Slide Header: Addressing Mode Comparison
% ============================================================
\begin{frame}{Addressing Mode Comparison}
\begin{center}
\small
\renewcommand{\arraystretch}{1.14}
\begin{tabular}{>{\raggedright\arraybackslash}m{2.6cm} >{\raggedright\arraybackslash}m{4.2cm} >{\raggedright\arraybackslash}m{2.7cm} >{\raggedright\arraybackslash}m{2.4cm}}
\rowcolor{navy}
{\color{white}\bfseries Form} &
{\color{white}\bfseries Example} &
{\color{white}\bfseries Access Address} &
{\color{white}\bfseries Base After} \\
\rowcolor{softgray}
Base &
\code{ldr x0, [x1]} &
\code{x1} &
unchanged \\
\rowcolor{white}
Immediate offset &
\code{ldr x0, [x1, \#16]} &
\code{x1 + 16} &
unchanged \\
\rowcolor{softgray}
Register offset &
\code{ldr x0, [x1, x2]} &
\code{x1 + x2} &
unchanged \\
\rowcolor{white}
Pre-index &
\code{ldr x0, [x1, \#8]!} &
\code{x1 + 8} &
\code{x1 + 8} \\
\rowcolor{softgray}
Post-index &
\code{ldr x0, [x1], \#8} &
\code{x1} &
\code{x1 + 8} \\
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{block}{Key Distinction}
Offset addressing computes an access address without changing the base register; pre-indexed and post-indexed forms also write an updated address back to the base.
\end{block}
\end{frame}

% ============================================================
\sectiondivider{Section 04}{Arrays, Pointers, and Byte Data}

% ============================================================
% Slide 23
% Slide Header: Array Address Calculation
% ============================================================
\begin{frame}{Array Address Calculation}
For an array with element size $S$, element \code{A[i]} begins at:
\[
\text{address}(A[i]) = \text{base}(A) + i \times S
\]
\vspace{0.1cm}
\begin{block}{Example: 64-Bit Array}
If \code{x1} contains the array base and \code{x2} contains index \code{i}:
\par
\vspace{0.08cm}
\large
\hspace{3.0cm}\mbox{\code{ldr x0, [x1, x2, lsl \#3]}}
\normalsize
\end{block}
\vspace{0.15cm}
\begin{itemize}
\item \code{lsl \#3} multiplies the index by $2^3 = 8$.
\item The computed byte offset selects the requested 64-bit element.
\end{itemize}
\end{frame}

% ============================================================
% Slide 24
% Slide Header: Traversing an Array with a Pointer
% ============================================================
\begin{frame}{Traversing an Array with a Pointer}
Assume \code{x0} points to the current 64-bit array element.
\vspace{0.25cm}
\large
\hspace{3.4cm}\mbox{\code{ldr x1, [x0], \#8}}
\normalsize
\vspace{0.3cm}
\begin{itemize}
\item The load copies the current 64-bit element into \code{x1}.
\item The memory access uses the original address in \code{x0}.
\item After the load, post-index writeback advances \code{x0} by 8 bytes.
\item \code{x0} therefore points to the next array element.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Pointer Traversal}
Post-index addressing combines the current access with advancement to the next element. Topic 7 will place this operation inside a loop.
\end{block}
\end{frame}

% ============================================================
% Slide 25
% Slide Header: Byte Arrays and Pointer Advancement
% ============================================================
\begin{frame}{Byte Arrays and Pointer Advancement}
A C-style string is a byte array, so one character occupies one byte.
\vspace{0.25cm}
\large
\hspace{3.3cm}\code{ldrb w1, [x0], \#1}
\normalsize
\vspace{0.25cm}
\begin{itemize}
\item \code{ldrb} reads the current byte and zero-extends it into \code{w1}.
\item The access uses the original address in \code{x0}.
\item Post-index writeback advances \code{x0} by one byte.
\item Testing for the null terminator and repeating the access are Topic 7 control-flow operations.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Scope Boundary}
This instruction demonstrates byte access and pointer movement only; the complete string-traversal loop belongs in the next topic.
\end{block}
\end{frame}

\sectiondivider{Section 05}{Alignment and Pair Transfers}

% ============================================================
% Slide 27
% Slide Header: Natural Alignment
% ============================================================
\begin{frame}{Natural Alignment}
\begin{block}{Natural Alignment Rule}
An $n$-byte object is naturally aligned when its starting address is a multiple of $n$.
\end{block}
\vspace{0.2cm}
\begin{itemize}
\item 2-byte halfword: address divisible by 2.
\item 4-byte word: address divisible by 4.
\item 8-byte doubleword: address divisible by 8.
\end{itemize}
\vspace{0.15cm}
\begin{block}{AArch64 Qualification}
Many ordinary AArch64 loads and stores to normal memory permit unaligned addresses, but natural alignment is preferred. Misaligned accesses may be slower or cross memory-system boundaries, and some access classes have stricter requirements.
\end{block}
\end{frame}

% ============================================================
% Slide 28

\begin{frame}{Load and Store Pair: \code{ldp} and \code{stp}}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Store Pair}
\large
\hspace{1.3cm}\code{stp x0, x1, [x2]}
\normalsize
\begin{itemize}
\item Stores two register values.
\item The memory locations are adjacent.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Load Pair}
\large
\hspace{1.4cm}\code{ldp x0, x1, [x2]}
\normalsize
\begin{itemize}
\item Loads two register values.
\item The memory locations are adjacent.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.2cm}
\begin{block}{Important Qualification}
A pair transfer is one architectural instruction, but it is not guaranteed to be atomic or to execute as one physical memory-bus transaction.
\end{block}
\end{frame}

\begin{frame}{Topic 6 Summary}
\begin{itemize}
\item AArch64 uses explicit load/store instructions to move values between registers and memory.
\item The instruction determines transfer width and whether a narrow loaded value is zero-extended or sign-extended.
\item Effective addresses can use a base register, immediate offset, register offset, or pre/post-indexed immediate writeback.
\item Scaled register offsets map naturally to array indexing; post-indexing maps naturally to pointer traversal.
\item Natural alignment is preferred, although many ordinary unaligned accesses to normal memory are permitted.
\item \code{ldp} and \code{stp} transfer adjacent register pairs without implying atomicity or a particular physical bus transaction.
\end{itemize}
\end{frame}
