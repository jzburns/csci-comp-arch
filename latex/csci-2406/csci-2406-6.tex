\begin{frame}{AArch64 Is a Load/Store Architecture}
\begin{block}{Separation of Memory Access and Computation}
Arithmetic and logical instructions operate on register values. Dedicated load and store instructions transfer values between registers and memory.
\end{block}
\vspace{0.25cm}
\large
\hspace{3.5cm}\code{ldr x1, [x0]}\\[0.12cm]
\hspace{3.5cm}\code{add x1, x1, \#5}\\[0.12cm]
\hspace{3.5cm}\code{str x1, [x0]}
\normalsize
\vspace{0.25cm}
\begin{itemize}
\item \code{ldr} reads a value from memory into a register.
\item \code{add} changes the register value.
\item \code{str} writes the register value back to memory.
\end{itemize}
\end{frame}

\begin{frame}{Memory Is Byte Addressed}
\begin{itemize}
\item Each memory address identifies one byte.
\item Multi-byte values occupy consecutive addresses.
\item The instruction determines how many bytes are transferred.
\end{itemize}
\vspace{0.2cm}
\begin{center}
\begin{tikzpicture}[x=1.4cm,y=1cm]
\node[anchor=west,font=\small\bfseries,text=navy] at (-0.5,1.35) {Byte offset (hex)};
\foreach \offset/\position in {07/0,06/1,05/2,04/3,03/4,02/5,01/6,00/7} {
  \node[font=\small\ttfamily,text=navy] at (\position,0.95) {\offset};
  \draw[draw=deepblue,fill=softgray,line width=0.8pt] (\position-0.5,0) rectangle (\position+0.5,0.65);
  \node[font=\small] at (\position,0.325) {1 byte};
}
\draw[->,thick,deepblue] (7.4,-0.4) -- (-0.4,-0.4)
  node[midway,below,font=\small] {Increasing byte addresses};
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\begin{block}{Example: One 64-Bit Value}
Starting address: \code{1000} (hexadecimal). The rightmost byte is at \code{1000}; the next byte to its left is at \code{1001}; the leftmost byte is at \code{1007}. Together, the eight bytes occupy 64 bits.
\end{block}
\end{frame}

\begin{frame}{Bracket Notation Means Memory Access}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Register Operand}
\large
\hspace{1.5cm}\code{add x0, x1, x2}
\normalsize
\vspace{0.1cm}
\begin{itemize}
\item Uses values already in registers.
\item No data-memory operand appears.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Memory Operand}
\large
\hspace{1.5cm}\code{ldr x0, [x1]}
\normalsize
\vspace{0.1cm}
\begin{itemize}
\item \code{x1} contains an address.
\item Brackets mean access memory at that address.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.25cm}
\begin{block}{Do Not Confuse Address and Data}
The value in \code{x1} is the address. The bytes stored at that address are the data transferred by the load or store.
\end{block}
\end{frame}

% ============================================================
% Slide 7
% Slide Header: Load Register
% ============================================================


% ============================================================
% Slide 6
% ============================================================

\begin{frame}{Loading Data with \code{ldr}}
\begin{block}{64-Bit Load}
\large
\hspace{4.0cm}\code{ldr x0, [x1]}
\normalsize
\end{block}
\vspace{0.2cm}
\begin{itemize}
\item The effective address is the 64-bit value in \code{x1}.
\item Eight bytes are read from memory.
\item The 64-bit value is written into \code{x0}.
\end{itemize}
\vspace{0.15cm}
\begin{block}{32-Bit Load}
\code{ldr w0, [x1]} reads four bytes into \code{w0}. Any write to \code{w0} clears bits [63:32] of the corresponding \code{x0} register.
\end{block}
\end{frame}

% ============================================================
% Slide 8
% Slide Header: Store Register
% ============================================================


% ============================================================
% Slide 7
% ============================================================

\begin{frame}{Storing Data with \code{str}}
\begin{block}{64-Bit Store}
\large
\hspace{4.0cm}\code{str x0, [x1]}
\normalsize
\end{block}
\vspace{0.2cm}
\begin{itemize}
\item The effective address is the value in \code{x1}.
\item The eight-byte value in \code{x0} is written to memory.
\item The source register is not changed by the store.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Narrow Stores}
\code{strb} writes the low 8 bits of a register, and \code{strh} writes the low 16 bits. \code{str w0, [x1]} writes the low 32 bits.
\end{block}
\end{frame}

\begin{frame}{Load Width Determines Bytes Read}
\begin{center}
\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{>{\raggedright\arraybackslash}m{3.0cm} >{\raggedright\arraybackslash}m{2.5cm} >{\raggedright\arraybackslash}m{5.6cm}}
\rowcolor{navy}
{\color{white}\bfseries Instruction} &
{\color{white}\bfseries Bytes Read} &
{\color{white}\bfseries Destination Result} \\
\rowcolor{softgray}
\code{ldrb w0, [x1]} &
1 byte &
Zero-extended to 32 bits in \code{w0} \\
\rowcolor{white}
\code{ldrh w0, [x1]} &
2 bytes &
Zero-extended to 32 bits in \code{w0} \\
\rowcolor{softgray}
\code{ldr w0, [x1]} &
4 bytes &
Written to \code{w0}; upper half of \code{x0} becomes zero \\
\rowcolor{white}
\code{ldr x0, [x1]} &
8 bytes &
Full 64-bit value written to \code{x0} \\
\end{tabular}
\end{center}
\vspace{0.25cm}
\begin{block}{Width Comes from the Instruction}
A memory address does not carry a type. The load/store instruction determines the number of bytes transferred.
\end{block}
\end{frame}

\begin{frame}{Signed Loads Preserve Negative Values}
\begin{block}{Sign Extension}
A signed load copies the sign bit of the narrow source into the new upper bits so the two's-complement value is preserved.
\end{block}
\vspace{0.15cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{>{\raggedright\arraybackslash}m{3.2cm} >{\raggedright\arraybackslash}m{2.4cm} >{\raggedright\arraybackslash}m{5.4cm}}
\rowcolor{navy}
{\color{white}\bfseries Instruction} &
{\color{white}\bfseries Source Width} &
{\color{white}\bfseries Result} \\
\rowcolor{softgray}
\code{ldrsb x0, [x1]} &
8 bits &
Sign-extended to 64 bits \\
\rowcolor{white}
\code{ldrsh x0, [x1]} &
16 bits &
Sign-extended to 64 bits \\
\rowcolor{softgray}
\code{ldrsw x0, [x1]} &
32 bits &
Sign-extended to 64 bits \\
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{block}{Example}
If memory contains the byte \hex{ff}, \code{ldrb} interprets the transferred bits as 255 after zero extension, while \code{ldrsb x0, [x1]} produces the 64-bit representation of $-1$.
\end{block}
\end{frame}

% ============================================================
% Slide 12
% Slide Header: Load Width and Extension Example
% ============================================================


% ============================================================
% Slide 10
% ============================================================

\begin{frame}{Load Width and Extension Example}
Assume \code{x1 = 0x2000} and memory at that address contains the byte \hex{80}.
\vspace{0.2cm}
\begin{itemize}
\item \textbf{Unsigned byte load:} \code{ldrb w0, [x1]}
  \begin{itemize}
  \item Reads one byte and fills the remaining bits with zeros.
  \item Result: \code{w0 = 0x00000080}, decimal value 128.
  \item Writing \code{w0} also clears the upper 32 bits of \code{x0}.
  \end{itemize}
\vspace{0.15cm}
\item \textbf{Signed byte load:} \code{ldrsb x0, [x1]}
  \begin{itemize}
  \item Reads one byte and copies its sign bit into all higher bits.
  \item Result: \code{x0 = 0xffffffffffffff80}, decimal value $-128$.
  \end{itemize}
\end{itemize}
\end{frame}

\sectiondivider{Section 3}{AArch64 Addressing Modes}

\begin{frame}{Three Addressing Types}
A \textbf{pointer} holds a memory address. In these examples, the \textbf{base register} holds that pointer: C pointer \code{ptr} corresponds to register \code{x1}.
\vspace{0.1cm}
\begin{enumerate}
\item \textbf{Offset:} access an address relative to the base; leave the base unchanged.
  \begin{itemize}
  \item \code{ldr x0, [x1, \#8]} \quad C: \code{value = ptr[1];}
  \end{itemize}
\item \textbf{Pre-index:} adjust the pointer, then access that address.
  \begin{itemize}
  \item \code{ldr x0, [x1, \#8]!} \quad C: \code{ptr++; value = *ptr;}
  \end{itemize}
\item \textbf{Post-index:} access the current address, then adjust the pointer.
  \begin{itemize}
  \item \code{ldr x0, [x1], \#8} \quad C: \code{value = *ptr; ptr++;}
  \end{itemize}
\end{enumerate}
\vspace{0.15cm}
\begin{block}{C Conventions Used in This Section}
\code{uint64\_t *ptr} corresponds to \code{x1}; \code{uint64\_t value} corresponds to \code{x0}. Each element occupies 8 bytes, so \code{ptr++} advances by 8 bytes. Examples assume valid array accesses.
\end{block}
\end{frame}

\begin{frame}{The Effective Address}
\begin{block}{Definition}
The effective address is the memory address computed for a particular load or store instruction.
\end{block}
\vspace{0.25cm}
\begin{itemize}
\item A base register holds the pointer that supplies the starting address.
\item An optional immediate or register offset can modify that address.
\item Some addressing forms also write an updated address back to the base register.
\end{itemize}
\vspace{0.2cm}
\begin{block}{Base Register}
AArch64 memory operands normally use a 64-bit general-purpose register or \code{sp} as the base. A \code{w} register is not used as the base address register.
\end{block}
\end{frame}

% ============================================================
% Slide 15
% Slide Header: Base Register Addressing
% ============================================================


% ============================================================
% Slide 12
% ============================================================

\begin{frame}{1. Offset: Base Register Only}
\begin{block}{Syntax}
\large
\hspace{4.0cm}\code{ldr x0, [x1]}
\normalsize
\end{block}
\vspace{0.25cm}
\begin{itemize}
\item Effective address $= \code{x1}$.
\item \code{x1} is unchanged.
\item This corresponds closely to dereferencing a pointer already holding the required address.
\end{itemize}
\vspace{0.2cm}
\begin{block}{C Analogy}
\code{value = *ptr;}\par This is offset addressing with a zero offset: read the current element and leave \code{ptr} unchanged.
\end{block}
\end{frame}

% ============================================================
% Slide 16
% Slide Header: Immediate Offset Addressing
% ============================================================


% ============================================================
% Slide 13
% ============================================================

\begin{frame}{1. Offset: Immediate Offset}
\begin{block}{Example}
\large
\hspace{3.6cm}\code{ldr x0, [x1, \#16]}
\normalsize
\end{block}
\vspace{0.25cm}
\begin{itemize}
\item Effective address $= \code{x1} + 16$.
\item The base register \code{x1} is unchanged.
\item Useful for structure fields and fixed-position array elements.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Equivalent C}
\code{value = ptr[2];}\par Read the element 16 bytes beyond \code{ptr}; \code{ptr} is unchanged.
\end{block}
\begin{block}{Encoding Detail}
A64 has more than one load/store immediate encoding. Common unsigned offsets are scaled by transfer size; unscaled forms use a smaller signed byte offset.
\end{block}
\end{frame}

% ============================================================
% Slide 17
% Slide Header: Register Offset Addressing
% ============================================================


% ============================================================
% Slide 14
% ============================================================

\begin{frame}{1. Offset: Register Offset}
\begin{block}{Pointer Access: \code{x2} Holds a Byte Displacement}
\code{ldr x0, [x1, x2]}\par
\code{value = *(ptr + offset);}\par
Here \code{ptr} is a pointer to 64-bit elements. C counts \code{offset} in elements; \code{x1} holds \code{ptr} and \code{x2} holds \code{offset * 8} bytes.
\end{block}
\vspace{0.15cm}
\begin{block}{Array Index: \code{x2} Holds Element Index \code{i}}
\code{ldr x0, [x1, x2, lsl \#3]}\par
\code{value = ptr[i];}\par
Multiply \code{i} by 8 to obtain the byte displacement for 64-bit elements.
\end{block}
\vspace{0.15cm}
Both forms leave \code{x1} (the pointer \code{ptr}) unchanged.
\end{frame}



% ============================================================
% Slide 19
% Slide Header: Pre-Indexed Addressing
% ============================================================


% ============================================================
% Slide 16
% ============================================================

\begin{frame}{2. Pre-Indexed Addressing}
\begin{block}{Assembly}
\code{ldr x0, [x1, \#8]!}
\end{block}
\begin{block}{Equivalent C: Advance, Then Read}
\code{ptr++;}\par
\code{value = *ptr;}\par
Or, in one expression: \code{value = *++ptr;}
\end{block}
\begin{itemize}
\item Access address: original \code{x1 + 8} (the next element).
\item Base after the instruction: original \code{x1 + 8}.
\item The exclamation mark indicates writeback; A64 pre-index uses an immediate offset.
\end{itemize}
\end{frame}

% ============================================================
% Slide 20
% Slide Header: Post-Indexed Addressing
% ============================================================


% ============================================================
% Slide 17
% ============================================================

\begin{frame}{3. Post-Indexed Addressing}
\begin{block}{Assembly}
\code{ldr x0, [x1], \#8}
\end{block}
\begin{block}{Equivalent C: Read, Then Advance}
\code{value = *ptr;}\par
\code{ptr++;}\par
Or, in one expression: \code{value = *ptr++;}
\end{block}
\begin{itemize}
\item Access address: original \code{x1} (the current element).
\item Base after the instruction: original \code{x1 + 8}.
\item The immediate outside the brackets specifies the pointer update.
\end{itemize}
\end{frame}

\begin{frame}{Addressing Mode Comparison}
\begin{center}
\small
\renewcommand{\arraystretch}{1.14}
\begin{tabular}{>{\raggedright\arraybackslash}m{2.6cm} >{\raggedright\arraybackslash}m{4.2cm} >{\raggedright\arraybackslash}m{2.7cm} >{\raggedright\arraybackslash}m{2.4cm}}
\rowcolor{navy}
{\color{white}\bfseries Form} &
{\color{white}\bfseries Example} &
{\color{white}\bfseries Access Address} &
{\color{white}\bfseries Base After} \\
\rowcolor{softgray}
Offset (zero) &
\code{ldr x0, [x1]} &
\code{x1} &
unchanged \\
\rowcolor{white}
Immediate offset &
\code{ldr x0, [x1, \#16]} &
\code{x1 + 16} &
unchanged \\
\rowcolor{softgray}
Register offset &
\code{ldr x0, [x1, x2]} &
\code{x1 + x2} &
unchanged \\
\rowcolor{white}
Pre-index &
\code{ldr x0, [x1, \#8]!} &
\code{x1 + 8} &
\code{x1 + 8} \\
\rowcolor{softgray}
Post-index &
\code{ldr x0, [x1], \#8} &
\code{x1} &
\code{x1 + 8} \\
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{block}{Key Distinction}
Offset addressing computes an access address without changing the base register; pre-indexed and post-indexed forms also write an updated address back to the base.
\end{block}
\end{frame}

\begin{frame}{Byte Access: Advance by One Byte}
\begin{block}{C: Read One Byte, Then Advance}
\code{uint8\_t *ptr;}\par
\code{value = *ptr++;}
\end{block}
\begin{block}{AArch64: \code{x0} Holds \code{ptr}}
\code{ldrb w1, [x0], \#1}
\end{block}
\begin{itemize}
\item Read one byte and zero-extend it into \code{w1}.
\item Advance by 1 byte for a byte array, compared with 8 bytes for a 64-bit array.
\item Match both the load width and pointer increment to the element size.
\end{itemize}
\end{frame}

\begin{frame}{Natural Alignment}
\begin{block}{Natural Alignment Rule}
An $n$-byte object is naturally aligned when its starting address is a multiple of $n$.
\end{block}
\vspace{0.2cm}
\begin{itemize}
\item 2-byte halfword: address divisible by 2.
\item 4-byte word: address divisible by 4.
\item 8-byte doubleword: address divisible by 8.
\end{itemize}
\vspace{0.15cm}
\begin{block}{AArch64 Qualification}
Many ordinary AArch64 loads and stores to normal memory permit unaligned addresses, but natural alignment is preferred. Misaligned accesses may be slower or cross memory-system boundaries, and some access classes have stricter requirements.
\end{block}
\end{frame}

\large

\large
\hspace{1.4cm}\code{ldp x0, x1, [x2]}
\normalsize
\begin{itemize}

\item The memory locations are adjacent.
\end{itemize}
\end{block}

\begin{block}{Important Qualification}
A pair transfer is one architectural instruction, but it is not guaranteed to be atomic or to execute as one physical memory-bus transaction.
\end{block}

\begin{frame}{Topic 6 Summary}
\begin{itemize}
\item AArch64 uses explicit load/store instructions to move values between registers and memory.
\item The instruction determines transfer width and whether a narrow loaded value is zero-extended or sign-extended.
\item The base register holds a pointer. Offset addressing leaves it unchanged; pre-index and post-index addressing update it before or after the access, respectively.
\item Scaled register offsets map naturally to array indexing; post-indexing maps naturally to pointer traversal.
\item Natural alignment is preferred, although many ordinary unaligned accesses to normal memory are permitted.
\item \code{ldp} and \code{stp} transfer adjacent register pairs without implying atomicity or a particular physical bus transaction.
\end{itemize}
\end{frame}
