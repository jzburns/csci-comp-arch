\sectiondivider{Section 1}{Alignment and Transfer Widths}

\begin{frame}{Memory Is Byte Addressed}
\begin{itemize}
\item Each memory address identifies one byte.
\item Multi-byte values occupy consecutive addresses.
\item The instruction determines how many bytes are transferred.
\end{itemize}
\vspace{0.2cm}
\begin{center}
\begin{tikzpicture}[x=1.4cm,y=1cm]
\node[anchor=west,font=\small\bfseries,text=navy] at (-0.5,1.35) {Byte offset (hex)};
\foreach \offset/\position in {07/0,06/1,05/2,04/3,03/4,02/5,01/6,00/7} {
  \node[font=\small\ttfamily,text=navy] at (\position,0.95) {\offset};
  \draw[draw=deepblue,fill=softgray,line width=0.8pt] (\position-0.5,0) rectangle (\position+0.5,0.65);
  \node[font=\small] at (\position,0.325) {1 byte};
}
\draw[->,thick,deepblue] (7.4,-0.4) -- (-0.4,-0.4)
  node[midway,below,font=\small] {Increasing byte addresses};
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\begin{block}{Example: One 64-Bit Value}
Starting address: \code{1000} (hexadecimal). The rightmost byte is at \code{1000}; the next byte to its left is at \code{1001}; the leftmost byte is at \code{1007}. Together, the eight bytes occupy 64 bits.
\end{block}
\end{frame}

\begin{frame}{Natural Alignment}
\begin{block}{Natural Alignment Rule}
An $n$-byte object is naturally aligned when its starting address is a multiple of $n$.
\end{block}
\vspace{0.2cm}
\begin{itemize}
\item 2-byte halfword: address divisible by 2.
\item 4-byte word: address divisible by 4.
\item 8-byte doubleword: address divisible by 8.
\end{itemize}
\vspace{0.15cm}
\begin{block}{AArch64 Qualification}
Many ordinary AArch64 loads and stores to normal memory permit unaligned addresses, but natural alignment is preferred. Misaligned accesses may be slower or cross memory-system boundaries, and some access classes have stricter requirements.
\end{block}
\end{frame}

\begin{center}

\begin{block}{Natural Alignment for Loads and Stores}
\small
Let $A$ be the effective byte address. For an $n$-byte transfer, natural alignment means $A \bmod n = 0$.
\par\smallskip
\begin{tabular}{@{}ll@{\qquad}ll@{}}
\code{ldrb/strb}: & $A \bmod 1 = 0$ & \code{ldrh/strh}: & $A \bmod 2 = 0$ \\
\code{ldr/str} (32-bit): & $A \bmod 4 = 0$ & \code{ldr/str} (64-bit): & $A \bmod 8 = 0$
\end{tabular}
\par\smallskip
\small Offset \code{00} starts at \code{1000} (hex); each group begins at its right edge. These are natural alignment conditions; ordinary loads/stores to normal memory may permit unaligned addresses.
\end{block}
\end{frame}

\begin{frame}{Consequences of Unaligned Access}
\begin{block}{Example}
An 8-byte access at address \code{1003} (hex) is not naturally aligned: $A \bmod 8 \ne 0$.
\end{block}
\begin{itemize}
\item \textbf{Correct access may still be possible:} ordinary AArch64 loads/stores to normal memory often support unaligned addresses.
\item \textbf{Lower performance:} crossing a cache-line boundary may require extra work.
\item \textbf{Possible fault:} alignment checking, certain instructions, or device memory can prohibit unaligned access.
\item \textbf{Atomicity may not be guaranteed:} an unaligned access may lack the guarantees of a naturally aligned access.
\end{itemize}
\begin{block}{Why Prefer Natural Alignment?}
An unaligned access does not automatically produce the wrong value. Natural alignment avoids these complications.
\end{block}
\end{frame}


\begin{frame}{Load/Store Transfer Widths}
\begin{center}
\small
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{>{\raggedright\arraybackslash}m{2.7cm} >{\raggedright\arraybackslash}m{2.7cm} >{\raggedright\arraybackslash}m{1.5cm} >{\raggedright\arraybackslash}m{4.6cm}}
\rowcolor{navy}
{\color{white}\bfseries Load} & {\color{white}\bfseries Store} & {\color{white}\bfseries Width} & {\color{white}\bfseries Load Result} \\
\rowcolor{softgray}
\code{ldrb} & \code{strb} & 1 byte & Zero-extended to 32 bits \\
\rowcolor{white}
\code{ldrh} & \code{strh} & 2 bytes & Zero-extended to 32 bits \\
\rowcolor{softgray}
\code{ldr} (32-bit) & \code{str} (32-bit) & 4 bytes & 32-bit value; upper 32 bits cleared \\
\rowcolor{white}
\code{ldr} (64-bit) & \code{str} (64-bit) & 8 bytes & Full 64-bit value \\
\end{tabular}
\end{center}
\begin{itemize}
\item A load into a 32-bit register clears the upper half of its 64-bit register.
\item Stores write the specified number of bytes from the low end of the source register.
\item The instruction determines transfer width; the memory address carries no type.
\end{itemize}
\end{frame}

\begin{frame}{Signed Loads Preserve Negative Values}
\begin{block}{Sign Extension}
A signed load copies the sign bit of the narrow source into the new upper bits so the two's-complement value is preserved.
\end{block}
\vspace{0.15cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{>{\raggedright\arraybackslash}m{3.2cm} >{\raggedright\arraybackslash}m{2.4cm} >{\raggedright\arraybackslash}m{5.4cm}}
\rowcolor{navy}
{\color{white}\bfseries Instruction} &
{\color{white}\bfseries Source Width} &
{\color{white}\bfseries Result} \\
\rowcolor{softgray}
\code{ldrsb x1, [x0]} &
8 bits &
Sign-extended to 64 bits \\
\rowcolor{white}
\code{ldrsh x1, [x0]} &
16 bits &
Sign-extended to 64 bits \\
\rowcolor{softgray}
\code{ldrsw x1, [x0]} &
32 bits &
Sign-extended to 64 bits \\
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{block}{Example}
If memory contains the byte \hex{ff}, \code{ldrb} interprets the transferred bits as 255 after zero extension, while \code{ldrsb x1, [x0]} produces the 64-bit representation of $-1$.
\end{block}
\end{frame}

% ============================================================
% Slide Header: Load Width and Extension Example
% ============================================================


% ============================================================
% ============================================================

\begin{frame}{Load Width and Extension Example}
Assume \code{x0 = 0x2000} and memory at that address contains the byte \hex{80}.
\vspace{0.2cm}
\begin{itemize}
\item \textbf{Unsigned byte load:} \code{ldrb w1, [x0]}
  \begin{itemize}
  \item Reads one byte and fills the remaining bits with zeros.
  \item Result: \code{w1 = 0x00000080}, decimal value 128.
  \item Writing \code{w1} also clears the upper 32 bits of \code{x1}.
  \end{itemize}
\vspace{0.15cm}
\item \textbf{Signed byte load:} \code{ldrsb x1, [x0]}
  \begin{itemize}
  \item Reads one byte and copies its sign bit into all higher bits.
  \item Result: \code{x1 = 0xffffffffffffff80}, decimal value $-128$.
  \end{itemize}
\end{itemize}
\end{frame}

\sectiondivider{Section 2}{AArch64 Addressing Modes}

\begin{frame}{Three Addressing Types}
\begin{block}{Pointer and Base Register}
A \textbf{pointer} holds a memory address. The \textbf{base register} holds that pointer. Throughout these examples, \code{x0} holds the C pointer \code{ptr}; loaded values go into \code{x1} (or \code{w1} for a narrow load).
\end{block}
\vspace{0.15cm}
\begin{enumerate}
\item \textbf{Offset:} access an address relative to the pointer, leaving the pointer unchanged.
\item \textbf{Pre-index:} adjust the pointer first, then access memory at the adjusted address.
\item \textbf{Post-index:} access memory at the current pointer, then adjust the pointer.
\end{enumerate}
\end{frame}


% ============================================================
% Slide Header: Effective Address
% ============================================================


% ============================================================
% ============================================================

\begin{frame}{The Effective Address}
\begin{block}{Definition}
The \textbf{effective address} is the memory address computed for a particular load or store instruction.
\end{block}
\vspace{0.25cm}
\begin{itemize}
\item A base register holds the pointer that supplies the \textbf{starting address}.
\item An optional immediate or register offset can \textbf{modify} that address.
\item Some addressing forms also \textbf{write back} an updated address to the base register.
\end{itemize}
\vspace{0.2cm}
\begin{block}{Base Register}
AArch64 memory operands normally use a 64-bit general-purpose register or \code{sp} as the base. \textbf{A \code{w} register is not used as the base address register.}
\end{block}
\end{frame}

% ============================================================
% Slide Header: Base Register Addressing
% ============================================================


% ============================================================
% ============================================================

\begin{frame}{1. Offset: Base Register Only}
\begin{block}{Syntax}
\large
\hspace{4.0cm}\code{ldr x1, [x0]}
\normalsize
\end{block}
\vspace{0.25cm}
\begin{itemize}
\item Effective address $= \code{x0}$.
\item \code{x0} is unchanged.
\item This corresponds closely to dereferencing a pointer already holding the required address.
\end{itemize}
\vspace{0.2cm}
\begin{block}{Equivalent C: \code{ptr} Points to 64-Bit Elements}
\code{uint64\_t *ptr;}\par
\code{value = *ptr;}\par This is offset addressing with a zero offset: read the current element and leave \code{ptr} unchanged.
\end{block}
\end{frame}

% ============================================================
% Slide Header: Immediate Offset Addressing
% ============================================================


% ============================================================
% ============================================================

\begin{frame}{1. Offset: Immediate Offset}
\begin{block}{Example}
\large
\hspace{3.6cm}\code{ldr x1, [x0, \#16]}
\normalsize
\end{block}
\vspace{0.15cm}
\begin{itemize}
\item Effective address $= \code{x0} + 16$.
\item The base register \code{x0} is unchanged.
\item Useful for structure fields and fixed-position array elements.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Equivalent C}
\code{value = *(ptr + 2);}\par Read the element 16 bytes beyond \code{ptr}; \code{ptr} is unchanged.
\end{block}

\end{frame}

% ============================================================
% Slide Header: Register Offset Addressing
% ============================================================


% ============================================================
% ============================================================

\begin{frame}{1. Offset: Register Offset}
\begin{block}{1. Register Holds the Byte Offset}
\textbf{AArch64:} \code{ldr x1, [x0, x2]}\par
\textbf{C:} \code{value = *(ptr + offset);}\par
C counts \code{offset} in elements. \code{x2} holds \code{offset * 8}, the corresponding byte offset.
\end{block}
\vspace{0.1cm}
\begin{block}{2. Element Index Converted to a Byte Displacement}
\textbf{AArch64:} \code{ldr x1, [x0, x2, lsl \#3]}\par
\textbf{C:} \code{value = *(ptr + i);}\par
\code{x2} holds index \code{i}; \code{lsl \#3} multiplies it by 8.
\end{block}
\vspace{0.1cm}
\begin{itemize}
\item In both examples, \code{x0} holds \code{ptr}, pointing to 64-bit elements.
\item The loaded value goes into \code{x1}; \code{x0} is unchanged.
\end{itemize}
\end{frame}



% ============================================================
% Slide Header: Pre-Indexed Addressing
% ============================================================


% ============================================================
% ============================================================

\begin{frame}{2. Pre-Indexed Addressing}
\begin{block}{Assembly}
\code{ldr x1, [x0, \#8]!}
\end{block}
\begin{block}{Equivalent C: Advance, Then Read}
\code{ptr++;}\par
\code{value = *ptr;}\par
Or, in one expression: \code{value = *++ptr;}
\end{block}
\begin{itemize}
\item Access address: original \code{x0 + 8} (the next element).
\item Base after the instruction: original \code{x0 + 8}.
\item The exclamation mark indicates writeback; A64 pre-index uses an immediate offset.
\end{itemize}
\end{frame}

% ============================================================
% Slide Header: Post-Indexed Addressing
% ============================================================


% ============================================================
% ============================================================

\begin{frame}{3. Post-Indexed Addressing}
\begin{block}{Assembly}
\code{ldr x1, [x0], \#8}
\end{block}
\begin{block}{Equivalent C: Read, Then Advance}
\code{value = *ptr;}\par
\code{ptr++;}\par
Or, in one expression: \code{value = *ptr++;}
\end{block}
\begin{itemize}
\item Access address: original \code{x0} (the current element).
\item Base after the instruction: original \code{x0 + 8}.
\item The immediate outside the brackets specifies the pointer update.
\end{itemize}
\end{frame}

% Slide Header: Addressing Mode Comparison
% ============================================================


% ============================================================
% ============================================================

\begin{frame}{Addressing Mode Comparison}
\begin{center}
\small
\renewcommand{\arraystretch}{1.14}
\begin{tabular}{>{\raggedright\arraybackslash}m{2.6cm} >{\raggedright\arraybackslash}m{4.2cm} >{\raggedright\arraybackslash}m{2.7cm} >{\raggedright\arraybackslash}m{2.4cm}}
\rowcolor{navy}
{\color{white}\bfseries Form} &
{\color{white}\bfseries Example} &
{\color{white}\bfseries Access Address} &
{\color{white}\bfseries Base After} \\
\rowcolor{softgray}
Offset (zero) &
\code{ldr x1, [x0]} &
\code{x0} &
unchanged \\
\rowcolor{white}
Immediate offset &
\code{ldr x1, [x0, \#16]} &
\code{x0 + 16} &
unchanged \\
\rowcolor{softgray}
Register offset &
\code{ldr x1, [x0, x2]} &
\code{x0 + x2} &
unchanged \\
\rowcolor{white}
Pre-index &
\code{ldr x1, [x0, \#8]!} &
\code{x0 + 8} &
\code{x0 + 8} \\
\rowcolor{softgray}
Post-index &
\code{ldr x1, [x0], \#8} &
\code{x0} &
\code{x0 + 8} \\
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{block}{Key Distinction}
Offset addressing computes an access address without changing the base register; pre-indexed and post-indexed forms also write an updated address back to the base.
\end{block}
\end{frame}

% Slide Header: Byte Arrays and Pointer Advancement
% ============================================================

\begin{frame}{Byte Access: Advance by One Byte}
\begin{block}{C: Read One Byte, Then Advance}
\code{uint8\_t *ptr;}\par
\code{value = *ptr++;}
\end{block}
\begin{block}{AArch64: \code{x0} Holds \code{ptr}}
\code{ldrb w1, [x0], \#1}
\end{block}
\begin{itemize}
\item Read one byte and zero-extend it into \code{w1}.
\item Advance by 1 byte for a byte array, compared with 8 bytes for a 64-bit array.
\item Match both the load width and pointer increment to the element size.
\end{itemize}
\end{frame}

\sectiondivider{Section 3}{Pair Transfers}

\begin{frame}{Load and Store Pair: \code{ldp} and \code{stp}}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Store Pair}
\large
\hspace{1.3cm}\code{stp x1, x2, [x0]}
\normalsize
\begin{itemize}
\item Stores two register values.
\item The memory locations are adjacent.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Load Pair}
\large
\hspace{1.4cm}\code{ldp x1, x2, [x0]}
\normalsize
\begin{itemize}
\item Loads two register values.
\item The memory locations are adjacent.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.2cm}
\begin{block}{Important Qualification}
A pair transfer is one architectural instruction, but it is not guaranteed to be atomic or to execute as one physical memory-bus transaction.
\end{block}
\end{frame}

\begin{frame}{Pair Transfers: Order in Memory}
\begin{columns}[T,onlytextwidth]
\column{0.46\textwidth}
\begin{center}
\begin{tikzpicture}[x=1cm,y=1cm]
\draw[draw=deepblue,fill=deepblue!12] (0,1.3) rectangle (3.6,2.6);
\draw[draw=deepblue,fill=gold!15] (0,0) rectangle (3.6,1.3);
\node[font=\normalsize\bfseries] at (1.8,1.95) {\code{x30} (LR)};
\node[font=\normalsize\bfseries] at (1.8,0.65) {\code{x29} (FP)};
\node[anchor=west,font=\small] at (3.75,1.3) {\code{sp + 8}};
\node[anchor=west,font=\small] at (3.75,0) {\code{sp}};

\begin{frame}{Topic 6 Summary}
\begin{itemize}
\item AArch64 uses explicit load/store instructions to move values between registers and memory.
\item The instruction determines transfer width and whether a narrow loaded value is zero-extended or sign-extended.
\item The base register holds a pointer. Offset addressing leaves it unchanged; pre-index and post-index addressing update it before or after the access, respectively.
\item Scaled register offsets map naturally to array indexing; post-indexing maps naturally to pointer traversal.
\item Natural alignment is preferred, although many ordinary unaligned accesses to normal memory are permitted.
\item \code{ldp} and \code{stp} transfer adjacent register pairs without implying atomicity or a particular physical bus transaction.
\end{itemize}
\end{frame}
