\begin{frame}{Arithmetic Primitives: Addition \& Subtraction}
\large

\begin{itemize}\setlength{\itemsep}{6pt}
  \item Many AArch64 data-processing instructions use a
        \key{3-operand non-destructive format}:
  $$\code{op\quad destination,\; source1,\; source2}$$
\end{itemize}

\vspace{0.15cm}

\begin{center}
\small
\setlength{\tabcolsep}{6pt}

\begin{tabular}{lll}
\toprule
\textbf{Instruction} & \textbf{Operation} & \textbf{Flags Updated?} \\
\midrule
\code{add  x0, x1, x2}  & $x_0 \leftarrow x_1 + x_2$ & No \\
\code{adds x0, x1, x2}  & $x_0 \leftarrow x_1 + x_2$ & \key{Yes (Updates NZCV)} \\
\code{sub  x0, x1, x2}  & $x_0 \leftarrow x_1 - x_2$ & No \\
\code{subs x0, x1, x2}  & $x_0 \leftarrow x_1 - x_2$ & \key{Yes (Updates NZCV)} \\
\code{neg  x0, x1}      & $x_0 \leftarrow 0 - x_1$   & No (Alias for \code{sub x0, xzr, x1}) \\
\bottomrule
\end{tabular}

\end{center}

\vspace{0.2cm}
\normalsize

\begin{block}{Immediate Constant Arithmetic}
\code{add x0, x1, \#42} adds a 12-bit immediate
($0\text{ to }4095$).
32-bit registers (\bits{w0}--\bits{w30})
follow identical syntax.
\end{block}

\end{frame}

\begin{frame}{Shifted-Register Operands and Shift Operations}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item The AArch64 ISA allows many instructions to encode
        a shifted register operand directly within the instruction word.
  \item Execution latency for shifted operands depends
        on the specific processor implementation.
\end{itemize}

\vspace{0.2cm}

\begin{center}
\small
\setlength{\tabcolsep}{6pt}

\begin{tabular}{lll}
\toprule
\textbf{Shift Type} & \textbf{Mnemonic} & \textbf{Operation Meaning} \\
\midrule
Logical Shift Left
&
\code{lsl \#k}
&
Equivalent to multiplication by $2^k$ modulo the register width (zero-fill).
\\

Logical Shift Right
&
\code{lsr \#k}
&
Equivalent to floor division by $2^k$ (zero-fill).
\\

Arithmetic Shift Right
&
\code{asr \#k}
&
Sign-preserving right shift
(replicates sign bit; differs from truncation toward zero for negative numbers).
\\
\bottomrule
\end{tabular}

\end{center}

\vspace{0.15cm}

\begin{block}{Integrated Shift-Add Examples}
\small

\begin{tabular}{lll}
\code{add x0, x1, x2, lsl \#3}
&
$\implies x_0 \leftarrow x_1 + (x_2 \times 8)$
&
(Array indexing by 8-byte doubleword)
\\

\code{sub x0, x1, x2, lsr \#1}
&
$\implies x_0 \leftarrow x_1 - (x_2 / 2)$
&
(Subtract half of unsigned value)
\\
\end{tabular}

\end{block}

\end{frame}

\begin{frame}{Integer Multiplication \& Division}
\large

\begin{itemize}\setlength{\itemsep}{6pt}
  \item AArch64 defines instructions for integer multiplication
        and division, leaving the underlying microarchitectural
        execution units implementation-dependent.
\end{itemize}

\vspace{0.15cm}

\begin{center}
\small
\setlength{\tabcolsep}{6pt}

\begin{tabular}{lll}
\toprule
\textbf{Instruction}
&
\textbf{RTL Semantic Action}
&
\textbf{Description}
\\
\midrule

\code{mul  x0, x1, x2}
&
$x_0 \leftarrow x_1 \times x_2$
&
64-bit integer product (lower 64 bits)
\\

\code{madd x0, x1, x2, x3}
&
$x_0 \leftarrow x_3 + (x_1 \times x_2)$
&
Multiply-Add
\\

\code{msub x0, x1, x2, x3}
&
$x_0 \leftarrow x_3 - (x_1 \times x_2)$
&
Multiply-Subtract
\\

\code{sdiv x0, x1, x2}
&
$x_0 \leftarrow x_1 / x_2$
&
Signed division
\\

\code{udiv x0, x1, x2}
&
$x_0 \leftarrow x_1 / x_2$
&
Unsigned division
\\
\bottomrule
\end{tabular}

\end{center}

\vspace{0.25cm}
\normalsize

\begin{block}{Division by Zero}
AArch64 division by zero does \textbf{not} raise an exception;
it returns \bits{0} in the destination register.
\end{block}

\end{frame}

\begin{frame}{Bitwise Logic Primitives}
\large

\begin{itemize}\setlength{\itemsep}{6pt}
  \item Bitwise operations apply independently
        to each corresponding bit position.
\end{itemize}

\vspace{0.15cm}

\begin{center}
\small
\setlength{\tabcolsep}{6pt}

\begin{tabular}{lll}
\toprule
\textbf{Instruction}
&
\textbf{Operation}
&
\textbf{Bitwise RTL Formula}
\\
\midrule

\code{and x0, x1, x2}
&
Bitwise AND
&
$x_0 \leftarrow x_1 \land x_2$
\\

\code{orr x0, x1, x2}
&
Bitwise OR
&
$x_0 \leftarrow x_1 \lor x_2$
\\

\code{eor x0, x1, x2}
&
Bitwise XOR
&
$x_0 \leftarrow x_1 \oplus x_2$
\\

\code{bic x0, x1, x2}
&
Bit Clear (AND NOT)
&
$x_0 \leftarrow x_1 \land (\sim x_2)$
\\

\code{mvn x0, x1}
&
Bitwise NOT
&
$x_0 \leftarrow \sim x_1$
\quad (Alias: \code{orn x0, xzr, x1})
\\
\bottomrule
\end{tabular}

\end{center}

\vspace{0.25cm}
\normalsize

\begin{block}{Bit Clear (\texttt{bic}) Idiom}
\code{bic x0, x1, x2}
sets bits to \bits{0} in $x_1$
wherever the mask in $x_2$ has \bits{1}s.
\end{block}

\end{frame}

\begin{frame}{Bitfield Extraction: \texttt{ubfx}}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item AArch64 provides bitfield instructions
        that extract packed fields without explicit
        manual mask-and-shift steps.
  \item \code{ubfx xd, xn, \#lsb, \#width}:
        Unsigned bitfield extract.
\end{itemize}

\vspace{0.2cm}

\centering

\begin{tikzpicture}[scale=0.9, every node/.style={transform shape}]

  \draw[fill=softgray, draw=navy]
    (0,0) rectangle (9.0,0.55);

  \draw[fill=gold!40, draw=navy, thick]
    (4.0,0) rectangle (6.5,0.55);

  \node at (5.25,0.275)
    {\small\textbf{Extracted Field (\bits{width} bits)}};

  \node[anchor=south,font=\tiny\bfseries]
    at (4.0,0.55)
    {Bit $(\text{lsb}+\text{width}-1)$};

  \node[anchor=south,font=\tiny\bfseries]
    at (6.5,0.55)
    {Bit $\text{lsb}$};

  \node[anchor=south,font=\tiny\bfseries]
    at (0.2,0.55)
    {Bit 63};

  \node[anchor=south,font=\tiny\bfseries]
    at (8.8,0.55)
    {Bit 0};

\end{tikzpicture}

\vspace{0.3cm}

\begin{block}{Example: Extract 8-bit Field Starting at Bit 16}
\small

\begin{tabular}{lll}
\textbf{Standard Approach:}
&
\code{lsr x0, x1, \#16}
\quad $\to$ \quad
\code{and x0, x0, \#0xff}
&
(2 instructions)
\\

\textbf{Bitfield Extract:}
&
\code{ubfx x0, x1, \#16, \#8}
&
\key{1 instruction!}
\\
\end{tabular}

\end{block}

\end{frame}

\begin{frame}{The NZCV Condition Flags}

\begin{itemize}\setlength{\itemsep}{5pt}
  \item Flag bits reside in \bits{pstate}
        and record status of the most recent instruction that updates NZCV.
  \item Common flag-setting forms include instructions
        with an ``\code{s}'' suffix and explicit comparisons.
\end{itemize}

\begin{center}
\small
\setlength{\tabcolsep}{6pt}

\begin{tabular}{cll}
\toprule
\textbf{Flag}
&
\textbf{Name}
&
\textbf{Condition Set to 1}
\\
\midrule

\key{N}
&
\textbf{Negative}
&
The most-significant bit of the result is 1.
\\

\key{Z}
&
\textbf{Zero}
&
Result is exactly zero.
\\

\key{C}
&
\textbf{Carry}
&
Unsigned carry-out (addition) or \key{no borrow} (subtraction).
\\

\key{V}
&
\textbf{Overflow}
&
The signed mathematical result cannot be represented in the destination width.
\\
\bottomrule
\tabularnewline
\end{tabular}

\end{center}

\vspace{-0.5cm}

\begin{block}{Carry Flag in Subtraction}
In AArch64 subtraction ($A-B$),
$\text{C}=1$ indicates $A \ge B$
(no borrow required).
$\text{C}=0$ indicates $A < B$
(unsigned borrow occurred).
\end{block}

\end{frame}

% ============================================================
% Slide 9
% ============================================================

\begin{frame}{Comparison Instructions: \texttt{cmp} and \texttt{tst}}
\large

\begin{itemize}\setlength{\itemsep}{6pt}
  \item Comparisons evaluate values,
        set NZCV flags,
        and discard the computed output.
\end{itemize}

\vspace{0.15cm}

\begin{center}
\small
\setlength{\tabcolsep}{6pt}

\begin{tabular}{lll}
\toprule
\textbf{Instruction}
&
\textbf{Equivalent Machine Operation}
&
\textbf{Flags Set}
\\
\midrule

\code{cmp x0, x1}
&
\code{subs xzr, x0, x1}
&
Sets NZCV based on $x_0-x_1$
\\

\code{cmn x0, x1}
&
\code{adds xzr, x0, x1}
&
Sets NZCV based on $x_0+x_1$ (Compare Negative)
\\

\code{tst x0, x1}
&
\code{ands xzr, x0, x1}
&
Sets N and Z from result; clears C and V to 0 (Bit Test)
\\
\bottomrule
\tabularnewline
\end{tabular}

\end{center}

\vspace{0.25cm}
\normalsize

\begin{block}{Checking Flags After \texttt{cmp x0, x1}}

\begin{itemize}\setlength{\itemsep}{2pt}
  \item If $x_0 == x_1$:
        \key{Z flag is set to 1}.
  \item If $x_0 < x_1$ (signed):
        \key{$\text{N} \neq \text{V}$}.
  \item If $x_0 < x_1$ (unsigned):
        \key{$\text{C}=0$}.
\end{itemize}

\end{block}

\end{frame}

\begin{frame}{Condition Codes: Signed vs.\ Unsigned Branches}
\vspace{-0.15cm}

\begin{center}
\small
\setlength{\tabcolsep}{6pt}

\begin{tabular}{llll}
\toprule
\textbf{Branch}
&
\textbf{Meaning}
&
\textbf{Flag Condition}
&
\textbf{Comparison Context}
\\
\midrule

\code{b.eq}
&
Equal
&
$\text{Z} == 1$
&
Signed / Unsigned
\\

\code{b.ne}
&
Not Equal
&
$\text{Z} == 0$
&
Signed / Unsigned
\\

\midrule

\code{b.lt}
&
Signed Less Than ($<$)
&
$\text{N} \neq \text{V}$
&
\key{Signed}
\\

\code{b.le}
&
Signed Less or Equal ($\le$)
&
$(\text{Z} == 1) \lor (\text{N} \neq \text{V})$
&
\key{Signed}
\\

\code{b.gt}
&
Signed Greater Than ($>$)
&
$(\text{Z} == 0) \land (\text{N} == \text{V})$
&
\key{Signed}
\\

\code{b.ge}
&
Signed Greater or Equal ($\ge$)
&
$\text{N} == \text{V}$
&
\key{Signed}
\\

\midrule

\code{b.lo}
&
Unsigned Lower ($<$)
&
$\text{C} == 0$
&
\key{Unsigned / Addresses}
\\

\code{b.hs}
&
Unsigned Higher or Same ($\ge$)
&
$\text{C} == 1$
&
\key{Unsigned / Addresses}
\\

\code{b.hi}
&
Unsigned Higher ($>$)
&
$(\text{C} == 1) \land (\text{Z} == 0)$
&
\key{Unsigned / Addresses}
\\

\code{b.ls}
&
Unsigned Lower or Equal ($\le$)
&
$(\text{C} == 0) \lor (\text{Z} == 1)$
&
\key{Unsigned / Addresses}
\\

\bottomrule
\end{tabular}

\end{center}

\vspace{-0.35cm}
\normalsize

\begin{block}{Usage Context}
Unsigned conditions are appropriate
when the program semantics treat the values
as unsigned or address-like quantities.
\end{block}

\end{frame}

\begin{frame}{Fused Branch-and-Test: \texttt{cbz} and \texttt{cbnz}}
\large

\begin{itemize}\setlength{\itemsep}{6pt}

  \item \key{Compare and Branch on Zero (\code{cbz}) / Non-Zero (\code{cbnz}):}
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item Tests register contents directly;
          does \textbf{not} alter NZCV flags.
    \item Saves 1 instruction compared to an explicit
          \code{cmp} $+$ \code{b.eq}.
  \end{itemize}

\end{itemize}

\vspace{0.2cm}

\begin{columns}[T,onlytextwidth]

\begin{column}{0.48\textwidth}

\begin{block}{Null Pointer Validation}
\small

\begin{tabular}{ll}
\code{// If ptr == NULL, jump}
&
\\
\code{cbz x0, .Lhandle\_null}
&
\\
\code{// Safe memory access}
&
\\
\code{ldr x1, [x0]}
&
\\
\end{tabular}

\end{block}

\end{column}

\begin{column}{0.48\textwidth}

\begin{block}{Loop Counter Decrement}
\small

\begin{tabular}{ll}
\code{.Lloop:} & \\
\code{sub x1, x1, \#1}
&
\text{\scriptsize // count-- (flags not needed)}
\\
\code{cbnz x1, .Lloop}
&
\text{\scriptsize // Loop until count == 0}
\\
\end{tabular}

\end{block}

\end{column}

\end{columns}

\vspace{0.25cm}
\normalsize

\key{Single Bit Check:}
\code{tbz x0, \#0, label}
branches if bit 0 is zero
(even number test).

\end{frame}

\begin{frame}[fragile]{Translating \texttt{if-then-else} Blocks}

\begin{columns}[T,onlytextwidth]

\begin{column}{0.44\textwidth}

\begin{block}{High-Level C Code}
\small

\begin{verbatim}
if (x > y) {
    result = x - y;
} else {
    result = y - x;
}
\end{verbatim}

\end{block}

\end{column}

\begin{column}{0.52\textwidth}

\begin{block}{AArch64 Assembly Mapping}
\small

\begin{tabular}{ll}
\multicolumn{2}{l}{\textit{// Assume: x, y are signed int64\_t}} \\
\multicolumn{2}{l}{\textit{// x0=x, x1=y, x2=result}} \\
\code{cmp  x0, x1}
&
\text{\scriptsize // Compare x and y}
\\
\code{b.le .Lelse}
&
\text{\scriptsize // Invert test: jump if x <= y}
\\
\code{sub  x2, x0, x1}
&
\text{\scriptsize // Then-body: result = x - y}
\\
\code{b    .Lend}
&
\text{\scriptsize // Skip else block}
\\
\code{.Lelse:}
&
\\
\code{sub  x2, x1, x0}
&
\text{\scriptsize // Else-body: result = y - x}
\\
\code{.Lend:}
&
\\
\end{tabular}

\end{block}

\end{column}

\end{columns}

\vspace{0.3cm}
\normalsize

\key{Assembly Pattern:}
Invert the high-level condition
to fall through into the ``then'' path.

\end{frame}

\begin{frame}[fragile]{Loop Translation: Inverted Loop Pattern}

\begin{columns}[T,onlytextwidth]

\begin{column}{0.44\textwidth}

\begin{block}{High-Level C Loop}
\small

\begin{verbatim}
// Compute array sum
int64_t i = 0;
int64_t n = ...; // signed bound
while (i < n) {
    sum += arr[i];
    i++;
}
\end{verbatim}

\end{block}

\end{column}

\begin{column}{0.52\textwidth}

\begin{block}{Bottom-Tested AArch64 Loop}
\small

\begin{tabular}{ll}
\multicolumn{2}{l}{\textit{// x0=arr, x1=n, x2=i, x3=sum}} \\
\code{mov  x2, xzr}
&
\text{\scriptsize // i = 0}
\\
\code{mov  x3, xzr}
&
\text{\scriptsize // sum = 0}
\\
\code{b    .Ltest}
&
\text{\scriptsize // Jump to test first}
\\
\code{.Lbody:}
&
\\
\code{ldr  x4, [x0, x2, lsl \#3]}
&
\text{\scriptsize // Load arr[i]}
\\
\code{add  x3, x3, x4}
&
\text{\scriptsize // sum += arr[i]}
\\
\code{add  x2, x2, \#1}
&
\text{\scriptsize // i++}
\\
\code{.Ltest:}
&
\\
\code{cmp  x2, x1}
&
\text{\scriptsize // i < n?}
\\
\code{b.lt .Lbody}
&
\text{\scriptsize // Loop if true}
\\
\end{tabular}

\end{block}

\end{column}

\end{columns}

\vspace{0.25cm}
\normalsize

\key{Efficiency Rule:}
Placing the branch at the bottom
requires only one loop-closing conditional branch
in the steady-state loop body.

\end{frame}

\begin{frame}{Branch Penalties and Prediction}

\begin{itemize}\setlength{\itemsep}{6pt}

  \item Modern processors often fetch instructions
        before a conditional branch is resolved.

  \item The processor predicts the branch direction
        and continues along the predicted path.

  \item A correct prediction allows execution
        to continue with little disruption.

  \item A misprediction requires incorrect work
        to be discarded and execution redirected.

  \item The \key{misprediction penalty}
        depends on pipeline depth
        and front-end organization.

  \item Predictable branches are usually inexpensive;
        irregular, data-dependent branches
        are more difficult to predict.

  \item Therefore, a branch is not inherently slow:
        the main cost comes from
        \key{incorrect prediction}.

\end{itemize}

\end{frame}

\begin{frame}{Branchless Conditional Selection}

\begin{block}{Conditional Select}
AArch64 provides \code{csel}
to choose between two register values
using the NZCV condition flags
without introducing a control-flow branch.
\end{block}

\vspace{0.2cm}

\begin{block}{Signed Maximum}
\small

Suppose the desired operation is:

\begin{center}
\code{result = (x1 > x2) ? x1 : x2}
\end{center}

\begin{center}
\begin{tabular}{ll}
\code{cmp  x1, x2}
&
\text{\scriptsize // Set NZCV from x1 - x2}
\\
\code{csel x0, x1, x2, gt}
&
\text{\scriptsize // Signed greater-than selection}
\\
\end{tabular}
\end{center}

For unsigned values,
the corresponding condition would be \code{hi}.
\end{block}

\begin{frame}{When Is Branchless Code Better?}

\begin{itemize}\setlength{\itemsep}{6pt}

  \item Branchless selection avoids
        \key{branch-misprediction risk}
        because control flow does not depend
        on the condition.

  \item \code{csel} replaces a control-flow dependency
        with a data dependency on the condition flags.

  \item Branchless code is often attractive
        when the condition is difficult to predict
        or when both candidate values are already available.

  \item A highly predictable branch
        can still perform very well
        and may outperform conditional selection.

  \item Performance therefore depends on
        the processor microarchitecture
        and the behavior of the workload.

\end{itemize}

\vspace{0.15cm}

\begin{center}
\colorbox{lightblue}{%
  \parbox{0.88\textwidth}{%
    \centering
    \textbf{Key Point:}
    Branchless code is not automatically faster;
    it trades branch-prediction risk for data dependencies.
  }%
}
\end{center}

\end{frame}

\begin{frame}{Topic 4: Comprehensive Summary}

\begin{itemize}\setlength{\itemsep}{6pt}

  \item \key{Instruction Encoding:}
        Many AArch64 instructions encode
        a shifted register operand within a single instruction.

  \item \key{Bitwise Primitives:}
        Bitwise operations apply independently
        to each corresponding bit position,
        with native support for
        \code{and}, \code{orr}, \code{eor},
        and bit clearing via \code{bic};
        field extraction via \code{ubfx}.

  \item \key{Flag-Setting Forms:}
        Common flag-setting forms include
        ``\code{s}'' variants
        (\code{adds}, \code{subs})
        and explicit comparisons
        (\code{cmp}, \code{tst});
        carry flag is active ($C=1$)
        when subtraction does not borrow.

  \item \key{Control Flow Idioms:}
        A common control-flow layout
        inverts the condition so one path falls through;
        loop inversion cuts branch overhead
        to one loop-closing branch in steady state.

  \item \key{Branch Elimination:}
        Conditional select instructions
        (\code{csel}, \code{cset})
        avoid misprediction risk,
        though performance trade-offs depend
        on pipeline depth and dependency chains.

\end{itemize}

\end{frame}
