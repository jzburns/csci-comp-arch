% ============================================================

% Slide Header: Why Integer Arithmetic Matters
% ============================================================
\begin{frame}{Why Integer Arithmetic Matters}
\begin{block}{Integer Operations Are Everywhere}
Integer arithmetic is not limited to explicit calculator-like expressions. It appears constantly in address calculation, loop control, array indexing, comparisons, and resource accounting.
\end{block}

\vspace{0.25cm}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Program-Level Examples}
\begin{itemize}
  \item Increment loop counters.
  \item Compute array offsets.
  \item Compare bounds and indices.
  \item Update pointer values.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Hardware-Level Consequence}
\begin{itemize}
  \item Arithmetic uses fixed-width registers.
  \item Results can wrap around.
  \item Flags record selected facts about results.
  \item Interpretation is supplied by software.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% ============================================================
% Slide 5
% Slide Header: Arithmetic Operates on Bit Patterns
% ============================================================
\begin{frame}{Arithmetic Operates on Bit Patterns}
\begin{block}{Core Principle}
The ALU operates on finite-width bit patterns. Signedness is not a property of the adder itself; it is a property of how software interprets the bits and which branch conditions it later uses.
\end{block}

\vspace{0.25cm}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Same Instruction}
\large
\hspace{1.5cm}\code{add x0, x1, x2}
\normalsize

\begin{itemize}
  \item Adds two 64-bit values.
  \item Produces one 64-bit result.
  \item Does not by itself say signed or unsigned.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Different Readings}
\begin{itemize}
  \item Unsigned: magnitude from $0$ to $2^{64}-1$.
  \item Signed: two's-complement range from $-2^{63}$ to $2^{63}-1$.
  \item Flags help evaluate special cases.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% ============================================================
% Slide 6
% Slide Header: Binary Addition and Carry
% ============================================================
\begin{frame}{Binary Addition and Carry}
\begin{columns}[T,onlytextwidth]

\column{0.46\textwidth}
\begin{block}{Column Addition}
Binary addition proceeds bit by bit, carrying into the next higher column when needed.

\vspace{0.15cm}

\begin{center}
\large
\begin{tabular}{@{}r@{\;}r@{}}
& \code{1111} \\
\code{+} & \code{0001} \\[0.03cm]
\multicolumn{2}{c}{\rule{1.65cm}{0.5pt}} \\[0.03cm]
& \code{1 0000}
\end{tabular}
\end{center}
\end{block}

\column{0.50\textwidth}
\begin{block}{Unsigned Meaning}
\begin{itemize}
  \item In 4 bits, only the lower \,\code{0000}\, remains.
  \item The extra leading bit is carry-out.
  \item For unsigned arithmetic, carry-out means the mathematical result exceeded the available range.
\end{itemize}
\end{block}
\end{columns}

\vspace{0.15cm}

\begin{block}{Fixed-Width Result}
An $n$-bit register keeps only $n$ result bits. Extra information may be reflected in condition flags.
\end{block}
\end{frame}

% ============================================================
% Slide 7
% Slide Header: Unsigned Ranges
% ============================================================
\begin{frame}{Unsigned Ranges}
\begin{block}{Range Formula}
An $n$-bit unsigned integer represents values from $0$ to $2^n - 1$.
\end{block}

\vspace{0.25cm}

\begin{center}
\small
\renewcommand{\arraystretch}{1.18}
\begin{tabular}{
  >{\raggedright\arraybackslash}m{3.0cm}
  >{\raggedright\arraybackslash}m{4.0cm}
  >{\raggedright\arraybackslash}m{4.0cm}
}
\rowcolor{navy}
{\color{white}\bfseries Width} &
{\color{white}\bfseries Smallest} &
{\color{white}\bfseries Largest} \\
\rowcolor{softgray}
8 bits & $0$ & $255$ \\
\rowcolor{white}
32 bits & $0$ & $2^{32}-1$ \\
\rowcolor{softgray}
64 bits & $0$ & $2^{64}-1$ \\
\end{tabular}
\end{center}

\vspace{0.25cm}

\begin{block}{Teaching Point}
Unsigned overflow is not mysterious. The mathematical result is too large for the fixed-width container, so the stored result wraps modulo $2^n$.
\end{block}
\end{frame}

% ============================================================
% Slide 8
% Slide Header: Two's-Complement Subtraction
% ============================================================
\begin{frame}{Two's-Complement Subtraction}
\begin{block}{Hardware Idea}
Subtraction can reuse the same adder by adding the two's-complement negation.
\end{block}

\vspace{0.05cm}

\begin{center}
$A - B = A + (\sim B) + 1$
\end{center}

\vspace{-0.05cm}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{AArch64 Example}
\large
\hspace{1.5cm}\code{sub x0, x1, x2}
\normalsize

\begin{itemize}
  \item Computes \code{x1 - x2}.
  \item Writes the fixed-width result to \code{x0}.
  \item Does not update NZCV flags.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Important Interpretation}
\begin{itemize}
  \item Negative values are encoded using two's complement.
  \item The same adder can serve add and subtract.
  \item Flags are needed to reason about borrow and overflow.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% ============================================================
% Slide 9
% Slide Header: Signed vs. Unsigned Interpretation
% ============================================================
\begin{frame}{Signed vs. Unsigned Interpretation}
\begin{block}{Same Bits, Different Values}
The same bit pattern can have different numeric meanings depending on whether software treats it as unsigned or signed two's complement.
\end{block}

\vspace{0.2cm}

\begin{center}
\small
\renewcommand{\arraystretch}{1.18}
\begin{tabular}{
  >{\raggedright\arraybackslash}m{3.2cm}
  >{\raggedright\arraybackslash}m{3.8cm}
  >{\raggedright\arraybackslash}m{4.2cm}
}
\rowcolor{navy}
{\color{white}\bfseries 8-bit pattern} &
{\color{white}\bfseries Unsigned} &
{\color{white}\bfseries Signed two's complement} \\
\rowcolor{softgray}
\code{01111111} & $127$ & $127$ \\
\rowcolor{white}
\code{10000000} & $128$ & $-128$ \\
\rowcolor{softgray}
\code{11111111} & $255$ & $-1$ \\
\end{tabular}
\end{center}

\vspace{0.2cm}

\begin{block}{Connection to AArch64}
The instruction result is a bit pattern. The choice of signed or unsigned branch determines how flags are interpreted later.
\end{block}
\end{frame}

% ============================================================
% Slide 10
% Slide Header: Signed Overflow
% ============================================================
\begin{frame}{Signed Overflow}
\begin{block}{Definition}
Signed overflow occurs when the true mathematical result of a two's-complement operation falls outside the signed range representable in the destination width.
\end{block}

\vspace{0.2cm}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Positive Overflow}
\large
\hspace{1.1cm}\code{0111 + 0001 = 1000}
\normalsize

\vspace{0.15cm}

In 4-bit signed arithmetic, $7+1$ cannot be represented. The stored pattern looks like $-8$.
\end{block}

\column{0.48\textwidth}
\begin{block}{Rule of Thumb}
\begin{itemize}
  \item Add two positives and get a negative: overflow.
  \item Add two negatives and get a positive: overflow.
  \item Add opposite signs: no signed overflow.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% ============================================================
% Slide 11
% Slide Header: Carry and Overflow Are Different
% ============================================================
\begin{frame}{Carry and Overflow Are Different}
\begin{block}{Do Not Treat Them as Synonyms}
Carry describes unsigned range behavior. Overflow describes signed two's-complement range behavior.
\end{block}

\vspace{0.2cm}

\begin{center}
\small
\renewcommand{\arraystretch}{1.18}
\begin{tabular}{
  >{\raggedright\arraybackslash}m{3.2cm}
  >{\raggedright\arraybackslash}m{4.2cm}
  >{\raggedright\arraybackslash}m{4.0cm}
}
\rowcolor{navy}
{\color{white}\bfseries Flag} &
{\color{white}\bfseries Main use} &
{\color{white}\bfseries Example idea} \\
\rowcolor{softgray}
\code{C} & unsigned carry or no-borrow & Did unsigned arithmetic exceed range? \\
\rowcolor{white}
\code{V} & signed overflow & Did signed result exceed two's-complement range? \\
\end{tabular}
\end{center}

\vspace{0.25cm}

\begin{block}{Course Connection}
Later branch instructions use different flag combinations for signed and unsigned comparisons.
\end{block}
\end{frame}

\begin{center}

\end{tikzpicture}
\end{center}

\vspace{0.25cm}

\begin{itemize}
  \item \key{N}: result sign bit is 1.
  \item \key{Z}: result is exactly zero.
  \item \key{C}: carry-out for addition; no-borrow for subtraction.
  \item \key{V}: signed two's-complement overflow.
\end{itemize}

\vspace{0.1cm}

\begin{block}{Interpretation Rule}
\code{C} is mainly used for unsigned reasoning; \code{V} is mainly used for signed overflow reasoning.
\end{block}
\end{frame}

% ============================================================
% Slide 14
% Slide Header: Flag-Setting Instructions
% ============================================================
\begin{frame}{Flag-Setting Instructions}
\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}
\begin{block}{No Flag Update}
\large
\hspace{2cm}\code{add x0, x1, x2}\\[0.12cm]
\hspace{2cm}\code{sub x0, x1, x2}
\normalsize

\begin{itemize}
  \item Write a register result.
  \item Leave NZCV unchanged.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Flag Update}
\large
\hspace{2cm}\code{adds x0, x1, x2}\\[0.12cm]
\hspace{2cm}\code{subs x0, x1, x2}
\normalsize

\begin{itemize}
  \item Write a register result.
  \item Update NZCV from that result.
\end{itemize}
\end{block}
\end{columns}

\vspace{0.2cm}

\begin{block}{Important Limitation}
The \code{s} suffix is useful here, but do not teach it as a universal rule that applies to every AArch64 mnemonic.
\end{block}
\end{frame}

% ============================================================
% Slide 15
% Slide Header: Compare as Flag-Setting Subtraction
% ============================================================
\begin{frame}{\code{cmp} as Flag-Setting Subtraction}
\begin{block}{Comparison Without Keeping the Difference}
\code{cmp} subtracts for flags and discards the numeric result.
\end{block}

\vspace{0.25cm}

\large
\hspace{4.3cm}\code{cmp x0, x1}\\[0.15cm]
\hspace{4.3cm}\code{subs xzr, x0, x1}
\normalsize

\vspace{0.25cm}

\begin{itemize}
  \item Both lines express the same underlying comparison mechanism.
  \item \code{x0 - x1} updates NZCV.
  \item The result is written to \code{xzr}, so it is discarded.
\end{itemize}
\end{frame}

% ============================================================
% Slide 16
% Slide Header: Carry Means No-Borrow After Subtraction
% ============================================================
\begin{frame}{Carry Means No-Borrow After Subtraction}
\begin{block}{AArch64 Subtraction Rule}
After \code{subs} or \code{cmp}, \code{C=1} means the subtraction did \key{not} require an unsigned borrow.
\end{block}

\vspace{0.2cm}

\begin{center}
\small
\renewcommand{\arraystretch}{1.18}
\begin{tabular}{
  >{\raggedright\arraybackslash}m{3.2cm}
  >{\raggedright\arraybackslash}m{3.8cm}
  >{\raggedright\arraybackslash}m{4.2cm}
}
\rowcolor{navy}
{\color{white}\bfseries Operation} &
{\color{white}\bfseries Unsigned relation} &
{\color{white}\bfseries Carry flag} \\
\rowcolor{softgray}
\code{5 - 3} & $5 \ge 3$ & \code{C=1}: no borrow \\
\rowcolor{white}
\code{3 - 5} & $3 < 5$ & \code{C=0}: borrow needed \\
\end{tabular}
\end{center}

\vspace{0.2cm}

\begin{block}{Common Mistake}
Do not read \code{C=1} after subtraction as ``borrow occurred.'' In AArch64, it means \key{no borrow}.
\end{block}
\end{frame}

\begin{block}{After a Compare}
The same \code{cmp} can support signed or unsigned comparisons. The branch mnemonic determines how the flags are interpreted.
\end{block}

\vspace{0.15cm}

\begin{center}
\small
\renewcommand{\arraystretch}{1.18}
\begin{tabular}{

}
\rowcolor{navy}

{\color{white}\bfseries Condition} \\
\rowcolor{softgray}

\rowcolor{white}

\end{center}

\vspace{0.15cm}

\large

\quad
\code{b.lt signed\_less}
\quad

\normalsize
\end{frame}

\begin{frame}{Multiplication and Division Basics}
\begin{block}{Architectural Category}
Multiply and divide read registers and write register results; latency is implementation-dependent.
\end{block}

\vspace{0.25cm}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}
\begin{block}{Multiply}
\large
\hspace{1.5cm}\code{mul x0, x1, x2}
\normalsize

\begin{itemize}
  \item Computes a register product.
  \item May use a distinct multiply unit internally.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Divide}
\large
\hspace{1.25cm}\code{udiv x0, x1, x2}\\[0.12cm]
\hspace{1.25cm}\code{sdiv x0, x1, x2}
\normalsize

\begin{itemize}
  \item \code{udiv}: unsigned division.
  \item \code{sdiv}: signed division.
\end{itemize}
\end{block}

\end{columns}
\end{frame}

% ============================================================
% Slide 20
% Slide Header: Shifts and Powers of Two
% ============================================================
\begin{frame}{Shifts and Powers of Two}
\begin{center}
\small
\renewcommand{\arraystretch}{1.18}
\begin{tabular}{
  >{\raggedright\arraybackslash}m{2.5cm}
  >{\raggedright\arraybackslash}m{4.0cm}
  >{\raggedright\arraybackslash}m{5.0cm}
}
\rowcolor{navy}
{\color{white}\bfseries Shift} &
{\color{white}\bfseries Meaning} &
{\color{white}\bfseries Typical interpretation} \\
\rowcolor{softgray}
\code{lsl \#n} &
shift left, zeros enter right &
multiply by $2^n$ if overflow is ignored \\
\rowcolor{white}
\code{lsr \#n} &
logical right, zeros enter left &
unsigned division by powers of two \\
\rowcolor{softgray}
\code{asr \#n} &
arithmetic right, sign bit is replicated &
preserves signed negative values \\
\rowcolor{white}
\code{ror \#n} &
rotate right, bits wrap around &
bit manipulation, not division \\
\end{tabular}
\end{center}

\vspace{0.2cm}

\begin{block}{Key Distinction}
\code{lsr} and \code{asr} differ when the top bit is set. That difference matters for unsigned versus signed interpretation.
\end{block}
\end{frame}

% ============================================================
% Slide 21
% Slide Header: Shift Example
% ============================================================
\begin{frame}{Shift Example}
\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}
\begin{block}{Logical Shift Right}
\large
\hspace{1.4cm}\code{lsr x0, x1, \#1}
\normalsize

\begin{itemize}
  \item Zero enters the top bit.
  \item Good for unsigned bit patterns.
  \item Does not preserve a negative sign.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Arithmetic Shift Right}
\large
\hspace{1.4cm}\code{asr x0, x1, \#1}
\normalsize

\begin{itemize}
  \item Original sign bit is copied.
  \item Preserves signed interpretation.
  \item Useful for signed right shifts.
\end{itemize}
\end{block}

\end{columns}

\vspace{0.25cm}

\begin{block}{Qualification}
Shifts can resemble multiplication or division by powers of two, but finite width, overflow, and signed rounding behavior must be considered.
\end{block}
\end{frame}

% Slide Header: Execution Time and Performance
% ============================================================
\begin{frame}{Execution Time and Performance}
\begin{block}{Primary Metric}
For a fixed workload, performance is the inverse of execution time.
\end{block}

\vspace{0.2cm}

\begin{center}
\large
$\text{Performance} = \dfrac{1}{\text{Execution Time}}$
\end{center}

\vspace{0.2cm}

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}
\begin{block}{Latency}
\begin{itemize}
  \item Time to complete one task.
  \item Important for response time.
  \item Example: seconds per program run.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Throughput}
\begin{itemize}
  \item Work completed per unit time.
  \item Important for pipelines and servers.
  \item Example: instructions per cycle.
\end{itemize}
\end{block}

\end{columns}
\end{frame}

% Slide 25
% Slide Header: Instruction Count Is Dynamic
% ============================================================
\begin{frame}{Instruction Count Is Dynamic}
\begin{block}{Dynamic Instruction Count}
Dynamic instruction count is the number of instructions actually executed, not merely the number of instructions visible in a source or assembly listing.
\end{block}

\vspace{0.25cm}

\begin{itemize}
  \item Loops multiply the effect of a small instruction sequence.
  \item Branches choose which instructions execute.
  \item Function calls, recursion, and input data can change the count.
  \item Optimizing one program region may not matter if it rarely executes.
\end{itemize}

\vspace{0.2cm}

\begin{block}{Simple Example}
A three-instruction loop body executed $N$ times contributes approximately $3N$ dynamic instructions, before considering branch and loop-control instructions.
\end{block}
\end{frame}

% ============================================================
% Slide 26
% Slide Header: CPI and IPC
% ============================================================
\begin{frame}{CPI and IPC}
\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}
\begin{block}{Cycles Per Instruction}
\begin{center}
\large
$\text{CPI}=\dfrac{\text{cycles}}{\text{instructions}}$
\end{center}

\begin{itemize}
  \item Average over a workload.
  \item Can include stalls and cache misses.
  \item Lower is usually better.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Instructions Per Cycle}
\begin{center}
\large
$\text{IPC}=\dfrac{1}{\text{CPI}}$
\end{center}

\begin{itemize}
  \item Average completed instructions per cycle.
  \item Can exceed 1 on superscalar cores.
  \item Higher is usually better.
\end{itemize}
\end{block}

\end{columns}

\vspace{0.2cm}

\begin{block}{No Universal Cycle Count}
Instruction latency and throughput are microarchitecture-dependent, not fixed by the AArch64 ISA alone.
\end{block}
\end{frame}

\begin{frame}{Clock Frequency Is Not Performance}
\begin{block}{Common Mistake}
A higher clock rate does not automatically mean a shorter runtime. Instruction count and CPI matter just as much.
\end{block}

\vspace{0.2cm}

\begin{center}
\small
\renewcommand{\arraystretch}{1.18}

\end{frame}

% ============================================================
\begin{frame}{Topic 4 Takeaways}
\begin{itemize}
  \item Integer arithmetic operates on fixed-width bit patterns.
  \item Signedness comes from interpretation, not from the adder itself.
  \item \code{C} supports unsigned reasoning; \code{V} supports signed overflow reasoning.
  \item After subtraction, \code{C=1} means no unsigned borrow occurred.
  \item Shifts can support power-of-two scaling, but overflow and signed behavior matter.
  \item Execution time depends jointly on instruction count, CPI, and clock period.
\end{itemize}
\end{frame}
