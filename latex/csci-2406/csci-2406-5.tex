% ============================================================

% Slide Header: Arithmetic Operates on Bit Patterns
% ============================================================
\begin{frame}{Arithmetic Operates on Bit Patterns}
\begin{block}{Architectural View}
AArch64 integer arithmetic operates on fixed-width register contents. The instruction computes a bit-pattern result; signedness is an interpretation applied by software and branch conditions.
\end{block}
\vspace{0.2cm}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Same Operation}
\begin{center}
\large
\code{add x0, x1, x2}
\end{center}
\begin{itemize}
  \item Adds two 64-bit register values.
  \item Produces one 64-bit result.
  \item Does not by itself mean signed or unsigned.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Different Interpretations}
\begin{itemize}
  \item Unsigned: range is $0$ to $2^{64}-1$.
  \item Signed: two's complement range is $-2^{63}$ to $2^{63}-1$.
  \item Flags help interpret exceptional cases.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% ============================================================
% Slide 5
% Slide Header: Add and Subtract
% ============================================================
\begin{frame}{Add and Subtract}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Addition}
\begin{center}
\large
\code{add x0, x1, x2}
\end{center}
\begin{itemize}
  \item Computes \code{x1 + x2}.
  \item Writes the result to \code{x0}.
  \item Does not update NZCV flags.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Subtraction}
\begin{center}
\large
\code{sub x0, x1, x2}
\end{center}
\begin{itemize}
  \item Computes \code{x1 - x2}.
  \item Uses fixed-width modular arithmetic.
  \item Does not update NZCV flags.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.25cm}
\begin{block}{Key Point}
The same binary datapath can support signed and unsigned arithmetic. Interpretation comes from how the result and flags are used.
\end{block}
\end{frame}

% ============================================================
% Slide 6
% Slide Header: Subtraction Uses Two's Complement
% ============================================================
\begin{frame}{Subtraction Uses Two's Complement}
\begin{block}{Hardware Idea}
Subtraction can be implemented as addition of the two's-complement negation.
\end{block}
\vspace{0.15cm}
\begin{center}
\large
$A - B = A + (\sim B) + 1$
\end{center}
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Architectural Example}
\begin{center}
\large
\code{sub x0, x1, x2}
\end{center}
\begin{itemize}
  \item Reads \code{x1} and \code{x2}.
  \item Produces \code{x1 - x2}.
  \item Result is fixed-width.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Pedagogical Warning}
\begin{itemize}
  \item Negative results wrap modulo $2^n$.
  \item The same bits can be read as signed or unsigned.
  \item Flags are needed to reason about overflow and borrow.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}{NZCV Flags}
\begin{center}
\begin{tikzpicture}[
  fld/.style={draw=navy,thick,minimum width=2.2cm,minimum height=1.0cm,align=center,font=\bfseries}
]
\node[fld,fill=navy,text=white] (n) {N\\Negative};
\node[fld,fill=deepblue,text=white,right=.15cm of n] (z) {Z\\Zero};
\node[fld,fill=gold!35,text=navy,right=.15cm of z] (c) {C\\Carry};
\node[fld,fill=softgray,text=navy,right=.15cm of c] (v) {V\\Overflow};
\end{tikzpicture}
\end{center}
\vspace{0.25cm}
\begin{itemize}
  \item \key{N}: result sign bit is 1.
  \item \key{Z}: result is exactly zero.
  \item \key{C}: unsigned carry-out for addition; no-borrow for subtraction.
  \item \key{V}: signed two's-complement overflow.
\end{itemize}
\vspace{0.1cm}
\begin{block}{Interpretation Rule}
\code{C} is mainly useful for unsigned reasoning; \code{V} is mainly useful for signed reasoning.
\end{block}
\end{frame}

% ============================================================
% Slide 9
% Slide Header: Flag-Setting Arithmetic
% ============================================================
\begin{frame}{Flag-Setting Arithmetic}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{No Flag Update}
\begin{center}
\large
\code{add x0, x1, x2}\\[0.12cm]
\code{sub x0, x1, x2}
\end{center}
\begin{itemize}
  \item Write a register result.
  \item Leave NZCV unchanged.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Flag Update}
\begin{center}
\large
\code{adds x0, x1, x2}\\[0.12cm]
\code{subs x0, x1, x2}
\end{center}
\begin{itemize}
  \item Write a register result.
  \item Update NZCV from that result.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.25cm}
\begin{block}{Important Limitation}
Do not overgeneralize this as “add \code{s} to every mnemonic.” AArch64 has specific flag-setting instruction forms and aliases.
\end{block}
\end{frame}

% ============================================================
% Slide 10
% Slide Header: Carry and Overflow Are Different
% ============================================================
\begin{frame}{Carry and Overflow Are Different}
\begin{block}{One Operation, Two Interpretations}
The same addition can be valid in one interpretation and overflow in another.
\end{block}
\vspace{0.15cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{>{\raggedright\arraybackslash}m{3.4cm}>{\raggedright\arraybackslash}m{3.6cm}>{\raggedright\arraybackslash}m{4.2cm}}
\rowcolor{navy}
{\color{white}\bfseries View} & {\color{white}\bfseries Example} & {\color{white}\bfseries Meaning} \\
\rowcolor{softgray}
Unsigned & $255 + 1 = 0$ in 8 bits & Carry-out indicates unsigned wraparound. \\
\rowcolor{white}
Signed & $127 + 1 = -128$ in 8 bits & Overflow indicates signed range exceeded. \\
\end{tabular}
\end{center}
\vspace{0.25cm}
\begin{block}{AArch64 Consequence}
Unsigned branches inspect \code{C}; signed branches inspect combinations involving \code{N} and \code{V}.
\end{block}
\end{frame}

% ============================================================
% Slide 11
% Slide Header: Carry Means No-Borrow After Subtraction
% ============================================================
\begin{frame}{Carry Means No-Borrow After Subtraction}
\begin{block}{AArch64 Subtraction Rule}
After \code{subs} or \code{cmp}, \code{C=1} means the subtraction did \key{not} require an unsigned borrow.
\end{block}
\vspace{0.15cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{>{\raggedright\arraybackslash}m{3.4cm}>{\raggedright\arraybackslash}m{3.4cm}>{\raggedright\arraybackslash}m{4.5cm}}
\rowcolor{navy}
{\color{white}\bfseries Operation} & {\color{white}\bfseries Unsigned Relation} & {\color{white}\bfseries Carry Flag} \\
\rowcolor{softgray}
\code{5 - 3} & $5 \ge 3$ & \code{C=1}: no borrow \\
\rowcolor{white}
\code{3 - 5} & $3 < 5$ & \code{C=0}: borrow needed \\
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{block}{Common Mistake}
Do not read \code{C=1} after subtraction as “borrow occurred.” In AArch64, it means the opposite: \key{no borrow}.
\end{block}
\end{frame}

% ============================================================
% Slide 12
% Slide Header: cmp Is an Alias
% ============================================================
\begin{frame}{\code{cmp} Is an Alias}
\begin{block}{Comparison Without Keeping the Difference}
\code{cmp} performs a subtraction for flags, but discards the subtraction result.
\end{block}
\vspace{0.25cm}
\begin{center}
\begin{minipage}{0.42\textwidth}
\raggedright
\large
\code{cmp x0, x1}\\[0.15cm]
\code{subs xzr, x0, x1}
\end{minipage}
\end{center}
\vspace{0.25cm}
\begin{itemize}
  \item The two lines express the same underlying operation.
  \item \code{x0 - x1} is used to update NZCV.
  \item The result is written to \code{xzr}, so it is discarded.
\end{itemize}
\end{frame}

% ============================================================
% Slide 13
% Slide Header: Signed and Unsigned Branches
% ============================================================
\begin{frame}{Signed and Unsigned Branches}
\begin{block}{After a Compare}
The same \code{cmp} can support signed or unsigned comparisons depending on which conditional branch is used.
\end{block}
\vspace{0.15cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{>{\raggedright\arraybackslash}m{3.0cm}>{\raggedright\arraybackslash}m{4.0cm}>{\raggedright\arraybackslash}m{4.0cm}}
\rowcolor{navy}
{\color{white}\bfseries Branch} & {\color{white}\bfseries Interpretation} & {\color{white}\bfseries Condition} \\
\rowcolor{softgray}
\code{b.lt label} & signed less than & $N \ne V$ \\
\rowcolor{white}
\code{b.ge label} & signed greater/equal & $N = V$ \\
\rowcolor{softgray}
\code{b.lo label} & unsigned lower & $C=0$ \\
\rowcolor{white}
\code{b.hs label} & unsigned higher/same & $C=1$ \\
\end{tabular}
\end{center}
\vspace{0.15cm}
\begin{center}
\large
\code{cmp x0, x1}\qquad \code{b.lt signed\_less}\qquad \code{b.lo unsigned\_lower}
\end{center}
\end{frame}

% Slide Header: Worked Comparison Example
% ============================================================
\begin{frame}{Worked Comparison Example}
\begin{block}{Same Bits, Different Ordering}
Consider 8-bit values for intuition: \code{0xff} and \code{0x01}.
\end{block}
\vspace{0.15cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{>{\raggedright\arraybackslash}m{3.2cm}>{\raggedright\arraybackslash}m{3.6cm}>{\raggedright\arraybackslash}m{4.2cm}}
\rowcolor{navy}
{\color{white}\bfseries Interpretation} & {\color{white}\bfseries Values} & {\color{white}\bfseries Result} \\
\rowcolor{softgray}
Unsigned & $255$ and $1$ & $255 > 1$ \\
\rowcolor{white}
Signed two's complement & $-1$ and $1$ & $-1 < 1$ \\
\end{tabular}
\end{center}
\vspace{0.25cm}
\begin{block}{Teaching Point}
The bit patterns are unchanged. The branch mnemonic determines whether the flags are interpreted as signed or unsigned evidence.
\end{block}
\end{frame}

% ============================================================
% Slide 16
% Slide Header: Logical Instructions Operate Bit-by-Bit
% ============================================================
\begin{frame}{Logical Instructions Operate Bit-by-Bit}
\begin{block}{Bitwise Meaning}
Logical instructions operate independently on corresponding bit positions. They do not treat the register as a signed or unsigned number.
\end{block}
\vspace{0.2cm}
\begin{center}
\large
\code{and x0, x1, x2}\qquad
\code{orr x0, x1, x2}\qquad
\code{eor x0, x1, x2}
\end{center}
\vspace{0.2cm}
\begin{itemize}
  \item \code{and}: keeps bits set in both operands.
  \item \code{orr}: sets bits present in either operand.
  \item \code{eor}: sets bits that differ between operands.
\end{itemize}
\end{frame}

% ============================================================
% Slide 17
% Slide Header: Bitwise AND Example
% ============================================================
\begin{frame}{Bitwise AND Example}
\begin{block}{Masking}
A mask selects which bit positions should survive an operation.
\end{block}
\vspace{0.15cm}
\begin{center}
\large
\begin{minipage}{0.33\textwidth}
\raggedright
\code{  11001010}\\[0.08cm]
\code{\& 10110100}\\[0.08cm]
\code{----------}\\[0.08cm]
\code{  10000000}
\end{minipage}
\end{center}
\vspace{0.15cm}
\begin{block}{AArch64 Example}
\begin{center}
\large
\code{and x0, x1, x2}
\end{center}
The result contains only the bit positions that were set in both source registers.
\end{block}
\end{frame}

% ============================================================
% Slide 18
% Slide Header: tst Is an Alias
% ============================================================
\begin{frame}{\code{tst} Is an Alias}
\begin{block}{Testing Bits Without Keeping the Result}
\code{tst} computes a bitwise AND for flags, but discards the AND result.
\end{block}
\vspace{0.25cm}
\begin{center}
\begin{minipage}{0.42\textwidth}
\raggedright
\large
\code{tst x0, x1}\\[0.15cm]
\code{ands xzr, x0, x1}
\end{minipage}
\end{center}
\vspace{0.25cm}
\begin{itemize}
  \item The AND result updates NZCV.
  \item The result is written to \code{xzr}, so it is discarded.
  \item This is useful for asking whether selected bits are zero or nonzero.
\end{itemize}
\end{frame}

\begin{minipage}{0.7\textwidth}

\code{orr x0, x1, x2}\\[0.12cm]

\end{block}

\end{columns}

% ============================================================
% Slide 21
% Slide Header: Shift Operations
% ============================================================
\begin{frame}{Shift Operations}
\begin{center}
\small
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{>{\raggedright\arraybackslash}m{2.5cm}>{\raggedright\arraybackslash}m{4.0cm}>{\raggedright\arraybackslash}m{5.0cm}}
\rowcolor{navy}
{\color{white}\bfseries Shift} & {\color{white}\bfseries Meaning} & {\color{white}\bfseries Typical Interpretation} \\
\rowcolor{softgray}
\code{lsl \#n} & shift left, zeros enter right & multiply by $2^n$ if overflow is ignored \\
\rowcolor{white}
\code{lsr \#n} & logical right, zeros enter left & unsigned division by powers of two \\
\rowcolor{softgray}
\code{asr \#n} & arithmetic right, sign bit is replicated & preserves signed negative values \\
\rowcolor{white}
\code{ror \#n} & rotate right, bits wrap around & bit manipulation, not division \\
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{block}{Key Distinction}
\code{lsr} and \code{asr} differ when the top bit is set. That difference matters for unsigned versus signed interpretation.
\end{block}
\end{frame}

% ============================================================
% Slide 22
% Slide Header: Shift Example
% ============================================================
\begin{frame}{Shift Example}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Logical Shift Right}
\begin{center}
\large
\code{lsr x0, x1, \#1}
\end{center}
\begin{itemize}
  \item Zero enters the top bit.
  \item Good for unsigned bit patterns.
  \item Does not preserve a negative sign.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Arithmetic Shift Right}
\begin{center}
\large
\code{asr x0, x1, \#1}
\end{center}
\begin{itemize}
  \item Original sign bit is copied.
  \item Preserves signed interpretation.
  \item Useful for signed right shifts.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.25cm}
\begin{block}{Qualification}
Shifts can resemble multiplication or division by powers of two, but finite width, overflow, and signed rounding behavior must be considered.
\end{block}
\end{frame}

% ============================================================
% Slide 23
% Slide Header: Immediate Operands Are Encoded in the Instruction
% ============================================================
\begin{frame}{Immediate Operands Are Encoded in the Instruction}
\begin{block}{Immediate Operand}
An immediate is a constant stored directly inside the instruction encoding.
\end{block}
\vspace{0.2cm}
\large
\hspace{3cm}\code{add x0, x1, \#5}\\[0.15cm]
\hspace{3cm}\code{and x2, x3, \#0xff}
\vspace{0.2cm}
\begin{itemize}
  \item Immediate operands avoid loading a separate constant from memory.
  \item They are limited by the available instruction bits.
  \item AArch64 immediate forms are constrained by instruction class.
\end{itemize}
\vspace{0.1cm}
\begin{block}{Preparation for Encoding}
Do not assume \code{add x0, x1, \#anything} can encode any 64-bit constant. The exact rules are covered later with instruction encoding.
\end{block}
\end{frame}

% ============================================================
% Slide 24
% Slide Header: mov Is Often an Alias
% ============================================================
\begin{frame}{\code{mov} Is Often an Alias}
\begin{block}{Assembler Convenience}
In AArch64, \code{mov} is frequently assembler syntax that maps to different underlying A64 instructions depending on the operands.
\end{block}
\vspace{0.25cm}
\large
\hspace{3cm}\code{mov x0, x1}\\[0.15cm]
\hspace{3cm}\code{mov x0, \#5}\\[0.15cm]
\hspace{3cm}\code{mov x0, xzr}
\vspace{0.25cm}
\begin{itemize}
  \item Students should read \code{mov} as “put this value there.”
  \item The assembler chooses an appropriate encoding.
  \item The exact expansion is an encoding topic, not a semantic mystery.
\end{itemize}
\end{frame}

\begin{frame}{Multiplication Is Data Processing, Not Necessarily Same Latency}
\begin{block}{Architectural Category}
Multiplication is a data-processing operation: it reads registers and writes a register result.
\end{block}
\vspace{0.25cm}
\begin{center}
\large
\code{mul x0, x1, x2}
\end{center}
\vspace{0.25cm}
\begin{itemize}
  \item Architecturally, it belongs with register-based computation.
  \item Microarchitecturally, it may use a separate multiply unit.
  \item Its latency can differ from simple add/sub/logic instructions.
\end{itemize}
\vspace{0.1cm}
\begin{block}{No Fixed Cycle Claim}
AArch64 specifies the instruction's architectural effect, not a universal number of cycles.
\end{block}
\end{frame}

% ============================================================
\begin{frame}{Short Worked Program}
\begin{block}{Goal}
Load a value, test whether it is zero, and add 1 only if it is nonzero.
\end{block}
\vspace{0.2cm}
\begin{center}
\begin{minipage}{0.45\textwidth}
\raggedright
\large
\code{ldr x1, [x0]}\\[0.10cm]
\code{cmp x1, xzr}\\[0.10cm]
\code{b.eq done}\\[0.10cm]
\code{add x1, x1, \#1}\\[0.10cm]
\code{str x1, [x0]}\\[0.10cm]
\code{done:}
\end{minipage}
\end{center}
\vspace{0.15cm}
\begin{block}{Mechanism}
\code{cmp} updates flags using a discarded subtraction. \code{b.eq} branches if the zero flag is set.
\end{block}
\end{frame}

% Slide Header: Topic 5 Summary
% ============================================================
\begin{frame}{Topic 5 Summary}
\begin{itemize}
  \item AArch64 integer arithmetic operates on fixed-width bit patterns.
  \item \code{add} and \code{sub} do not inherently mean signed or unsigned arithmetic.
  \item \code{C} supports unsigned reasoning; \code{V} supports signed overflow reasoning.
  \item After subtraction, \code{C=1} means no unsigned borrow occurred.
  \item \code{cmp} and \code{tst} are aliases that update flags and discard the computed result.
  \item Shifts are useful arithmetic tools, but overflow and signed interpretation matter.
\end{itemize}
\end{frame}

  \item What is the difference between the \code{C} and \code{V} flags?
  \item After \code{cmp x0, x1}, what does \code{C=1} mean for unsigned comparison?
  \item Explain why \code{cmp x0, x1} can be understood as \code{subs xzr, x0, x1}.
  \item Which branch should be used for signed less-than: \code{b.lt} or \code{b.lo}?

\begin{frame}{Topic 5 Summary}
\begin{block}{Core Message}
AArch64 arithmetic and logical instructions operate on fixed-width bit patterns; interpretation depends on the instruction form, the flags, and the surrounding program context.
\end{block}
\vspace{0.15cm}
\begin{itemize}
  \item Arithmetic instructions define architectural state changes.
  \item Flag-setting forms connect arithmetic results to conditional control flow.
  \item Aliases simplify assembly syntax without changing the underlying operation.
  \item Shifts and logical operations manipulate bit patterns directly.
  \item Microarchitectural performance depends on implementation, not only the ISA.
\end{itemize}
\end{frame}
