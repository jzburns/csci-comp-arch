% ============================================================
% Slide 4
% Slide Header: Data Processing in Registers
% ============================================================
\begin{frame}{Data Processing in Registers}
\begin{block}{Load/Store Architecture}
AArch64 arithmetic instructions operate primarily on values already held in registers. Memory access is handled separately by load and store instructions.
\end{block}
\vspace{0.2cm}
\begin{center}
\large
\hspace{0.5cm}\code{add x0, x1, x2}\qquad\code{sub x3, x4, x5}
\end{center}
\vspace{0.2cm}
\begin{itemize}
\item The first operand is the destination register.
\item The remaining operands provide the values used by the operation.
\item The result replaces the previous contents of the destination register.
\item Register operands make the data path regular and predictable.
\end{itemize}
\end{frame}

% ============================================================
% Slide 5
% Slide Header: Addition and Subtraction
% ============================================================
\begin{frame}{Addition and Subtraction}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Addition}
\large
\hspace{1.35cm}\code{add x0, x1, x2}
\normalsize
\begin{itemize}
\item Computes \code{x1 + x2}.
\item Stores the result in \code{x0}.
\item Operands remain unchanged unless reused as the destination.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Subtraction}
\large
\hspace{1.35cm}\code{sub x0, x1, x2}
\normalsize
\begin{itemize}
\item Computes \code{x1 - x2}.
\item Stores the result in \code{x0}.
\item Uses the same fixed-width register datapath.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.2cm}
\begin{block}{Connection to Topic 4}
Flag-setting forms such as \code{adds} and \code{subs} were covered with NZCV in Topic 4. Here the focus is instruction use and operand forms.
\end{block}
\end{frame}

% ============================================================
% Slide 6
% Slide Header: Register and Immediate Arithmetic
% ============================================================
\begin{frame}{Register and Immediate Arithmetic}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Register Operand}
\large
\hspace{1.1cm}\code{add x0, x1, x2}\\[0.12cm]
\hspace{1.1cm}\code{sub x3, x4, x5}
\normalsize
\begin{itemize}
\item Both source values come from registers.
\item Useful when both values were computed earlier.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Immediate Operand}
\large
\hspace{1.0cm}\code{add x0, x1, \#24}\\[0.12cm]
\hspace{1.0cm}\code{sub x3, x4, \#8}
\normalsize
\begin{itemize}
\item One source value is encoded in the instruction.
\item Useful for small fixed constants.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% ============================================================
% Slide 7
% Slide Header: 32-Bit and 64-Bit Arithmetic
% ============================================================
\begin{frame}{32-Bit and 64-Bit Arithmetic}
\begin{block}{Register Width Selects the Operation Width}
The same instruction family is available for 32-bit \code{w} registers and 64-bit \code{x} registers.
\end{block}
\vspace{0.2cm}
\begin{center}
\large
\hspace{0.6cm}\code{add w0, w1, w2}\qquad\code{add x0, x1, x2}
\end{center}
\vspace{0.2cm}
\begin{itemize}
\item \code{w0}--\code{w30} operate on 32-bit values.
\item \code{x0}--\code{x30} operate on 64-bit values.
\item Writing a \code{w} register clears the upper 32 bits of the corresponding \code{x} register.
\item Operand widths must match the instruction form.
\end{itemize}
\end{frame}

% ============================================================
% Slide 8
% Slide Header: Reusing a Source as the Destination
% ============================================================
\begin{frame}{Reusing a Source as the Destination}
\begin{block}{In-Place Update}
The destination register may also be one of the source registers.
\end{block}
\vspace{0.2cm}
\large
\hspace{3.0cm}\code{add x0, x0, \#1}\\[0.12cm]
\hspace{3.0cm}\code{sub x5, x5, x2}
\normalsize
\vspace{0.25cm}
\begin{itemize}
\item The sources are read before the destination value is written.
\item \code{add x0, x0, \#1} increments \code{x0}.
\item \code{sub x5, x5, x2} replaces \code{x5} with \code{x5 - x2}.
\item This form is common in counters and address calculations.
\end{itemize}
\end{frame}

% ============================================================
% Slide 9
% Slide Header: Multiplication
% ============================================================
\begin{frame}{Multiplication}
\begin{block}{Basic Multiply}
\large
\hspace{3.7cm}\code{mul x0, x1, x2}
\normalsize
\end{block}
\vspace{0.2cm}
\begin{itemize}
\item Multiplies the two source registers and writes the low 64 bits to the destination.
\item The 32-bit form uses \code{w} registers and writes a 32-bit result.
\item Multiplication is architecturally a register data-processing operation.
\item Its execution latency is a microarchitectural property, not fixed by the ISA.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Example}
If \code{x1 = 6} and \code{x2 = 7}, then \code{mul x0, x1, x2} leaves \code{x0 = 42}.
\end{block}
\end{frame}

% ============================================================
% Slide 10
% Slide Header: Signed and Unsigned Division
% ============================================================
\begin{frame}{Signed and Unsigned Division}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Unsigned Division}
\large
\hspace{1.2cm}\code{udiv x0, x1, x2}
\normalsize
\begin{itemize}
\item Interprets the operands as unsigned integers.
\item Quotient is truncated toward zero.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Signed Division}
\large
\hspace{1.2cm}\code{sdiv x0, x1, x2}
\normalsize
\begin{itemize}
\item Interprets the operands as signed two's-complement integers.
\item Quotient is truncated toward zero.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.2cm}
\begin{block}{Why Two Instructions?}
Unlike \code{add} and \code{sub}, division must know whether the operands are signed or unsigned because the quotient depends on that interpretation.
\end{block}
\end{frame}

% ============================================================
% Slide 13
% Slide Header: Bitwise Operations
% ============================================================
\begin{frame}{Bitwise Operations}
\begin{block}{Operate on Corresponding Bits}
Logical instructions process each bit position independently. Signed and unsigned numeric interpretation is irrelevant to the bitwise operation itself.
\end{block}
\vspace{0.2cm}
\begin{center}
\large
\code{and x0, x1, x2}\qquad
\code{orr x0, x1, x2}\qquad
\code{eor x0, x1, x2}
\end{center}
\vspace{0.2cm}
\begin{itemize}
\item \code{and}: output bit is 1 only when both input bits are 1.
\item \code{orr}: output bit is 1 when either input bit is 1.
\item \code{eor}: output bit is 1 when the input bits differ.
\end{itemize}
\end{frame}

% ============================================================
% Slide 14
% Slide Header: AND for Selecting Bits
% ============================================================
\begin{frame}{AND for Selecting Bits}
\begin{block}{Masking}
A bit mask uses 1s in positions that should be preserved and 0s in positions that should be cleared.
\end{block}
\vspace{0.15cm}
\begin{center}
\large
\begin{tabular}{r@{\hspace{0.18cm}}l}
 & \code{11001010} \\
\code{\&} & \code{00111100} \\[0.06cm]
\small result & \code{00001000}
\end{tabular}
\normalsize
\end{center}
\vspace{0.15cm}
\begin{block}{AArch64 Example}
\large
\hspace{3.5cm}\code{and x0, x1, x2}
\normalsize
\end{block}
\end{frame}

% ============================================================
% Slide 15
% Slide Header: ORR for Setting Bits
% ============================================================
\begin{frame}{ORR for Setting Bits}
\begin{block}{Setting Selected Bits}
A mask containing 1s can force selected result bits to 1 while leaving other positions unchanged.
\end{block}
\vspace{0.15cm}
\begin{center}
\large
\begin{tabular}{r@{\hspace{0.18cm}}l}
 & \code{10000001} \\
\code{|} & \code{00110000} \\[0.06cm]
\small result & \code{10110001}
\end{tabular}
\normalsize
\end{center}
\vspace{0.15cm}
\begin{block}{AArch64 Example}
\large
\hspace{3.5cm}\code{orr x0, x1, x2}
\normalsize
\end{block}
\end{frame}

% ============================================================
% Slide 16
% Slide Header: EOR for Toggling Bits
% ============================================================
\begin{frame}{EOR for Toggling Bits}
\begin{block}{Toggle Selected Bits}
XOR with 1 flips a bit; XOR with 0 leaves the bit unchanged.
\end{block}
\vspace{0.15cm}
\begin{center}
\large
\begin{tabular}{r@{\hspace{0.18cm}}l}
 & \code{10100110} \\
\code{\^} & \code{00111100} \\[0.06cm]
\small result & \code{10011010}
\end{tabular}
\normalsize
\end{center}
\vspace{0.15cm}
\begin{block}{AArch64 Example}
\large
\hspace{3.5cm}\code{eor x0, x1, x2}
\normalsize
\end{block}
\end{frame}

% Slide Header: Inverted-Operand Logical Forms
% ============================================================
\begin{frame}{Inverted-Operand Logical Forms}
\begin{block}{Why These Instructions Exist}
Some bit-mask operations need one operand inverted before AND, OR, or XOR. These forms combine the inversion with the logical operation.
\end{block}
\vspace{0.12cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.14}
\begin{tabular}{>{\raggedright\arraybackslash}m{2.4cm}>{\raggedright\arraybackslash}m{4.2cm}>{\raggedright\arraybackslash}m{4.4cm}}
\rowcolor{navy}
{\color{white}\bfseries Instruction} & {\color{white}\bfseries Operation} & {\color{white}\bfseries Typical use} \\
\rowcolor{softgray}
\code{bic} & $A \land \sim B$ & clear selected bits \\
\rowcolor{white}
\code{orn} & $A \lor \sim B$ & OR with an inverted operand \\
\rowcolor{softgray}
\code{eon} & $A \oplus \sim B$ & XOR with an inverted operand \\
\end{tabular}
\end{center}
\vspace{0.12cm}
\begin{block}{Example: Clearing Selected Bits}
\large
\hspace{0.6cm}\code{bic x0, x1, x2}
\normalsize
\par
\vspace{0.08cm}
\small
\code{x2} is the mask: 1-bits in \code{x2} clear the corresponding bits of \code{x1}; the result is written to \code{x0}.
\normalsize
\end{block}
\end{frame}

% ============================================================
% Slide 20
% Slide Header: Four Common Shift and Rotate Operations
% ============================================================
\begin{frame}{Four Common Shift and Rotate Operations}
\begin{center}
\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{>{\raggedright\arraybackslash}m{2.1cm}>{\raggedright\arraybackslash}m{4.4cm}>{\raggedright\arraybackslash}m{4.8cm}}
\rowcolor{navy}
{\color{white}\bfseries Instruction} & {\color{white}\bfseries Bit movement} & {\color{white}\bfseries Typical use} \\
\rowcolor{softgray}
\code{lsl} & left; zeros enter on right & scale by powers of two when high bits are not lost \\
\rowcolor{white}
\code{lsr} & right; zeros enter on left & unsigned right shift \\
\rowcolor{softgray}
\code{asr} & right; sign bit replicated & signed right shift \\
\rowcolor{white}
\code{ror} & right; outgoing bits re-enter left & rotate a bit pattern \\
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{block}{Shift Amount}
The shift amount specifies how many bit positions the value moves.
\end{block}
\end{frame}

% ============================================================
% Slide 21
% Slide Header: Logical Shift Left
% ============================================================
\begin{frame}{Logical Shift Left}
\begin{block}{Operation}
\large
\hspace{3.4cm}\code{lsl x0, x1, \#3}
\normalsize
\end{block}
\vspace{0.2cm}
\begin{itemize}
\item Moves every bit in \code{x1} three positions toward the most-significant end.
\item Zeros enter the three vacated low-order bit positions.
\item Bits shifted beyond the register width are discarded.
\item When no significant high bits are lost, shifting left by 3 corresponds to multiplication by 8.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Finite Width}
The register does not become wider during the shift; only the destination-width bits remain.
\end{block}
\end{frame}

% ============================================================
% Slide 22
% Slide Header: Logical and Arithmetic Right Shift
% ============================================================
\begin{frame}{Logical and Arithmetic Right Shift}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Logical Right Shift}
\large
\hspace{1.1cm}\code{lsr x0, x1, \#1}
\normalsize
\begin{itemize}
\item Moves bits right by one.
\item Inserts 0 into the top bit.
\item Appropriate for unsigned bit patterns.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Arithmetic Right Shift}
\large
\hspace{1.1cm}\code{asr x0, x1, \#1}
\normalsize
\begin{itemize}
\item Moves bits right by one.
\item Replicates the original sign bit.
\item Preserves signed two's-complement interpretation.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.15cm}
\begin{block}{Difference}
For values whose top bit is 1, \code{lsr} inserts 0s while \code{asr} inserts 1s.
\end{block}
\end{frame}

% ============================================================
% Slide 23
% Slide Header: Rotate Right
% ============================================================
\begin{frame}{Rotate Right}
\begin{block}{Operation}
\large
\hspace{3.5cm}\code{ror x0, x1, \#8}
\normalsize
\end{block}
\vspace{0.2cm}
\begin{itemize}
\item Bits move toward lower-order positions.
\item Bits leaving the right end re-enter at the left end.
\item No bit positions are replaced by zeros or replicated sign bits.
\item Rotation is therefore different from both logical and arithmetic shifting.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Typical Uses}
Rotations appear in bit manipulation, hashing, cryptographic algorithms, and packed-data transformations.
\end{block}
\end{frame}

% ============================================================
% Slide 24
% Slide Header: Shifted Register Operands
% ============================================================
\begin{frame}{Shifted Register Operands}
\begin{block}{Combine Shift and Operation}
Many AArch64 data-processing register forms can shift one source operand as part of the same instruction.
\end{block}
\vspace{0.2cm}
\large
\hspace{2.4cm}\mbox{\code{add x0, x1, x2, lsl \#3}}\\[0.14cm]
\hspace{2.4cm}\mbox{\code{orr x3, x4, x5, lsr \#8}}
\normalsize
\vspace{0.25cm}
\begin{itemize}
\item The shift applies to the final source register before the main operation.
\item The source register itself is not modified.
\item This can combine scaling and arithmetic or bit manipulation in one architectural instruction.
\end{itemize}
\end{frame}

\sectiondivider{Section 04}{Immediate Values and Constants}

% ============================================================
% Slide 26
% Slide Header: What Is an Immediate?
% ============================================================
\begin{frame}{What Is an Immediate?}
\begin{block}{Constant Inside the Instruction}
An immediate operand is a constant represented within the instruction itself rather than read from a source register.
\end{block}
\vspace{0.2cm}
\large
\hspace{3.2cm}\code{add x0, x1, \#5}\\[0.14cm]
\hspace{3.2cm}\code{sub x2, x2, \#32}
\normalsize
\vspace{0.25cm}
\begin{itemize}
\item The \code{\#} prefix identifies an immediate constant in AArch64 assembly syntax.
\item Immediate forms are convenient for fixed constants known when the program is assembled.
\item The constant must fit the encoding permitted by that particular instruction form.
\end{itemize}
\end{frame}

% ============================================================
% Slide 27
% Slide Header: Immediate Encodings Depend on the Instruction
% ============================================================
\begin{frame}{Immediate Encodings Depend on the Instruction}
\begin{block}{No Universal Immediate Format}
AArch64 does not have one common immediate field that works identically for every instruction class.
\end{block}
\vspace{0.18cm}
\begin{itemize}
\item \code{add} and \code{sub} use an arithmetic-immediate form designed for common small constants.
\item Logical instructions use a specialized bit-mask immediate representation.
\item Shift instructions encode the shift amount according to their operation width.
\item Move-wide instructions place 16-bit pieces into selected portions of a register.
\end{itemize}
\vspace{0.15cm}
\begin{block}{Encoding Boundary}
The exact bit fields are studied later in Topic 9. Here the important point is that immediate availability is instruction-specific.
\end{block}
\end{frame}

% ============================================================
% Slide 28
% Slide Header: Logical Immediate Masks
% ============================================================
\begin{frame}{Logical Immediate Masks}
\begin{block}{Useful but Restricted}
Logical-immediate forms can encode many repeated bit-mask patterns directly, but not every arbitrary 32-bit or 64-bit constant.
\end{block}
\vspace{0.2cm}
\large
\hspace{2.7cm}\code{and x0, x1, \#0xff}\\[0.14cm]
\hspace{2.7cm}\code{orr x2, x2, \#0x100}
\normalsize
\vspace{0.25cm}
\begin{itemize}
\item These forms are especially useful for masking and setting fields.
\item Whether a particular mask is encodable is determined by the logical-immediate encoding rules.
\item If it is not encodable directly, the constant can first be constructed in a register.
\end{itemize}
\end{frame}

% ============================================================
% Slide 29
% Slide Header: Building Constants with MOVZ and MOVK
% ============================================================
\begin{frame}{Building Constants with \code{movz} and \code{movk}}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{\code{movz}}
Moves a 16-bit immediate into a selected 16-bit position and sets the remaining register bits to zero.
\vspace{0.15cm}
\large
\hspace{0.8cm}\code{movz x0, \#0x1234}
\normalsize
\end{block}
\column{0.48\textwidth}
\begin{block}{\code{movk}}
Moves a 16-bit immediate into a selected 16-bit position while keeping the other bits unchanged.
\vspace{0.15cm}
\large
\hspace{0.65cm}\code{movk x0, \#0xabcd, lsl \#16}
\normalsize
\end{block}
\end{columns}
\vspace{0.2cm}
\begin{block}{Purpose}
Several move-wide instructions can assemble a full 64-bit constant from 16-bit pieces without loading it from memory.
\end{block}
\end{frame}

% ============================================================
% Slide 30
% Slide Header: MOVN and MOV
% ============================================================
\begin{frame}{\code{movn} and \code{mov}}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{\code{movn}}
\begin{itemize}
\item Moves the bitwise NOT of a 16-bit immediate into a selected position.
\item Other bits are set consistently with the move-wide operation.
\item Can efficiently construct constants containing many 1s.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{\code{mov}}
\begin{itemize}
\item Common assembler syntax for putting a value into a register.
\item Depending on the operands, the assembler selects an appropriate underlying instruction form.
\item It should not be assumed to correspond to one universal hardware encoding.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.2cm}
\large
\hspace{2.3cm}\code{mov x0, x1}\qquad\code{mov x2, \#42}
\normalsize
\end{frame}

% ============================================================
% Slide 31
% Slide Header: Constant Construction Example
% ============================================================
\begin{frame}{Constant Construction Example}
\begin{block}{Construct \hex{123456789abcdef0}}
A 64-bit constant can be built as four 16-bit pieces.
\end{block}
\vspace{0.15cm}
\large
\hspace{2.8cm}\mbox{\code{movz x0, \#0xdef0}}\\[0.10cm]
\hspace{2.8cm}\mbox{\code{movk x0, \#0x9abc, lsl \#16}}\\[0.10cm]
\hspace{2.8cm}\mbox{\code{movk x0, \#0x5678, lsl \#32}}\\[0.10cm]
\hspace{2.8cm}\mbox{\code{movk x0, \#0x1234, lsl \#48}}
\normalsize
\vspace{0.2cm}
\begin{block}{Result}
After the sequence, \code{x0} contains \hex{123456789abcdef0}.
\end{block}
\end{frame}

% ============================================================
% Slide 32
% Slide Header: Choosing Between an Immediate and a Register
% ============================================================
\begin{frame}{Choosing Between an Immediate and a Register}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Use an Immediate When}
\begin{itemize}
\item the constant is known at assembly time,
\item it fits the instruction's immediate form, and
\item no register value is required for reuse elsewhere.
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{Use a Register When}
\begin{itemize}
\item the value is computed at runtime,
\item the constant does not fit the immediate form, or
\item the value will be reused by several instructions.
\end{itemize}
\end{block}
\end{columns}
\vspace{0.2cm}
\begin{block}{Compiler Role}
Compilers and assemblers choose among these forms to express the required operation using legal A64 encodings.
\end{block}
\end{frame}

% ============================================================
% Slide 33
% Slide Header: Combined Data-Processing Example
% ============================================================
\begin{frame}{Combined Data-Processing Example}
\begin{block}{Goal}
Scale a value by 8, add it to a base value, then keep only the low byte.
\end{block}
\vspace{0.2cm}
\large
\hspace{2.8cm}\mbox{\code{add x0, x1, x2, lsl \#3}}\\[0.14cm]
\hspace{2.8cm}\mbox{\code{and x0, x0, \#0xff}}
\normalsize
\vspace{0.25cm}
\begin{itemize}
\item The first instruction shifts \code{x2} left by 3 and adds it to \code{x1}.
\item The second instruction masks the result to retain only bits 7--0.
\item Both operations remain entirely within the register data-processing model.
\end{itemize}
\end{frame}

\begin{frame}{Topic 5 Summary}
\begin{itemize}
\item \code{add}, \code{sub}, \code{mul}, \code{sdiv}, and \code{udiv} provide the main arithmetic operations introduced here.
\item \code{and}, \code{orr}, \code{eor}, and \code{bic} manipulate selected bit positions.
\item \code{lsl}, \code{lsr}, \code{asr}, and \code{ror} rearrange bits in different ways.
\item Many AArch64 data-processing instructions can combine a shifted register operand with the main operation.
\item Immediate constants are encoded according to instruction-specific rules.
\item \code{movz}, \code{movk}, and \code{movn} help construct constants that do not fit a simple immediate form.
\end{itemize}
\end{frame}
